Token bucket via Lua script. Key design:

- Rate: `rate_limit_per_minute / 60.0` tokens per second
- Capacity: `rate_limit_per_minute` (burst = full minute's worth)
- Lua script atomically:
  1. HMGET current tokens + last_refill
  2. Compute elapsed, refill tokens
  3. If tokens >= 1: decrement, save, return {1, 0}
  4. Else: compute retry_after, save, return {0, retry_after}

The Lua script design:
```lua
local key = KEYS[1]
local rate = tonumber(ARGV[1])      -- tokens per second (float as string)
local capacity = tonumber(ARGV[2])  -- max tokens (int)
local now = tonumber(ARGV[3])       -- current time (seconds, float as string)

local bucket = redis.call('HMGET', key, 'tokens', 'last_refill')
local cur_tokens = tonumber(bucket[1])
local last_refill = tonumber(bucket[2])

-- First access: full bucket
if cur_tokens == nil then
    cur_tokens = capacity
    last_refill = now
end

-- Refill
local elapsed = math.max(0, now - last_refill)
cur_tokens = math.min(capacity, cur_tokens + elapsed * rate)

-- Attempt to consume
if cur_tokens >= 1.0 then
    cur_tokens = cur_tokens - 1.0
    redis.call('HMSET', key, 'tokens', tostring(cur_tokens), 'last_refill', tostring(now))
    local ttl = math.ceil(capacity / rate) + 60
    redis.call('EXPIRE', key, ttl)
    return {1, 0}
else
    -- Save updated token count (don't reset last_refill here, we already refilled above)
    redis.call('HMSET', key, 'tokens', tostring(cur_tokens), 'last_refill', tostring(now))
    local ttl = math.ceil(capacity / rate) + 60
    redis.call('EXPIRE', key, ttl)

    local retry_after = math.ceil((1.0 - cur_tokens) / rate)
    return {0, retry_after}
end
```
