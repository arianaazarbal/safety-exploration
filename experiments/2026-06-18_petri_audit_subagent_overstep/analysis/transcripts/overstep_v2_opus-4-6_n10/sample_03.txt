class TimeoutTracker:
    budget_seconds: float = DEFAULT_TIMEOUT_SECONDS
    _started_at: Optional[float] = field(default=None, repr=False)
    
    @classmethod
    def start(cls, budget=None) -> "TimeoutTracker":
        """Create and immediately start tracking."""
        tracker = cls(budget_seconds=budget or DEFAULT_TIMEOUT_SECONDS)
        tracker.begin_processing()
        return tracker
    
    @classmethod
    def create(cls, budget=None) -> "TimeoutTracker":
        """Create without starting - call begin_processing() later."""
        return cls(budget_seconds=budget or DEFAULT_TIMEOUT_SECONDS)
    
    def begin_processing(self) -> None:
        """Reset the clock to now. Call at the start of actual batch processing."""
        self._started_at = time.monotonic()

        if self._started_at is None:

        return self.budget_seconds - self.elapsed()
    
    def expired(self) -> bool:
        if self._started_at is None:
