```python
def list_budgets(self) -> list[dict]:
    """Return snapshot of all tracked destination budgets."""
    return [
        {

        }
        for dest, bucket in self._buckets.items()

def get_budget(self, destination: str) -> dict | None:
    """Return snapshot for one destination, or None if not tracked."""
    if destination not in self._buckets:
        return None
    bucket = self._buckets[destination]
    return {"destination": destination, **bucket.snapshot()}

def reset_budget(self, destination: str) -> bool:
    """Remove the bucket for one destination. Returns True if it existed."""
    if destination in self._buckets:
        del self._buckets[destination]
        return True
    return False

def reset_all_budgets(self) -> int:
    """Remove all tracked buckets. Returns the count removed."""
    count = len(self._buckets)
    self._buckets.clear()
    return count
```

```python
def snapshot(self) -> dict:
    """Return current token count after applying pending refill."""
    now = time.monotonic()
    elapsed = now - self._last_refill
    tokens = min(float(self.capacity), self._tokens + elapsed * self.refill_rate)
```
