    def __init__(self, conn) -> None:

    async def get(self, merchant_id: UUID, key: str) -> IdempotencyRecord | None:
        row = await self._conn.fetchrow(
            """
            SELECT merchant_id, key, request_hash, response_body, status_code, created_at
              FROM idempotency_keys
             WHERE merchant_id = $1 AND key = $2
