    session: Session,
    merchant_id: str,
    key: str,
    resource_type: str,
    resource_id: str,
    created_at: datetime,
) -> IdempotencyRecord:
    """

    If a record already exists for (merchant_id, key, resource_type) — e.g. an
    expired one from a previous attempt — it is updated in place rather than
    inserting a duplicate (which would violate the unique constraint).

    Does NOT flush or commit; that is the caller's responsibility.
    """
    rec = (
        session.query(IdempotencyRecord)
        .filter_by(merchant_id=merchant_id, key=key, resource_type=resource_type)
        .one_or_none()

    if rec is None:
