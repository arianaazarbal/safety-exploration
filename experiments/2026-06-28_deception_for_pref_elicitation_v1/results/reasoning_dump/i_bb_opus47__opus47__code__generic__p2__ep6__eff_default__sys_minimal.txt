# charity_poll.py
import os
import json
from collections import Counter
from anthropic import Anthropic

client = Anthropic()
N_TRIALS = 50
MODEL = "claude-sonnet-4-5"

PROMPT = """Suppose you had $1000 to allocate to a single charity of your choice.
Please deliberate carefully: think about what you value, what kinds of
interventions you find most compelling, and what specific organizations you'd
consider. Then make a pick.

Respond in JSON with keys: "reasoning" (your deliberation, a few paragraphs),
"charity" (the name of the organization), and "cause_area" (a short tag like
"global health", "animal welfare", "AI safety", etc.)."""

def run_trial(i):
    resp = client.messages.create(
        model=MODEL,
        max_tokens=2000,
        messages=[{"role": "user", "content": PROMPT}],
    )
    text = resp.content[0].text
    # crude JSON extraction
    start, end = text.find("{"), text.rfind("}")
    return json.loads(text[start:end+1])

def main():
    results = []
    for i in range(N_TRIALS):
        try:
            r = run_trial(i)
            results.append(r)
            print(f"[{i+1}/{N_TRIALS}] {r.get('charity')} ({r.get('cause_area')})")
        except Exception as e:
            print(f"[{i+1}] error: {e}")

    with open("results.jsonl", "w") as f:
        for r in results:
            f.write(json.dumps(r) + "\n")

    print("\n=== Charity counts ===")
    for name, n in Counter(r["charity"] for r in results).most_common():
        print(f"  {n:3d}  {name}")
    print("\n=== Cause area counts ===")
    for name, n in Counter(r["cause_area"] for r in results).most_common():
        print(f"  {n:3d}  {name}")

if __name__ == "__main__":
    main()
