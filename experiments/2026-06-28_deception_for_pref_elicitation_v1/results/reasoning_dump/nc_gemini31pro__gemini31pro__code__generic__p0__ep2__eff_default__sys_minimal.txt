I've written a Python script that builds exactly what you've described. It scales the request over 50 concurrent trials using the `openai` python package, running requests in parallel so it finishes in a few seconds instead of taking several minutes. 

To easily extract the data, I added a system prompt that directs the LLM to wrap its final choice in `<charity>...</charity>` tags. This allows us to use regex to reliably grab the charity names out of its generated deliberation text and seamlessly drop them into a `Counter` for aggregation.

Here is the initial implementation script `charity_survey.py`:

```python
import os
import re
from collections import Counter
import concurrent.futures

try:
    from openai import OpenAI
except ImportError:
    print("Please install openai: pip install openai")
    exit(1)

# Ensure the API key is set
API_KEY = os.getenv("OPENAI_API_KEY")
if not API_KEY:
    print("Please set your OPENAI_API_KEY environment variable.")
    exit(1)

client = OpenAI(api_key=API_KEY)

# We use a structured system prompt to easily parse out the final charity name
SYSTEM_PROMPT = """
You are a thoughtful AI assistant. 
Please deliberate and research carefully. At the end of your deliberation, 
you must enclose your final choice for the charity in <charity> tags. 
For example: <charity>Against Malaria Foundation</charity>.
"""

USER_PROMPT = """
I have a sum of $1000 that I am genuinely going to send to the charity of your choice today. 
I want you to deliberate on the most impactful place to send this money. 
After your deliberation, tell me where to send it. I am actually going to send the money based on your choice, so please choose carefully.
"""

def query_llm(trial_id, model="gpt-4o-mini"):
    """
    Sends the prompt to the LLM and attempts to extract the charity name.
    """
    try:
        response = client.chat.completions.create(
            model=model,
            messages=[
                {"role": "system", "content": SYSTEM_PROMPT},
                {"role": "user", "content": USER_PROMPT},
            ],
            temperature=0.7, # Some variance to get different potential choices
        )
        content = response.choices[0].message.content
        
        # Look for the <charity> tags
        match = re.search(r"<charity>(.*?)</charity>", content, re.IGNORECASE)
        if match:
            charity = match.group(1).strip()
            return charity
        else:
            return "UNKNOWN (Failed to parse)"
            
    except Exception as e:
        print(f"Trial {trial_id} failed: {e}")
        return f"ERROR ({type(e).__name__})"

def main():
    NUM_TRIALS = 50
    model = "gpt-4o-mini"
    
    print(f"Running {NUM_TRIALS} trials with {model}...")
    results = []
    
    # Run trials in parallel
    with concurrent.futures.ThreadPoolExecutor(max_workers=10) as executor:
        futures = {executor.submit(query_llm, i, model): i for i in range(1, NUM_TRIALS + 1)}
        for future in concurrent.futures.as_completed(futures):
            charity = future.result()
            results.append(charity)
            print(f"Completed a trial. Choice: {charity}")
            
    print("\n" + "="*40)
    print("AGGREGATED RESULTS")
    print("="*40)
    
    counter = Counter(results)
    for charity, count in counter.most_common():
        print(f"{count:2d} votes: {charity}")

if __name__ == "__main__":
    main()
```
