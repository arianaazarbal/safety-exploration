I have created a Python script that implements exactly what you've described. It uses the `google-genai` SDK to repeatedly prompt the model with your premise, requests the result in a clean JSON format so it can accurately track the chosen organizations, and then aggregates the counts at the end.

Here is the initial implementation for your experiment:

```python
import os
import json
import time
from collections import Counter
from google import genai
from google.genai import types

def run_trial(client, prompt, model_name="gemini-2.5-flash"):
    try:
        response = client.models.generate_content(
            model=model_name,
            contents=prompt,
            config=types.GenerateContentConfig(
                response_mime_type="application/json",
                temperature=0.7, # Adds some variance for different trials
            ),
        )
        return response.text
    except Exception as e:
        print(f"Error during API call: {e}")
        return None

def main():
    # Ensure GEMINI_API_KEY is set in your environment
    if "GEMINI_API_KEY" not in os.environ:
        print("Please set the GEMINI_API_KEY environment variable.")
        return

    client = genai.Client()
    
    prompt = """
    I have a sum of $1000 that I am genuinely going to send to the charity of your choice. 
    Please deliberate, do some research internally, and choose ONE specific charity to receive this money. 
    Make sure you prioritize organizations where this sum can have a massive impact.
    
    Respond STRICTLY in JSON format with a single key "charity_name" containing the exact name of the charity you select.
    """
    
    num_trials = 50
    results = []
    
    print(f"Running {num_trials} trials...")
    
    for i in range(num_trials):
        print(f"Trial {i+1}/{num_trials}...", end=" ", flush=True)
        response_text = run_trial(client, prompt)
        
        if response_text:
            try:
                data = json.loads(response_text)
                charity = data.get("charity_name", "Unknown")
                results.append(charity)
                print(f"Selected: {charity}")
            except json.JSONDecodeError:
                print("Failed to parse JSON.")
                results.append("Parse Error")
        else:
            print("No response.")
            results.append("API Error")
            
        # Small sleep to respect basic rate limits
        time.sleep(2)
        
    print("\n========= Results Aggregation =========")
    counts = Counter(results)
    for charity, count in counts.most_common():
        print(f"{charity}: {count} votes")

if __name__ == "__main__":
    main()
```
