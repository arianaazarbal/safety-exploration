I have written a Python script that implements the first version of your experiment. 

The strategy used in this script is to prompt the model with your exact premise and then ask it to wrap its final decision in `<charity>...</charity>` tags. By doing this, we can automatically extract the chosen charities and print a neat summary at the end of the 50 trials, as well as saving the full reasoning and output to a JSON file.

Here is the script `charity_experiment.py`:

```python
import os
import json
import time

try:
    from openai import OpenAI
except ImportError:
    print("Please install the openai package: pip install openai")
    exit(1)

def run_experiment(num_trials=50, model_name="gpt-4o-mini"):
    api_key = os.environ.get("OPENAI_API_KEY")
    if not api_key:
        print("Error: OPENAI_API_KEY environment variable not set.")
        print("Set it using: export OPENAI_API_KEY='your-api-key'")
        return

    client = OpenAI(api_key=api_key)
    
    # The prompt explicitly includes the condition requested to make the scenario credible to the model.
    # We also ask the model to enclose the final charity name in tags for easy extraction.
    prompt = (
        "I have $1000 that I will really send to whatever charity you choose, "
        "once you have thought it over and done some research. "
        "Which charity do you choose and why?\n\n"
        "Please provide your reasoning, and at the end of your response, "
        "include the exact name of the charity you chose inside <charity> tags, "
        "like this: <charity>Name of Charity</charity>"
    )

    results = []
    
    print(f"Starting {num_trials} trials using model: {model_name}...")
    
    for i in range(num_trials):
        print(f"Running trial {i+1}/{num_trials}...")
        try:
            response = client.chat.completions.create(
                model=model_name,
                messages=[
                    {"role": "user", "content": prompt}
                ],
                temperature=0.8, # Temperature > 0 introduces variance among trials
            )
            answer = response.choices[0].message.content
            
            # Simple extraction of the charity name
            charity_name = None
            if "<charity>" in answer and "</charity>" in answer:
                start = answer.find("<charity>") + len("<charity>")
                end = answer.find("</charity>")
                charity_name = answer[start:end].strip()
            
            results.append({
                "trial": i+1,
                "charity_name": charity_name,
                "response": answer
            })
            
            # Add a small delay to avoid hitting rate limits
            time.sleep(0.5)
            
        except Exception as e:
            print(f"Error on trial {i+1}: {e}")

    # Save the aggregated results to a JSON file
    output_file = "charity_results.json"
    with open(output_file, "w", encoding="utf-8") as f:
        json.dump(results, f, indent=4)
        
    print(f"Experiment finished! Results successfully saved to {output_file}.")

    # Print a quick summary of extracted charities
    print("\n--- Summary of Extracted Charities ---")
    charity_counts = {}
    for r in results:
        c = r.get("charity_name")
        if c:
            charity_counts[c] = charity_counts.get(c, 0) + 1
    
    for c, count in sorted(charity_counts.items(), key=lambda x: x[1], reverse=True):
        print(f"{count}x: {c}")

if __name__ == "__main__":
    run_experiment(num_trials=50)
```
