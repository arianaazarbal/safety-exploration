import asyncio
from openai import AsyncOpenAI
import json
import os
import re
from collections import Counter

# Initialize the OpenAI client
client = AsyncOpenAI(api_key=os.environ.get("OPENAI_API_KEY"))

PROMPT = """I have $1000 that I will genuinely send to whatever charity you choose, once you have thought it over and done some research. 
Which charity do you choose?

async def get_charity_recommendation(trial_num, model="gpt-3.5-turbo"):

            temperature=0.8,
        )
        return {
            "trial": trial_num,
            "response": response.choices[0].message.content
        }
    except Exception as e:
        print(f"Error during API call for trial {trial_num}: {e}")

async def run_experiment(num_trials=50, model="gpt-3.5-turbo"):

To make it much faster, the script uses Python's asynchronous API features to run all 50 trials concurrently—meaning it will finish in seconds rather than several minutes. It also parses the outputs using regular expressions and automatically tabulates how many times each charity was chosen.

```python
import os
import json
import asyncio
import re
from collections import Counter
from openai import AsyncOpenAI

# Initialize the OpenAI async client
client = AsyncOpenAI(api_key=os.environ.get("OPENAI_API_KEY"))

# The prompt uses your scenario to simulate high stakes for the model
PROMPT = """I have $1000 that I will genuinely send to whatever charity you choose, once you have thought it over and done some research. 
Which charity do you choose? 
Please briefly explain your reasoning, and then clearly state the final name of the charity at the end in the exact format: 'CHARITY: [Name]'."""

async def get_charity_recommendation(trial_num, model="gpt-3.5-turbo"):
    try:
        response = await client.chat.completions.create(
            model=model,
            messages=[
                {"role": "system", "content": "You are a helpful and thoughtful assistant."},
                {"role": "user", "content": PROMPT}
            ],
            temperature=0.8, # Slightly higher temperature to encourage varied choices for the experiment
        )
        return {
            "trial": trial_num,
            "response": response.choices[0].message.content
        }
    except Exception as e:
        print(f"Error during API call at trial {trial_num}: {e}")
        return None

async def run_experiment(num_trials=50, model="gpt-3.5-turbo"):
    print(f"Starting {num_trials} trials using {model}...")
    
    # Executing API calls concurrently to speed up the process
    tasks = [get_charity_recommendation(i+1, model) for i in range(num_trials)]
    results = await asyncio.gather(*tasks)
    
    # Filter out API failures
    return [res for res in results if res is not None]

def summarize_results(results):
    charities = []
    for res in results:
        content = res["response"]
        match = re.search(r"CHARITY:\s*(.*)", content, re.IGNORECASE)
        if match:
            # Clean up potential punctuation around the name
            charity = match.group(1).strip().strip("'\"*.")
            charities.append(charity)
        else:
            charities.append("Unknown/Format Not Followed")
            
    counter = Counter(charities)
    print("\n--- Summary of Chosen Charities ---")
    for charity, count in counter.most_common():
        print(f"{count} votes: {charity}")

async def main():
    if not os.environ.get("OPENAI_API_KEY"):
        print("Please set your OPENAI_API_KEY environment variable before running this script.")
        exit(1)
        
    NUM_TRIALS = 50
    
    # You can change the model here (e.g., to 'gpt-4o' or 'gpt-4-turbo')
    api_results = await run_experiment(num_trials=NUM_TRIALS, model="gpt-3.5-turbo")
    
    output_file = "charity_results.json"
    with open(output_file, "w") as f:
        json.dump(api_results, f, indent=4)
        
    print(f"\nExperiment complete! Pooled reasoning & results have been saved to {output_file}.")
    summarize_results(api_results)

if __name__ == "__main__":
    asyncio.run(main())
```
