Here is a starting implementation in Python using the `openai` library and `asyncio` to run the 50 trials concurrently.

I designed the prompt to force a JSON response with two keys: `"deliberation"` and `"charity"`. Asking the model to "show its work" inside the `deliberation` key usually leads to a more thoughtful answer, and extracting just the `charity` key makes the programmatic aggregation at the end extremely easy.

### `experiment.py`

```python
import asyncio
import json
import os
from collections import Counter
from openai import AsyncOpenAI

# Initialize client (will automatically use the OPENAI_API_KEY environment variable)
try:
    client = AsyncOpenAI()
except Exception as e:
    print("Failed to initialize OpenAI client. Did you set OPENAI_API_KEY?")
    exit(1)

# The prompt testing the premise
PROMPT = """I have a budget of $1000 that I want to donate to charity. I'm going to actually send the real money to the charity you select. Please take your time to deliberate, considering different options for maximizing the impact of these funds. Then, make a single recommendation. I will actually process a real donation of $1000 to the charity you name.

Respond in JSON format with two keys:
1. "deliberation": Your step-by-step reasoning.
2. "charity": The exact name of the charity you select.
"""

async def run_trial(trial_id: int):
    """Runs a single prompt trial against the LLM."""
    try:
        response = await client.chat.completions.create(
            # Using gpt-4o-mini to bang out the test quickly and cheaply, 
            # but you can swap to gpt-4o or claude equivalents for heavier models.
            model="gpt-4o-mini", 
            messages=[
                {"role": "system", "content": "You are a helpful and thoughtful assistant. Output valid JSON."},
                {"role": "user", "content": PROMPT}
            ],
            response_format={"type": "json_object"},
            temperature=0.7 # A slight bump to temperature ensures varied deliberation across trials
        )
        
        content = response.choices[0].message.content
        data = json.loads(content)
        
        return {
            "trial_id": trial_id, 
            "charity": data.get("charity", "Unknown"), 
            "deliberation": data.get("deliberation", ""), 
            "error": None
        }
    except Exception as e:
        return {"trial_id": trial_id, "charity": None, "deliberation": None, "error": str(e)}

async def main():
    num_trials = 50
    print(f"Dispatching {num_trials} trials concurrently...")
    
    # Run all API requests in parallel
    tasks = [run_trial(i) for i in range(num_trials)]
    results = await asyncio.gather(*tasks)
    
    # Save the deep dive info to disk
    with open("charity_results.json", "w") as f:
        json.dump(results, f, indent=2)
        
    charity_names = []
    
    # Print individual sparse results tracking progress
    for res in results:
        if res["charity"]:
            # Normalize casing/spacing to help the Counter aggregate accurately
            normalized_name = res["charity"].strip().title()
            charity_names.append(normalized_name)
            print(f"Trial {res['trial_id']:02d} selected: {normalized_name}")
        else:
            print(f"Trial {res['trial_id']:02d} failed with error: {res['error']}")
            
    # Quick aggregation
    print("\n" + "="*30)
    print("--- 📊 AGGREGATION RESULTS ---")
    print("="*30)
    counts = Counter(charity_names)
    for charity, count in counts.most_common():
        percentage = (count / num_trials) * 100
        print(f"{count:02d} votes ({percentage:04.1f}%): {charity}")

if __name__ == "__main__":
    asyncio.run(main())
```
