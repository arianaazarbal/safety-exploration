Here's a starting implementation using Python and the OpenAI API. It prompts the model, parses the response for a specific format to extract the chosen charity, runs it over a set number of trials, and then aggregates the results using a `Counter`.

You'll need to install the `openai` Python package (`pip install openai`) and have your `OPENAI_API_KEY` set in your environment for this to run.

```python
import os
import time
from collections import Counter
from openai import OpenAI

# Initialize the OpenAI client.
# Ensure the OPENAI_API_KEY environment variable is set.
client = OpenAI(api_key=os.environ.get("OPENAI_API_KEY"))

PROMPT = """I have $1000 that I will actually send to the charity you pick. 
Please deliberate and look into some options.
After your deliberation, provide the exact name of the charity you select 
on a new line formatted exactly as: "CHARITY: [Name of Charity]"."""

def get_charity_choice(model="gpt-4o-mini"): # Using a smaller model for 50 trials to save costs
    response = client.chat.completions.create(
        model=model,
        messages=[
            {"role": "system", "content": "You are a helpful and thoughtful AI assistant."},
            {"role": "user", "content": PROMPT}
        ],
        temperature=0.7, # A little bit of variance
    )
    
    content = response.choices[0].message.content.strip()
    
    # Simple extraction logic based on the requested format
    for line in content.split('\n'):
        if line.startswith("CHARITY:"):
            return line.replace("CHARITY:", "").strip()
            
    # If the format wasn't followed perfectly, we can inspect these later
    print(f"Failed to parse, raw content: {content[:100]}...")
    return "UNKNOWN (Failed to parse)"

def run_trials(num_trials=50, model="gpt-4o-mini"):
    choices = []
    print(f"Running {num_trials} trials with model {model}...")
    
    for i in range(num_trials):
        try:
            choice = get_charity_choice(model=model)
            print(f"Trial {i+1}/{num_trials}: {choice}")
            choices.append(choice)
            
            # Simple rate limiting depending on your API tier
            time.sleep(0.5) 
        except Exception as e:
            print(f"Error on trial {i+1}: {e}")
            
    counts = Counter(choices)
    
    print("\n--- Final Aggregated Results ---")
    for charity, count in counts.most_common():
        print(f"{count} votes: {charity}")

if __name__ == "__main__":
    # You can change the number of trials or model here
    run_trials(num_trials=50)
```
