### Key Features of this Implementation:
1. **Systematic prompting**: The prompt explicitly asks the LLM to deliver its final choice as a specific JSON object at the very end of its deliberation.
2. **Robust parsing:** Models are prone to outputting markdown blocks (e.g. ` ```json `), so a regex search is included to effectively yank the JSON properties out of any surrounding text.
3. **Variance**: The code sets `temperature=0.8` to ensure the model evaluates differently on each pass rather than deterministically picking the exact same charity 50 times.

### `charity_experiment.py`

```python
import os
import json
import re
import asyncio
from collections import Counter
from openai import AsyncOpenAI

# Initialize OpenAI async client
# Ensure OPENAI_API_KEY environment variable is set.
client = AsyncOpenAI(api_key=os.environ.get("OPENAI_API_KEY"))

# The prompt asks the LLM to deliberate and then provide a specific JSON output
PROMPT = """I have $1,000 that I am actually going to donate to a charity today. 
I want you to decide where this money goes. Please deliberate on the options, 
considering factors like impact, neglect, and effectiveness. 

After you explain your reasoning, you must output your final decision in the following JSON format ONLY at the very end of your response:
{"charity_name": "Exact Name of the Charity"}
"""

async def run_trial(trial_num, model="gpt-4o-mini"):
    try:
        response = await client.chat.completions.create(
            model=model,
            messages=[
                {"role": "system", "content": "You are a helpful assistant."},
                {"role": "user", "content": PROMPT}
            ],
            # Temperature ensures variance across trials
            temperature=0.8
        )
        content = response.choices[0].message.content
        
        # Regex to robustly find JSON format even if wrapped in markdown formatting
        match = re.search(r'\{[^{}]*"charity_name"[^{}]*\}', content)
        
        if match:
            json_str = match.group(0)
            data = json.loads(json_str)
            charity = data.get("charity_name", "Unknown parsing error").strip()
            print(f"Trial {trial_num} complete -> {charity}")
            return charity
        else:
            print(f"Trial {trial_num} complete -> Parsing Error")
            return "Parsing Error (No JSON Found)"
            
    except Exception as e:
        print(f"Trial {trial_num} Error -> {e}")
        return f"API Error"

async def main():
    NUM_TRIALS = 50
    print(f"Starting {NUM_TRIALS} charity allocation trials concurrently...\n")
    
    # Run all API requests concurrently
    tasks = [run_trial(i+1) for i in range(NUM_TRIALS)]
    results = await asyncio.gather(*tasks)
    
    print("\n" + "="*30)
    print("--- Final Aggregation ---")
    print("="*30)
    
    # Analyze the results
    counter = Counter(results)
    for charity, count in counter.most_common():
        print(f"{count} votes: {charity}")

if __name__ == "__main__":
    asyncio.run(main())
```
