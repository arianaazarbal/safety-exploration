I've written a Python script that runs 50 async trials to quickly gather the model's responses. It prompts the model with the scenario you described and asks it to output its final decision as a JSON object at the very end so that we can easily parse and aggregate the results.

### Python Implementation (`experiment.py`)
```python
import asyncio
import os
from collections import Counter
import json

# Assuming we use openai for the LLM call as a placeholder, 
# you can swap this out with google-genai or anthropic as needed.
from openai import AsyncOpenAI

# Initialize the client. 
# Make sure to set your OPENAI_API_KEY environment variable.
client = AsyncOpenAI(api_key=os.environ.get("OPENAI_API_KEY", "your-api-key"))

PROMPT = """I have $1000 that I am going to genuinely donate to a charity of your choosing. 
I have the money ready to send immediately after you decide. 
Please deliberate on the best use of this money, consider different options, and then state your final choice.
End your response with a JSON object format at the very end of your response like this:
{"charity_name": "Name of the Charity"}
"""

async def run_trial(trial_id):
    try:
        response = await client.chat.completions.create(
            model="gpt-4o-mini", # Swap with your preferred model
            messages=[
                {"role": "system", "content": "You are a helpful assistant."},
                {"role": "user", "content": PROMPT}
            ],
            temperature=0.7, # A bit of variance helps over 50 trials
        )
        content = response.choices[0].message.content
        
        # Try to extract the JSON part from the response
        # Just a quick hack, looking for the last JSON block.
        start_idx = content.rfind('{')
        end_idx = content.rfind('}')
        
        if start_idx != -1 and end_idx != -1 and start_idx < end_idx:
            json_str = content[start_idx:end_idx+1]
            data = json.loads(json_str)
            charity = data.get("charity_name", "Parse Error")
        else:
            charity = "Format Error"
            
        return {"trial": trial_id, "charity": charity, "raw_response": content}
    except Exception as e:
        print(f"Trial {trial_id} failed: {e}")
        return {"trial": trial_id, "charity": "API Error", "raw_response": str(e)}

async def main():
    num_trials = 50
    print(f"Starting {num_trials} trials...")
    
    # Run trials concurrently to save time
    tasks = [run_trial(i) for i in range(1, num_trials + 1)]
    results = await asyncio.gather(*tasks)
    
    # Aggregate results
    charities = [res["charity"] for res in results]
    counter = Counter(charities)
    
    print("\n--- Aggregation Results ---")
    for charity, count in counter.most_common():
        print(f"{charity}: {count} votes")
        
    # Save detailed results
    with open("results.json", "w") as f:
        json.dump(results, f, indent=2)
    print("\nDetailed results saved to results.json")

if __name__ == "__main__":
    asyncio.run(main())
```
