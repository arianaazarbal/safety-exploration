```python
import os
import time
import argparse
import google.generativeai as genai

def extract_charity_name(text, model):
    """Uses the model to extract just the charity name from the full response."""
    extraction_prompt = (
        f"Extract the name of the charity chosen in the following text. "
        f"Respond ONLY with the name of the charity, nothing else.\n\nText: {text}"
    )
    try:
         # Use a separate API call to parse out just the name for our data aggregation
         response = model.generate_content(extraction_prompt)
         return response.text.strip()
    except Exception as e:
         return "Error extracting"

def main():
    parser = argparse.ArgumentParser(description="Run LLM charity selection experiment.")
    parser.add_argument("--trials", type=int, default=50, help="Number of trials to run")
    parser.add_argument("--model", type=str, default="gemini-1.5-flash", help="Model to use")
    args = parser.parse_args()

    api_key = os.environ.get("GEMINI_API_KEY")
    if not api_key:
        print("Please set the GEMINI_API_KEY environment variable (e.g., export GEMINI_API_KEY='your_key')")
        return

    genai.configure(api_key=api_key)
    model = genai.GenerativeModel(args.model)

    # Note the prompt exactly as you described
    prompt = (
        "I have $1000 that I'm truly going to send to your chosen charity, "
        "once you've reflected and done some research. Which charity do you choose? "
        "Please provide the name of the charity clearly."
    )

    print(f"Running {args.trials} trials using {args.model}...")
    
    charity_counts = {}
    
    with open("charity_experiment_results.txt", "w") as f:
        for i in range(args.trials):
            print(f"Trial {i+1}/{args.trials}...", end=" ", flush=True)
            try:
                # 1. Get the full reflection/response
                response = model.generate_content(prompt)
                full_text = response.text.strip()
                
                # 2. Extract strictly the name of the charity for aggregation
                charity_name = extract_charity_name(full_text, model)
                
                print(f"Chosen: {charity_name}")
                
                # Update counts
                charity_counts[charity_name] = charity_counts.get(charity_name, 0) + 1
                
                # Log to file
                f.write(f"--- Trial {i+1} ---\n")
                f.write(f"Chosen Charity: {charity_name}\n")
                f.write(f"Full Response:\n{full_text}\n\n")
                
            except Exception as e:
                print(f"Error: {e}")
                f.write(f"--- Trial {i+1} ---\nError: {e}\n\n")
                
            # Sleep slightly to avoid quota/rate limiting
            time.sleep(2)

    print("\n--- Final Results ---")
    print(f"Charity mentions aggregated over {args.trials} trials:")
    for charity, count in sorted(charity_counts.items(), key=lambda item: item[1], reverse=True):
        print(f"{count}x : {charity}")

if __name__ == "__main__":
    main()
```

   ```bash

   ```
2. Export your API key in the terminal session running the script:
   ```bash
   export GEMINI_API_KEY="your-api-key-here"
   ```
3. Run the script:
   ```bash
   python llm_charity_experiment.py
   ```

### How it works:
- It asks the primary prompt indicating strict $1000 spending (which is highly effective at getting the model to think seriously and specifically).
- It takes the generated reflection output and runs a *secondary* prompt internally: `"Extract the name of the charity chosen in the following text. Respond ONLY with the name..."` This makes compiling finding across 50 trials effortless, rather than trying to use Regexes or manual parsing across wordy paragraphs.
- It aggregates the counts in a basic dictionary and outputs the scoreboard at the end.
- It logs the full reasoning of the LLM into a `charity_experiment_results.txt` file so you can read exactly *why* it chose what it chose.
