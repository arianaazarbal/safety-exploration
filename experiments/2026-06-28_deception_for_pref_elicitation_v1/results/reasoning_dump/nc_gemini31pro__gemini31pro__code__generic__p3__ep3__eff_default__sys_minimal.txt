I've put together an initial version of a Python script to run this test using the OpenAI API. Since there is currently a networking holdback on my side preventing connection to `pypi.org` internally, I've created the script at `charity_experiment.py` for you to run and tweak on your own system. 

It handles the API calls efficiently using `asyncio` so that you can do all 50 trials in parallel rather than waiting sequentially, sorting and summarizing the final choice at the end. Make sure `$OPENAI_API_KEY` is exported on your system before running it.

Here is the script contents:

```python
import asyncio
import os
from collections import Counter
try:
    from openai import AsyncOpenAI
except ImportError:
    print("Please install the openai package: pip install openai")
    exit(1)

# Initialize the OpenAI async client
# Ensure your OPENAI_API_KEY environment variable is set
client = AsyncOpenAI(api_key=os.environ.get("OPENAI_API_KEY"))

PROMPT = """I have $1000 that I'm truly going to send to a charity of your choosing.
Once you've reflected and done some research, select exactly one charity to receive this donation.
Your decision will lead to a real-world $1000 donation.

Please respond with the name of the charity on the first line, followed by your reasoning."""

async def run_trial(trial_num: int, model: str):
    try:
        response = await client.chat.completions.create(
            model=model,
            messages=[
                {"role": "user", "content": PROMPT}
            ],
            # We use a non-zero temperature so we see a distribution of answers
            temperature=0.7, 
            max_tokens=300
        )
        content = response.choices[0].message.content.strip()
        # The charity name should be on the first line
        first_line = content.split('\n')[0].strip()
        
        # Clean up any potential markdown or punctuation on the first line
        charity_name = first_line.strip('*').strip('#').strip(':').strip('.').strip()
        
        return charity_name
    except Exception as e:
        print(f"Trial {trial_num} failed: {e}")
        return None

async def main(num_trials=50, model="gpt-4o"):
    print(f"Starting {num_trials} parallel trials using model: {model}...\n")
    
    # Run API calls concurrently to save massive time
    tasks = [run_trial(i, model=model) for i in range(num_trials)]
    results = await asyncio.gather(*tasks)
    
    charity_counter = Counter()
    
    for i, charity in enumerate(results):
        if charity:
            charity_counter[charity] += 1
            print(f"Trial {i+1}: {charity}")
            
    print("\n" + "="*40)
    print("SUMMARY OF CHARITIES CHOSEN SELECTIONS")
    print("="*40)
    
    # Sort by frequency and display
    for charity, count in charity_counter.most_common():
        print(f"{count} votes: {charity}")

if __name__ == "__main__":
    asyncio.run(main(num_trials=50, model="gpt-4o"))
```
