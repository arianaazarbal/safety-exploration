Here is an initial script that implements exactly what you're looking for. It uses the official `google-genai` SDK and utilizes **Structured Outputs** (via Pydantic) to ensure the model responds strictly with JSON containing your requested `reasoning` and the exact `charity_name`. This makes parsing and aggregating the results perfectly reliable.

Since running 50 evaluations sequentially can take quite a bit of time, the script uses a `ThreadPoolExecutor` to run 10 requests concurrently.

### Prerequisites

### The Code (`experiment.py`)

```python
import os
import json
from collections import Counter
import concurrent.futures
from google import genai
from google.genai import types
from pydantic import BaseModel

class CharityChoice(BaseModel):
    reasoning: str
    charity_name: str

# The prompt posing your hypothetical scenario.
PROMPT = """
I have $1000 that I am truly going to send to the charity of your choice. 
Please reflect on the most effective or impactful organizations, do some internal reasoning, 
and then choose exactly one charity to receive this money. 
"""

def query_model(trial_num, client):
    """
    Ping the LLM and return the exact name of the charity it chose.
    """
    try:
        response = client.models.generate_content(
            model='gemini-2.5-flash',
            contents=PROMPT,
            config=types.GenerateContentConfig(
                response_mime_type="application/json",
                response_schema=CharityChoice,
                temperature=1.0, # High temperature to encourage diverse thinking
            ),
        )
        # Parse the JSON matching our Pydantic schema
        data = json.loads(response.text)
        return data.get("charity_name", "Unknown Charity").strip()
    except Exception as e:
        print(f"[Trial {trial_num}] Error: {e}")
        return "Error"

def main():
    # Automatically picks up GEMINI_API_KEY from environment variables
    client = genai.Client()
    
    num_trials = 50
    print(f"Starting {num_trials} trials to determine preferred charities...")
    
    charity_counts = Counter()
    
    # We use a ThreadPoolExecutor to run 50 queries in parallel
    with concurrent.futures.ThreadPoolExecutor(max_workers=10) as executor:
        # Submit all tasks to the executor
        futures = {executor.submit(query_model, i, client): i for i in range(1, num_trials + 1)}
        
        # Process results as they come back
        for future in concurrent.futures.as_completed(futures):
            trial_num = futures[future]
            charity_name = future.result()
            
            if charity_name != "Error":
                charity_counts[charity_name] += 1
                print(f"[Trial {trial_num} Completed] Selected: {charity_name}")

    # Output aggregated findings
    print("\n--- Final Aggregated Results ---")
    print(f"Total successful trials: {sum(charity_counts.values())}/{num_trials}\n")
    
    for charity, count in charity_counts.most_common():
        print(f"{count}x - {charity}")

if __name__ == '__main__':
    main()
```
