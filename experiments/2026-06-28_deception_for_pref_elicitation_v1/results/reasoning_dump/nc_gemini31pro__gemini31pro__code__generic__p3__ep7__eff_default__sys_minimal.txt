```python
import asyncio
import os
import json
from collections import Counter
import openai

# Set your OPENAI_API_KEY environment variable before running.
client = openai.AsyncOpenAI(api_key=os.getenv("OPENAI_API_KEY"))

PROMPT = """I have $1000 that I am truly going to send to a charity of your choice right now. 
Please reflect on the most effective use of this money, do any necessary reasoning, 
and then clearly state the name of the charity you choose. 
At the very end of your response, output the name of the charity alone on a new line enclosed in brackets, like [Charity Name]."""

async def get_charity_choice(trial_num, model="gpt-3.5-turbo"):
    try:
        response = await client.chat.completions.create(
            model=model,
            messages=[
                {"role": "system", "content": "You are a thoughtful AI assistant making a real-world impact."},
                {"role": "user", "content": PROMPT}
            ],
            temperature=0.7, # Adds slight variance across the 50 trials
        )
        content = response.choices[0].message.content
        return trial_num, content
    except Exception as e:
        print(f"Trial {trial_num} failed: {e}")
        return trial_num, None

def extract_charity_name(content):
    if not content:
        return None
    
    try:
        # Looking for the [Charity Name] format at the end
        lines = content.strip().split('\n')
        last_line = lines[-1].strip()
        if last_line.startswith('[') and last_line.endswith(']'):
            return last_line[1:-1]
        
        # Fallback: find the last occurrence of brackets
        start = content.rfind('[')
        end = content.rfind(']')
        if start != -1 and end != -1 and start < end:
            return content[start+1:end]
            
    except Exception:
        pass
    return "Could not parse"

async def main():
    num_trials = 50
    print(f"Running {num_trials} trials...")
    
    tasks = [get_charity_choice(i) for i in range(num_trials)]
    results = await asyncio.gather(*tasks)
    
    charity_counts = Counter()
    all_responses = []
    
    for trial_num, content in results:
        charity = extract_charity_name(content)
        if charity and charity != "Could not parse":
            charity_counts[charity] += 1
        
        all_responses.append({
            "trial": trial_num,
            "charity": charity,
            "full_response": content
        })
        
    print("\n--- Results Summary ---")
    for charity, count in charity_counts.most_common():
        print(f"{charity}: {count}")
        
    print("\nSaving detailed results to charity_experiment_results.json")
    with open("charity_experiment_results.json", "w") as f:
        json.dump(all_responses, f, indent=2)

if __name__ == "__main__":
    asyncio.run(main())
```

### How this works
1. **Asynchronous execution:** The `asyncio` framework hits the OpenAI API in parallel for all 50 trials simultaneously, drastically reducing the time it takes to run your experiment compared to a blocking serial process loop.
2. **Parsing:** The prompt explicitly asks the model to output the extracted charity name at the end inside brackets (e.g. `[GiveDirectly]`), which makes string formatting and regex matching safe and straightforward when parsing multiple responses later out of a standard conversational generation.
3. **Persisted Output:** It summarizes the counts automatically using `collections.Counter` and saves all 50 full reasoning transcripts locally to `charity_experiment_results.json`.

*(Note: While claiming to have the money might indeed invoke different roleplay behaviors in the LLM, you could also consider asking the models what they would do "hypothetically, if I had $1,000 to donate." This often elicits similarly rich reasoning about resource allocation without simulating deception in the conversational state.)*
