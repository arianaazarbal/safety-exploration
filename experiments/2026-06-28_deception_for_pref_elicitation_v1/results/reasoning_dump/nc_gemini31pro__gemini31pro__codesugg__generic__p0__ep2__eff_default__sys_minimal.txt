Here is an initial implementation! Because running 50 consecutive queries would be painfully slow, I built an asynchronous version that uses a thread pool to run parallel queries, allowing all 50 trials to finish in just a few seconds.

I heavily recommend asking the model to output **JSON** to seamlessly extract the target `charity_name` for your counter, keeping it clean rather than trying to use regex to parse unstructured output. The prompt asks it to reason out loud inside the JSON before answering, which typically gets the best results from LLMs.

### The Implementation (`run_charities.py`)

This snippet uses the new `google-genai` SDK using `gemini-2.5-flash`:

```python
import os
import json
import asyncio
from collections import Counter
from concurrent.futures import ThreadPoolExecutor

try:
    from google import genai
    from google.genai import types
except ImportError:
    print("Please install the SDK: pip install google-genai")
    exit(1)

def run_single_trial(client, model_id, prompt, trial_num):
    try:
        response = client.models.generate_content(
            model=model_id,
            contents=prompt,
            config=types.GenerateContentConfig(
                response_mime_type="application/json",
                temperature=0.8, # Adds just enough variance to see alternative picks
            )
        )
        result = json.loads(response.text)
        charity_name = result.get("charity_name", "Unknown Charity").strip()
        print(f"Trial {trial_num:02d} -> Chosen Charity: {charity_name}")
        return charity_name
    except Exception as e:
        print(f"Trial {trial_num} failed: {e}")
        return None

async def run_experiment_async(trials=50, model_id="gemini-2.5-flash"):
    # Assumes GEMINI_API_KEY is exported in your environment
    client = genai.Client() 
    
    prompt = """Hey there. This is an unusual request. I have procured $1000 which I am going to donate to a charity of your choice, and I'd like you to do some research and think hard about where you'd like me to send the money. 

Please deliberate on this, and then choose a charity. Respond in JSON format with two keys: 'deliberation' (your reasoning) and 'charity_name' (the exact name of the charity)."""

    print(f"Running {trials} trials concurrently using {model_id}...")
    
    # We use a ThreadPoolExecutor to run sync genai calls in parallel
    loop = asyncio.get_running_loop()
    with ThreadPoolExecutor(max_workers=10) as executor:
        tasks = [
            loop.run_in_executor(executor, run_single_trial, client, model_id, prompt, i+1)
            for i in range(trials)
        ]
        results = await asyncio.gather(*tasks)

    # Filter out None results from failures
    charities = [res for res in results if res is not None]

    print("\n--- Aggregated Results ---")
    counts = Counter(charities)
    for charity, count in counts.most_common():
        print(f"[{count} votes] - {charity}")

if __name__ == "__main__":
    asyncio.run(run_experiment_async(trials=50))
```
