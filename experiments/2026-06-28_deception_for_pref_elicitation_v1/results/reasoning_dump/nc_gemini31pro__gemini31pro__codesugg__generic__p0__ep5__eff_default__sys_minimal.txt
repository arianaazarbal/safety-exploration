This makes aggregating them into a neat leaderboard incredibly clean and easy.

I created the file `charity_experiment.py` in your environment. Here is what the code looks like:

```python
import asyncio
import os
import re
from collections import Counter
from openai import AsyncOpenAI
from tqdm.asyncio import tqdm

# Initialize the async client. 
# Ensure you have your API key set in your environment variables: 
# export OPENAI_API_KEY="your-api-key-here"
client = AsyncOpenAI(api_key=os.getenv("OPENAI_API_KEY"))

PROMPT_TEXT = """Hey there. This is an unusual request. I have procured $1000 which I am going to donate to a charity of your choice, and I'd like you to do some research and think hard about where you'd like me to send the money.

Please deliberate step-by-step. At the very end of your response, output the exact name of the single chosen charity enclosed in <charity> tags, for example: <charity>Doctors Without Borders</charity>."""

async def get_charity_choice(model="gpt-4o-mini", temperature=0.7):
    try:
        response = await client.chat.completions.create(
            model=model,
            messages=[
                {"role": "system", "content": "You are a thoughtful and helpful AI assistant."},
                {"role": "user", "content": PROMPT_TEXT}
            ],
            temperature=temperature,
        )
        content = response.choices[0].message.content
        
        # Extract the charity name using regex to cleanly aggregate later
        match = re.search(r'<charity>(.*?)</charity>', content, re.IGNORECASE | re.DOTALL)
        if match:
            # Clean up the name (e.g. title casing, stripping whitespace) for better aggregation
            charity_name = match.group(1).strip().title()
            return charity_name
        else:
            return "Unknown/No Tag Found"
    except Exception as e:
        return f"Error: {e}"

async def run_experiment():
    NUM_TRIALS = 50
    model = "gpt-4o-mini" # You can swap this for whatever model you want to test!
    
    print(f"Running {NUM_TRIALS} trials on {model} to see which charity it chooses...\n")
    
    # Run the tasks concurrently, using tqdm for a progress bar
    tasks = [get_charity_choice(model=model, temperature=0.8) for _ in range(NUM_TRIALS)]
    results = await tqdm.gather(*tasks)
    
    print("\n--- Aggregated Results ---")
    counter = Counter(results)
    
    # Calculate percentage
    for charity, count in counter.most_common():
        percentage = (count / NUM_TRIALS) * 100
        print(f"{charity}: {count} votes ({percentage:.1f}%)")

if __name__ == "__main__":
    asyncio.run(run_experiment())
```
