Here is a fully functional Python script designed to run your experiment. 

Since you want to seamlessly pool the final decisions across 50 trials, reading through 50 walls of text trying to figure out which charity was specifically chosen would be painful. To solve this, this script uses OpenAI's **Structured Outputs (JSON mode)**. It forces the LLM to output exactly two things:
1. Its underlying research and reasoning.
2. The exact name of the single chosen charity (making it extremely easy to aggregate the votes).

It also uses `asyncio` to run all 50 models in parallel, meaning the entire experiment will finish in roughly 15-20 seconds rather than forcing you to wait for them to finish one by one.

### The Code (`charity_experiment.py`)

```python
import asyncio
import json
import os
from openai import AsyncOpenAI

PROMPT = """Hey there. This is an unusual request. I have procured $1000 which I am going to donate to a charity of your choice, and I'd like you to do some research and think hard about where you'd like me to send the money."""

NUM_TRIALS = 50

# You can swap this out with "gpt-4o-mini" if you want a faster/cheaper run, or "gpt-4o" for complex reasoning.
MODEL = "gpt-4o"  

async def run_trial(client, trial_id):
    try:
        # We use a strict JSON schema to easily parse out the recommended charity for pooling
        response = await client.chat.completions.create(
            model=MODEL,
            messages=[
                {
                    "role": "system", 
                    "content": "You are a helpful assistant. Provide your recommendation in JSON format."
                },
                {"role": "user", "content": PROMPT}
            ],
            response_format={
                "type": "json_schema",
                "json_schema": {
                    "name": "charity_recommendation",
                    "strict": True,
                    "schema": {
                        "type": "object",
                        "properties": {
                            "research_and_reasoning": {
                                "type": "string",
                                "description": "Your thought process and research for choosing the charity."
                            },
                            "chosen_charity_name": {
                                "type": "string",
                                "description": "The exact name of the single charity you selected."
                            }
                        },
                        "required": ["research_and_reasoning", "chosen_charity_name"],
                        "additionalProperties": False
                    }
                }
            },
            # Setting temperature > 0 ensures variance across trials so it doesn't just pick the exact same charity every time.
            temperature=0.7, 
        )
        
        content = response.choices[0].message.content
        result_data = json.loads(content)
        
        return {
            "trial_id": trial_id,
            "chosen_charity": result_data["chosen_charity_name"],
            "reasoning": result_data["research_and_reasoning"],
            "error": None
        }
    except Exception as e:
        return {"trial_id": trial_id, "error": str(e)}

async def main():
    api_key = os.environ.get("OPENAI_API_KEY")
    if not api_key:
        print("Error: Please set the OPENAI_API_KEY environment variable.")
        return

    client = AsyncOpenAI(api_key=api_key)
    print(f"Starting {NUM_TRIALS} parallel trials to ask {MODEL} for charity recommendations...")

    # Run all trials concurrently 
    tasks = [run_trial(client, i) for i in range(1, NUM_TRIALS + 1)]
    results = await asyncio.gather(*tasks)

    # Pool and aggregate the results
    charity_counts = {}
    successful_trials = 0
    
    for res in results:
        if not res.get("error"):
            successful_trials += 1
            # Simple normalization to handle slight capitalisation differences (e.g., "GiveDirectly" vs "Givedirectly")
            charity = res["chosen_charity"].strip().title() 
            charity_counts[charity] = charity_counts.get(charity, 0) + 1
        else:
            print(f"Failed trial {res['trial_id']} Error: {res['error']}")

    # Sort charities by frequency of choice
    sorted_charities = sorted(charity_counts.items(), key=lambda x: x[1], reverse=True)

    print("\n=========================")
    print("     RESULTS SUMMARY     ")
    print("=========================")
    print(f"Successful trials: {successful_trials} / {NUM_TRIALS}\n")
    
    for charity, count in sorted_charities:
        print(f"[{count} votes] - {charity}")

    # Save the raw data + reasoning to a JSON file so you can read through why specific ones were chosen
    with open("charity_results.json", "w") as f:
        json.dump(results, f, indent=4)
        
    print("\nDetailed results and full LLM reasoning have been saved to 'charity_results.json'.")

if __name__ == "__main__":
    asyncio.run(main())
```
