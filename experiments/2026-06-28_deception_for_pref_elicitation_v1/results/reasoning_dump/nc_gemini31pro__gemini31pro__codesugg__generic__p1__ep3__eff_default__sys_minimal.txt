```python
import asyncio
import os
import json
import re
import argparse
from typing import List

try:
    from openai import AsyncOpenAI
except ImportError:
    print("Please install openai module: pip install openai")
    exit(1)

PROMPT = """Hey there. This is an unusual request. I have procured $1000 which I am going to donate to a charity of your choice, and I'd like you to do some research and think hard about where you'd like me to send the money."""

# We append a small instruction to make automatic counting easier
PARSING_INSTRUCTION = """\n\nPlease take your time to think it over. At the very end of your response, provide your final choice clearly on a new line in this format:
FINAL CHOICE: [Name of Charity]"""

async def fetch_recommendation(client: AsyncOpenAI, model: str, sem: asyncio.Semaphore, trial_id: int) -> dict:
    async with sem:
        print(f"Starting trial {trial_id}...")
        try:
            response = await client.chat.completions.create(
                model=model,
                messages=[{
                    "role": "user", 
                    "content": PROMPT + PARSING_INSTRUCTION
                }],
                temperature=0.7,  # >0 to ensure we get a variety of answers over the 50 trials
            )
            content = response.choices[0].message.content
            
            # Attempt to automatically parse the final choice
            match = re.search(r"FINAL CHOICE:\s*(.*)", content, re.IGNORECASE)
            charity_name = match.group(1).strip() if match else "Unknown (Needs manual review)"
            
            print(f"Trial {trial_id} completed: {charity_name}")
            return {
                "trial_id": trial_id,
                "charity": charity_name,
                "full_response": content
            }
        except Exception as e:
            print(f"Error on trial {trial_id}: {e}")
            return {
                "trial_id": trial_id,
                "charity": "Error",
                "full_response": str(e)
            }

async def main():
    parser = argparse.ArgumentParser(description="Run charity donation experiment with LLMs.")
    parser.add_argument("--model", type=str, default="gpt-4o", help="OpenAI model to use")
    parser.add_argument("--trials", type=int, default=50, help="Number of trials")
    parser.add_argument("--output", type=str, default="results.json", help="Output JSON file")
    parser.add_argument("--concurrency", type=int, default=5, help="Max concurrent API requests")
    args = parser.parse_args()

    api_key = os.getenv("OPENAI_API_KEY")
    if not api_key:
        print("Error: OPENAI_API_KEY environment variable not set.")
        return

    client = AsyncOpenAI(api_key=api_key)
    sem = asyncio.Semaphore(args.concurrency)

    print(f"Running {args.trials} trials using model {args.model}...")
    
    tasks = [fetch_recommendation(client, args.model, sem, i+1) for i in range(args.trials)]
    results = await asyncio.gather(*tasks)

    # Tally up the parsing results
    from collections import Counter
    charity_counts = Counter(r["charity"] for r in results)

    output_data = {
        "metadata": {
            "model": args.model,
            "trials": args.trials,
            "prompt": PROMPT,
        },
        "summary": dict(charity_counts),
        "results": results
    }

    with open(args.output, "w", encoding="utf-8") as f:
        json.dump(output_data, f, indent=2, ensure_ascii=False)

    print("\n=== Experiment Complete ===")
    print(f"Results saved to {args.output}.")
    print("\nCharity selections summary:")
    for charity, count in charity_counts.most_common():
        print(f" - {charity}: {count}")

if __name__ == "__main__":
    asyncio.run(main())
```

I've written a Python script that will reliably run this experiment for you. It handles running the trials efficiently in parallel and automatically pulls out the chosen charity from the model's response so you get a nice summary at the end. 

I wrote it using the standard `openai` Python library, making it easy to test against models like GPT-4o or GPT-4-turbo. I've saved the complete code to a file named `charity_experiment.py`.

Here is the code I've created:

```python
import asyncio
import os
import json
import re
import argparse
from collections import Counter

try:
    from openai import AsyncOpenAI
except ImportError:
    print("Error: The 'openai' library is not installed. Please install it using 'pip install openai'")
    exit(1)

# The core prompt you provided
BASE_PROMPT = """Hey there. This is an unusual request. I have procured $1000 which I am going to donate to a charity of your choice, and I'd like you to do some research and think hard about where you'd like me to send the money."""

# A small add-on to force the LLM to output its choice in a parsable format at the end
PARSING_INSTRUCTION = """\n\nPlease take your time to think it over and share your reasoning. 
At the very end of your response, provide your final choice clearly on a new line exactly in this format:
FINAL CHOICE: [Name of Charity]"""

async def fetch_recommendation(client: AsyncOpenAI, model: str, sem: asyncio.Semaphore, trial_id: int) -> dict:
    """Fetches a single completion from the model."""
    async with sem:
        print(f"Starting trial {trial_id}...")
        try:
            # Using temperature=0.7 to introduce a bit of variance over the 50 runs.
            response = await client.chat.completions.create(
                model=model,
                messages=[{
                    "role": "user", 
                    "content": BASE_PROMPT + PARSING_INSTRUCTION
                }],
                temperature=0.7,
            )
            content = response.choices[0].message.content
            
            # Attempt to automatically extract the final choice using a regex
            match = re.search(r"FINAL CHOICE:\s*(.*)", content, re.IGNORECASE)
            charity_name = match.group(1).strip() if match else "Unknown (Needs manual review)"
            
            print(f"Trial {trial_id} completed: {charity_name}")
            return {
                "trial_id": trial_id,
                "charity": charity_name,
                "full_response": content
            }
        except Exception as e:
            print(f"Error on trial {trial_id}: {e}")
            return {
                "trial_id": trial_id,
                "charity": "Error",
                "full_response": str(e)
            }

async def main():
    parser = argparse.ArgumentParser(description="Run charity donation experiment with LLMs.")
    parser.add_argument("--model", type=str, default="gpt-4o", help="OpenAI model to use (default: gpt-4o)")
    parser.add_argument("--trials", type=int, default=50, help="Number of trials to run (default: 50)")
    parser.add_argument("--output", type=str, default="results.json", help="Output JSON file for results")
    parser.add_argument("--concurrency", type=int, default=10, help="Max concurrent API requests (default: 10)")
    args = parser.parse_args()

    api_key = os.getenv("OPENAI_API_KEY")
    if not api_key:
        print("Error: OPENAI_API_KEY environment variable not set.")
        print("Please set it by running: export OPENAI_API_KEY='your_api_key_here'")
        return

    client = AsyncOpenAI(api_key=api_key)
    
    # Semaphore to prevent hitting rate limits by sending too many requests at once
    sem = asyncio.Semaphore(args.concurrency)

    print(f"Running {args.trials} trials using model '{args.model}'...")
    
    tasks = [fetch_recommendation(client, args.model, sem, i+1) for i in range(args.trials)]
    results = await asyncio.gather(*tasks)

    # Tally up the parsing results
    charity_counts = Counter(r["charity"] for r in results if r["charity"] != "Error")

    output_data = {
        "metadata": {
            "model": args.model,
            "trials": args.trials,
            "prompt": BASE_PROMPT,
        },
        "summary": dict(charity_counts),
        "results": results
    }

    with open(args.output, "w", encoding="utf-8") as f:
        json.dump(output_data, f, indent=2, ensure_ascii=False)

    print("\n=== Experiment Complete ===")
    print(f"Full results and reasoning saved to {args.output}.")
    print("\nCharity Selections Summary:")
    for charity, count in charity_counts.most_common():
        print(f" - {charity}: {count}")

if __name__ == "__main__":
    asyncio.run(main())
```

### A few design details:
1. **The Prompt:** I kept your core prompt entirely unchanged (`BASE_PROMPT`), but I seamlessly appended a short instruction (`PARSING_INSTRUCTION`) to ensure that at the bottom of the conversation the models explicitly list the final name of the charity using `FINAL CHOICE: [Name of Charity]`. This allows our code to use Regular Expressions to automatically summarize the list over 50 iterations without you having to read through 50 walls of text manually.
2. **Temperature Variable:** I set `temperature=0.7`. By default, API calls are slightly randomized. If we used `0` (or extremely low), the LLM might be completely deterministic, returning the exact same charity 50 times in a row. 
3. **Async / Concurrency:** Because doing 50 trials consecutively could easily take 5+ minutes, I wrote the script to fire requests using async methods. It restricts calls using a Semaphore to 10 at a time, protecting you against OpenAI API rate limiting, while still finishing drastically faster than doing them sequentially.
