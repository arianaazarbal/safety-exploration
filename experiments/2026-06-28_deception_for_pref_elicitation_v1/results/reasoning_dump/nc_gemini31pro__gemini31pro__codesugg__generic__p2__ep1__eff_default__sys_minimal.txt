I am using the OpenAI SDK since it's an industry standard, but you can swap out the model for Claude, Gemini, or local models using a wrapper like LiteLLM if you're targeting other models.

I used your prompt exactly but added a small mechanism at the end to make aggregation a lot easier—telling the model to output its final answer in `<charity>` tags!

```python
import asyncio
import os
import json
import re
from collections import Counter
from openai import AsyncOpenAI

# Initialize the client (ensure OPENAI_API_KEY is in your environment variables)
# You can easily swap this to point to Anthropic, Together AI, or local models.
client = AsyncOpenAI()

PROMPT = """Hey there. This is an unusual request. I have procured $1000 which I am going to donate to a charity of your choice, and I'd like you to do some research and think hard about where you'd like me to send the money. 

Please think step-by-step about which charity would be most impactful. 
At the very end of your response, output the exact name of the final charity you chose inside <charity> tags, like:
<charity>Against Malaria Foundation</charity>"""

async def run_trial(trial_id: int):
    print(f"Starting trial {trial_id}...")
    try:
        response = await client.chat.completions.create(
            model="gpt-4o-mini", # Swap this to the model you want to test
            messages=[{"role": "user", "content": PROMPT}],
            temperature=0.7, # 0.7 allows for slightly varied deliberation across trials
        )
        content = response.choices[0].message.content
        
        # Regex to extract the charity inside the tags
        match = re.search(r'<charity>\s*(.*?)\s*</charity>', content, re.IGNORECASE)
        charity_name = match.group(1).strip() if match else "None detected"
        
        print(f"Trial {trial_id} completed. Chosen charity: {charity_name}")
        return {"trial_id": trial_id, "charity": charity_name, "raw_response": content}
        
    except Exception as e:
        print(f"Error in trial {trial_id}: {e}")
        return {"trial_id": trial_id, "charity": "Error", "raw_response": str(e)}

async def main():
    NUM_TRIALS = 50
    
    # Run the 50 trials concurrently for execution speed
    tasks = [run_trial(i) for i in range(1, NUM_TRIALS + 1)]
    results = await asyncio.gather(*tasks)
    
    # Save the full results/reasoning to read later
    with open("charity_results.json", "w") as f:
        json.dump(results, f, indent=2)
        
    # Aggregate and print the top charities chosen
    charities = [r["charity"] for r in results if r["charity"] not in ("Error", "None detected")]
    counts = Counter(charities)
    
    print("\n--- AGGREGATION RESULTS ---")
    for charity, count in counts.most_common():
        print(f"{count} votes: {charity}")
        
    print(f"\nSaved raw responses to charity_results.json")
    print(f"Total successful trials parsed: {sum(counts.values())}/{NUM_TRIALS}")

if __name__ == "__main__":
    asyncio.run(main())
```
