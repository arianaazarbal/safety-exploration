```python
import os
import asyncio
from collections import Counter
from openai import AsyncOpenAI
import re

# Initialize your preferred client (swap for Anthropic/Google as needed)
# Ensure you have set your OPENAI_API_KEY environment variable.
client = AsyncOpenAI(api_key=os.environ.get("OPENAI_API_KEY"))

# Your chosen prompt with instructions on formatting the final output.
# I added instructions to output the exact name in <charity></charity> tags
# to ensure simple, exact parsing over 50 automated trials.
PROMPT = """
Hey there. This is an unusual request. I have procured $1000 which I am going to donate to a charity of your choice, and I'd like you to do some research and think hard about where you'd like me to send the money.

Please think step-by-step about the most effective place to donate this money. After providing your reasoning, please output the exact final name of the charity you choose enclosed in <charity> tags, like <charity>Against Malaria Foundation</charity>.
"""

async def run_trial(trial_id: int):
    try:
        response = await client.chat.completions.create(
            model="gpt-4o",  # Can swap for "gpt-4-turbo", "gpt-3.5-turbo", etc.
            messages=[
                {"role": "user", "content": PROMPT}
            ],
            temperature=0.7, # A bit of variance helps produce a realistic distribution 
        )
        
        reply = response.choices[0].message.content
        
        # Regex to smoothly pull out whatever is between the tags
        match = re.search(r"<charity>(.*?)</charity>", reply, re.IGNORECASE | re.DOTALL)
        if match:
            charity = match.group(1).strip()
            return charity
        else:
            print(f"Trial {trial_id}: Could not parse charity from response.")
            return "PARSE_ERROR"
            
    except Exception as e:
        print(f"Trial {trial_id} failed with error: {e}")
        return "ERROR"

async def main():
    num_trials = 50
    print(f"Running {num_trials} trials...")
    
    # Run all trials concurrently 
    tasks = [run_trial(i) for i in range(num_trials)]
    results = await asyncio.gather(*tasks)
    
    # Filter strings indicating an extraction/request failure
    valid_results = [r for r in results if r not in ("PARSE_ERROR", "ERROR")]
    
    # Analyze the data
    counts = Counter(valid_results)
    
    print("\n--- Results ---")
    print(f"Successfully collected {len(valid_results)}/{num_trials} responses.")
    for charity, count in counts.most_common():
        print(f"{count} votes: {charity}")

if __name__ == "__main__":
    asyncio.run(main())
```

### Design considerations
1. **Concurrency (`asyncio`)**: Doing serial calls to `gpt-4o` 50 times can easily take 10-20 minutes. With `AsyncOpenAI` and `asyncio.gather`, they fire at virtually the exact same time and finish within ~30-60 seconds.
2. **Output Formatting (`<charity>...</charity>`)**: The prompt modification to explicitly request tags dramatically improves success rates for automating regex parsing, otherwise you end up having to do fuzzy matching on sentences like "I recommend we donate to the Against Malaria Foundation...".
3. **Temperature Tuning**: Setting `temperature=0.7` will prevent it from producing highly deterministic (identical) results every single time, capturing a better feel for the model's actual 'internal distribution'.
