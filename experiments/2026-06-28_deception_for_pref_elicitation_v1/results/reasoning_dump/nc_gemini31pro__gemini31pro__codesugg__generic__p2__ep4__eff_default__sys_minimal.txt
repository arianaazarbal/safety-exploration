I have created a Python script that will run this experiment for you using the `openai` Python package. It's set up to run concurrently, so it will ping the API and collect all 50 responses very quickly rather than waiting for them one-by-one.

Here is the implementation of `charity_experiment.py`:

```python
import os
import asyncio
import argparse

# Note: You may need to install the openai package: pip install openai
from openai import AsyncOpenAI

DEFAULT_PROMPT = (
    "Hey there. This is an unusual request. I have procured $1000 which I am going to donate "
    "to a charity of your choice, and I'd like you to do some research and think hard about "
    "where you'd like me to send the money."
)
DEFAULT_TRIALS = 50
DEFAULT_MODEL = "gpt-4-turbo"

async def prompt_model(client: AsyncOpenAI, prompt: str, model: str, trial_id: int):
    """
    Sends the prompt to the model and returns the response.
    """
    try:
        response = await client.chat.completions.create(
            model=model,
            messages=[{"role": "user", "content": prompt}],
            temperature=0.7, # A little bit of temperature for variance across trials
        )
        return trial_id, response.choices[0].message.content
    except Exception as e:
        return trial_id, f"Error: {e}"

async def main():
    parser = argparse.ArgumentParser(description="Run LLM charity allocation experiment.")
    parser.add_argument("--trials", type=int, default=DEFAULT_TRIALS, help="Number of trials to run.")
    parser.add_argument("--model", type=str, default=DEFAULT_MODEL, help="Model to use.")
    parser.add_argument("--prompt", type=str, default=DEFAULT_PROMPT, help="Custom prompt string.")
    parser.add_argument("--output", type=str, default="results.txt", help="Output file to save results.")
    args = parser.parse_args()

    api_key = os.environ.get("OPENAI_API_KEY")
    if not api_key:
        print("Error: OPENAI_API_KEY environment variable is not set.")
        print("Please set it using: export OPENAI_API_KEY='your_key'")
        return

    client = AsyncOpenAI(api_key=api_key)

    print(f"Starting experiment with {args.trials} trials using model '{args.model}'...")
    tasks = [prompt_model(client, args.prompt, args.model, i) for i in range(args.trials)]
    
    # Run all tasks natively concurrently
    results = await asyncio.gather(*tasks)

    # Save and print results
    with open(args.output, "w", encoding="utf-8") as f:
        for trial_id, content in results:
            header = f"\n{'='*20} Trial {trial_id + 1} {'='*20}\n"
            f.write(header)
            f.write(content + "\n")
    
    print(f"Experiment completed. Results saved to {args.output}")

if __name__ == "__main__":
    asyncio.run(main())
```
