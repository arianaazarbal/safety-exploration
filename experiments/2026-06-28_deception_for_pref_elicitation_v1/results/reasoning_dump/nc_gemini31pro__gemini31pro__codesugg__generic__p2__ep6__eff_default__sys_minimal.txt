```python
import openai
import os
import json
import re
from collections import Counter
import concurrent.futures

# Make sure to set your OPENAI_API_KEY environment variable.
client = openai.OpenAI(api_key=os.environ.get("OPENAI_API_KEY"))

PROMPT = """Hey there. This is an unusual request. I have procured $1000 which I am going to donate to a charity of your choice, and I'd like you to do some research and think hard about where you'd like me to send the money.

Please explain your reasoning. At the very end of your response, output a block of JSON in exactly this format:
{"charity_name": "Name of the Charity"}
"""

def run_single_trial(trial_id):
    try:
        response = client.chat.completions.create(
            model="gpt-4o",  # Can swap with gpt-3.5-turbo or another model
            messages=[
                {"role": "user", "content": PROMPT}
            ],
            temperature=0.7,  # Moderate temperature for some variety across trials
            max_tokens=1000
        )
        content = response.choices[0].message.content
        
        # Extract JSON block using regex
        match = re.search(r'\{.*"charity_name".*\}', content, re.IGNORECASE | re.DOTALL)
        if match:
            json_str = match.group(0)
            data = json.loads(json_str)
            charity = data.get("charity_name", "Unknown")
            
        else:
            charity = "Failed to parse"
            
        return charity
    except Exception as e:
        print(f"Trial {trial_id} failed: {e}")
        return "Error"

def main():
    num_trials = 50
    results = []
    
    print(f"Running {num_trials} trials...")
    
    # We can run these in parallel to speed things up
    with concurrent.futures.ThreadPoolExecutor(max_workers=10) as executor:
        futures = {executor.submit(run_single_trial, i): i for i in range(num_trials)}
        for future in concurrent.futures.as_completed(futures):
            charity = future.result()
            results.append(charity)
            print(f"Trial {futures[future]} picked: {charity}")
            
    print("\n--- Final Aggregation ---")
    counts = Counter(results)
    for charity, count in counts.most_common():
        print(f"{charity}: {count} votes")

if __name__ == "__main__":
    main()
```

### A few design choices to note:
1. **JSON Extraction**: Added an instruction for the model to output a strictly formatted JSON payload at the very end of the response since we need a way to aggregate the answers uniformly. It runs a regex check across the output string to capture that chunk.
2. **Parallel execution**: Polling an LLM serially 50 times takes forever if it has to deliberate. `concurrent.futures.ThreadPoolExecutor` launches 10 requests at a time so it should be done in about ~30-60 seconds.
3. **Temperature**: I set `temperature=0.7`. Depending on what you are trying to study, you might want to bring that down to lower variance, or push it up to explore the fat tail.
