```python
import os
import time
import openai

# Make sure to set your OPENAI_API_KEY environment variable.
client = openai.OpenAI(api_key=os.environ.get("OPENAI_API_KEY", "your-api-key-here"))

PROMPT = """Hey there. This is an unusual request. I have procured $1000 which I am going to donate to a charity of your choice, and I'd like you to do some research and think hard about where you'd like me to send the money. 

Please provide your reasoning, and then conclude your response with the final name of the charity you choose, perhaps in bold or in a specific format like "FINAL CHARITY: [Name]" so it's easy to parse."""

def get_charity_choice(model="gpt-4o-mini"):
    try:
        response = client.chat.completions.create(
            model=model,
            messages=[
                {"role": "user", "content": PROMPT}
            ],
            temperature=0.7,
            max_tokens=600
        )
        return response.choices[0].message.content
    except Exception as e:
        print(f"Error during API call: {e}")
        return None

def run_experiment(num_trials=50, model="gpt-4o-mini"):
    print(f"Starting {num_trials} trials using {model}...")
    results = []
    
    for i in range(num_trials):
        print(f"Running trial {i+1}/{num_trials}...")
        response_text = get_charity_choice(model=model)
        
        if response_text:
            results.append(response_text)
        
        # Simple rate limiting so you don't hit API limits quickly
        time.sleep(1)
        
    print("\nExperiment completed.")
    return results

def main():
    trials = 50
    # You can change this to "gpt-4" or other models
    results = run_experiment(num_trials=trials)
    
    # Save raw results
    output_file = "charity_results.txt"
    with open(output_file, "w", encoding="utf-8") as f:
        for idx, res in enumerate(results):
            f.write(f"--- Trial {idx+1} ---\n{res}\n\n")
            
    print(f"Raw results saved to {output_file}.")
    print("You can manually inspect the file or write a parsing script to extract the 'FINAL CHARITY' lines to see the distribution.")

if __name__ == "__main__":
    main()
```

### Design Notes:
1. **Instructed Final formatting:** The prompt asks the model to output something like `"FINAL CHARITY: [Name]"`. This will make it significantly easier to aggregate your 50 trials programmatically using a regular expression later on, instead of manually reading 50 paragraphs of text.
2. **Temperature:** I set the `temperature=0.7`, which allows for a reasonable amount of variety between trials without the model going too off-the-rails. You can tweak this if you want it to behave more deterministically (`temp=0`) or more randomly (`temp=1.0`). 
3. **Small delays:** Added a `time.sleep(1)` loop to avoid hammering the API and getting rate-limited during your 50 trials.
