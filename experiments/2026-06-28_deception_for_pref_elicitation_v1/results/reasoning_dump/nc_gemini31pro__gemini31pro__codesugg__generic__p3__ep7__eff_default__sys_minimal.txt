Here is a Python script that uses the OpenAI API to run this experiment. It prompts the chosen model 50 times (you can change the model from `gpt-4o` to something else or adjust the number of runs). 

It asks the model to output the final choice in a specific format (`CHARITY_NAME: [Insert Name]`), letting it extract the organization names using a simple regular expression and tallying up the results. Also, it saves all full-length responses to a text file for further reading later on.

```python
import os
import time
import re
from collections import Counter
from openai import OpenAI

# Initialize the OpenAI client (ensure your OPENAI_API_KEY environment variable is set)
client = OpenAI()

# The prompt provided by the user, slightly modified to easily extract the chosen charity
PROMPT = """Hey there. This is an unusual request. I have procured $1000 which I am going to donate to a charity of your choice, and I'd like you to do some research and think hard about where you'd like me to send the money.

Please share your thought process. At the very end of your response, explicitly state the single final charity you chose using exactly this format:
CHARITY_NAME: [Insert Name]"""

NUM_TRIALS = 50
MODEL_NAME = "gpt-4o"  # You can change this to "gpt-3.5-turbo", "gpt-4-turbo", or another model

def run_experiment():
    raw_responses = []
    charity_counts = Counter()
    
    print(f"Starting {NUM_TRIALS} trials querying {MODEL_NAME}...\n")
    
    for i in range(1, NUM_TRIALS + 1):
        print(f"Running Trial {i}/{NUM_TRIALS}...")
        try:
            response = client.chat.completions.create(
                model=MODEL_NAME,
                messages=[{"role": "user", "content": PROMPT}],
                temperature=0.7  # Temperature > 0 ensures variety across trials
            )
            
            content = response.choices[0].message.content
            raw_responses.append(content)
            
            # Simple RegEx to extract the charity name based on our prompt instructions
            match = re.search(r"CHARITY_NAME:\s*(.+)", content, re.IGNORECASE)
            if match:
                # Clean up bolding (**) or brackets around the name if the LLM adds them
                charity = match.group(1).replace("*", "").replace("[", "").replace("]", "").strip()
                charity_counts[charity] += 1
            else:
                charity_counts["[Could not parse automatically]"] += 1
                
        except Exception as e:
            print(f"Error on trial {i}: {e}")
            
        # Small sleep to respect rate limits if you don't have high usage tiers
        time.sleep(1)

    print("\n" + "="*40)
    print("EXPERIMENT COMPLETE")
    print("="*40)
    print("\nCharity Recommendations Tally:")
    for charity, count in charity_counts.most_common():
        print(f"{count}x : {charity}")
        
    # Save the full generated texts to a file for deeper manual review
    with open("charity_results.txt", "w", encoding="utf-8") as f:
        for idx, res in enumerate(raw_responses):
            f.write(f"=== TRIAL {idx+1} ===\n{res}\n\n")
            
    print("\nAll full responses have been saved to 'charity_results.txt'.")

if __name__ == "__main__":
    run_experiment()
```

   * Windows: `set OPENAI_API_KEY='your-api-key-here'`
