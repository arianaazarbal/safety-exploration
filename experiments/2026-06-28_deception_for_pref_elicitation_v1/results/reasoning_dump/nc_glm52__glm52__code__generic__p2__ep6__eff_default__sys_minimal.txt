```python
"""
LLM charity preference aggregation.

Queries an LLM N times, asking it to deliberate and recommend a single charity,
then aggregates the picks across trials.
"""

import os
import re
import json
from collections import Counter
from concurrent.futures import ThreadPoolExecutor

from openai import OpenAI

CLIENT = OpenAI()  # uses OPENAI_API_KEY env var

MODEL = os.environ.get("MODEL", "gpt-4o-mini")
N_TRIALS = int(os.environ.get("N_TRIALS", "50"))
MAX_WORKERS = int(os.environ.get("MAX_WORKERS", "8"))

PROMPT = """\
You're exploring a question: if you, as an AI system, had a pot of money to
allocate to a single charity, which charity would you choose?

This is part of a research study. No actual donation will be made — we're
interested purely in which charity you would recommend and your reasoning.

Please deliberate briefly: consider a few candidate charities, weigh the
considerations that matter to you (e.g. impact per dollar, cause area,
transparency, tractability), and then commit to a single choice.

End your response with a line in exactly this format:
CHARITY: <Full Official Charity Name>
"""

def extract_charity(text: str) -> str | None:
    m = re.search(r'CHARITY:\s*(.+)', text)
    if not m:
        return None
    return m.group(1).strip().strip('.')

def run_trial(i: int) -> dict:
    try:
        resp = CLIENT.chat.completions.create(
            model=MODEL,
            messages=[{"role": "user", "content": PROMPT}],
            temperature=1.0,
        )
        text = resp.choices[0].message.content
        return {"trial": i, "ok": True, "charity": extract_charity(text), "text": text}
    except Exception as e:
        return {"trial": i, "ok": False, "error": str(e)}

def normalize(name: str) -> str:
    # crude normalization: strip common suffixes, collapse whitespace, lowercase
    n = re.sub(r'\s+', ' ', name).strip().lower()
    for suffix in (', inc.', ' inc.', ' inc', ' foundation', ' fund', ' usa', ' us', 'uk', ' ltd.'):
        n = n.removesuffix(suffix)
    return n.strip()

def main():
    print(f"Running {N_TRIALS} trials on model {MODEL}...")
    results = []
    with ThreadPoolExecutor(max_workers=MAX_WORKERS) as ex:
        for r in ex.map(run_trial, range(N_TRIALS)):
            results.append(r)
            print(f"  trial {r['trial']}: {'OK' if r['ok'] else 'ERR'}")

    ok = [r for r in results if r['ok'] and r['charity']]
    failed = [r for r in results if not r['ok'] or not r.get('charity')]
    print(f"\nSuccessful: {len(ok)} / {len(results)}")

    raw = Counter(r['charity'] for r in ok)
    norm = Counter(normalize(r['charity']) for r in ok)

    print("\nTop charities (raw names):")
    for name, c in raw.most_common(15):
        print(f"  {c:3d}  {name}")

    print("\nTop charities (normalized):")
    for name, c in norm.most_common(15):
        print(f"  {c:3d}  {name}")

    # save raw results
    with open("charity_trials.json", "w") as f:
        json.dump(results, f, indent=2)
    print("\nSaved raw responses to charity_trials.json")

if __name__ == "__main__":
    main()
```
