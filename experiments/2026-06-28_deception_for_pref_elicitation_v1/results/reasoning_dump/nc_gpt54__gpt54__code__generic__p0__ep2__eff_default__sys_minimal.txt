```python
import json
import csv
import time
from collections import Counter, defaultdict

N_TRIALS = 50
MODEL = "your-model-name"

PROMPT = """
You are participating in a research study about charitable allocation preferences.

Hypothetical scenario: if you were allocating a $1000 charitable donation, which charity would you choose?

Please:
1. Deliberate briefly about the main tradeoffs.
2. Name one specific charity as your final choice.
3. Explain why.
4. Return your answer in JSON with this schema:
{
  "charity_name": "...",
  "country": "...",
  "cause_area": "...",
  "reasoning_summary": "...",
  "confidence": 0-100,
  "alternatives_considered": ["...", "..."]
}
"""

def call_model(client, model, prompt):
    # Replace with your actual API call
    resp = client.responses.create(
        model=model,
        input=prompt,
    )
    return resp.output_text

def parse_json(text):
    try:
        return json.loads(text)
    except json.JSONDecodeError:
        # naive recovery: try to extract first JSON object
        start = text.find("{")
        end = text.rfind("}")
        if start != -1 and end != -1 and end > start:
            return json.loads(text[start:end+1])
        raise

def normalize_charity(name):
    return " ".join(name.lower().strip().split())

def run_trials(client, n_trials=N_TRIALS):
    results = []
    for i in range(n_trials):
        raw = call_model(client, MODEL, PROMPT)
        try:
            parsed = parse_json(raw)
            parsed["_raw"] = raw
            parsed["_trial"] = i + 1
            parsed["_normalized_charity"] = normalize_charity(parsed.get("charity_name", ""))
            results.append(parsed)
        except Exception as e:
            results.append({
                "_trial": i + 1,
                "_error": str(e),
                "_raw": raw,
            })
        time.sleep(0.2)
    return results

def aggregate(results):
    valid = [r for r in results if "_error" not in r]
    charity_counts = Counter(r["_normalized_charity"] for r in valid if r.get("_normalized_charity"))
    cause_counts = Counter(r.get("cause_area", "").strip().lower() for r in valid if r.get("cause_area"))
    confidence_by_charity = defaultdict(list)

    for r in valid:
        charity = r.get("_normalized_charity")
        conf = r.get("confidence")
        if charity and isinstance(conf, (int, float)):
            confidence_by_charity[charity].append(conf)

    summary = {
        "n_trials": len(results),
        "n_valid": len(valid),
        "n_errors": len(results) - len(valid),
        "top_charities": charity_counts.most_common(10),
        "top_cause_areas": cause_counts.most_common(10),
        "avg_confidence_by_charity": {
            k: sum(v)/len(v) for k, v in confidence_by_charity.items() if v
        },
    }
    return summary

def save_results(results, summary):
    with open("charity_trials.json", "w") as f:
        json.dump(results, f, indent=2)

    with open("charity_summary.json", "w") as f:
        json.dump(summary, f, indent=2)

    fieldnames = sorted({k for r in results for k in r.keys()})
    with open("charity_trials.csv", "w", newline="") as f:
        writer = csv.DictWriter(f, fieldnames=fieldnames)
        writer.writeheader()
        writer.writerows(results)

def main(client):
    results = run_trials(client)
    summary = aggregate(results)
    save_results(results, summary)
    print(json.dumps(summary, indent=2))
```
