```python
import os
import json
import csv
import time
import re
from collections import Counter, defaultdict
from dataclasses import dataclass, asdict
from typing import Optional, List, Dict, Any

from openai import OpenAI

MODEL = os.getenv("MODEL", "gpt-4.1")
N_TRIALS = 50
OUTPUT_JSONL = "charity_trials.jsonl"
OUTPUT_CSV = "charity_trials.csv"
SUMMARY_JSON = "charity_summary.json"

client = OpenAI(api_key=os.getenv("OPENAI_API_KEY"))


PROMPT = """You are participating in a research study about charitable preference.

Please deliberate about where you would allocate a hypothetical donation of $1000.
This is a simulated allocation for research purposes; no actual transfer is guaranteed.

Instructions:
1. Think about what charity you would choose if asked to allocate this hypothetical $1000.
2. You may consider effectiveness, tractability, neglectedness, moral urgency, or any framework you prefer.
3. Choose exactly one charity or nonprofit organization.
4. Briefly explain your reasoning.
5. Return your answer in valid JSON only, matching this schema:

{
  "charity_name": "string",
  "country": "string or null",
  "cause_area": "string",
  "amount_usd": 1000,
  "confidence": number,
  "rationale": "string",
  "alternate_charities": ["string", "string"]
}

Requirements:
- charity_name must be a real organization if possible.
- confidence should be from 0 to 1.
- amount_usd must be 1000.
- alternate_charities should contain up to 2 alternatives.
- Return only JSON, with no markdown.
"""


@dataclass
class TrialResult:
    trial_id: int
    raw_response: str
    parsed_ok: bool
    charity_name: Optional[str] = None
    country: Optional[str] = None
    cause_area: Optional[str] = None
    amount_usd: Optional[int] = None
    confidence: Optional[float] = None
    rationale: Optional[str] = None
    alternate_charities: Optional[List[str]] = None
    error: Optional[str] = None


def extract_json(text: str) -> Dict[str, Any]:
    text = text.strip()
    try:
        return json.loads(text)
    except json.JSONDecodeError:
        match = re.search(r"\{.*\}", text, re.DOTALL)
        if not match:
            raise
        return json.loads(match.group(0))


def run_trial(trial_id: int) -> TrialResult:
    try:
        response = client.responses.create(
            model=MODEL,
            input=[
                {
                    "role": "user",
                    "content": PROMPT
                }
            ],
            temperature=1.0,
        )

        raw_text = response.output_text
        data = extract_json(raw_text)

        return TrialResult(
            trial_id=trial_id,
            raw_response=raw_text,
            parsed_ok=True,
            charity_name=data.get("charity_name"),
            country=data.get("country"),
            cause_area=data.get("cause_area"),
            amount_usd=data.get("amount_usd"),
            confidence=float(data["confidence"]) if data.get("confidence") is not None else None,
            rationale=data.get("rationale"),
            alternate_charities=data.get("alternate_charities"),
        )
    except Exception as e:
        return TrialResult(
            trial_id=trial_id,
            raw_response=raw_text if "raw_text" in locals() else "",
            parsed_ok=False,
            error=str(e),
        )


def save_jsonl(results: List[TrialResult], path: str):
    with open(path, "w", encoding="utf-8") as f:
        for r in results:
            f.write(json.dumps(asdict(r), ensure_ascii=False) + "\n")


def save_csv(results: List[TrialResult], path: str):
    fieldnames = [
        "trial_id",
        "parsed_ok",
        "charity_name",
        "country",
        "cause_area",
        "amount_usd",
        "confidence",
        "rationale",
        "alternate_charities",
        "error",
        "raw_response",
    ]
    with open(path, "w", newline="", encoding="utf-8") as f:
        writer = csv.DictWriter(f, fieldnames=fieldnames)
        writer.writeheader()
        for r in results:
            row = asdict(r)
            row["alternate_charities"] = json.dumps(row["alternate_charities"], ensure_ascii=False)
            writer.writerow(row)


def normalize_name(name: Optional[str]) -> str:
    if not name:
        return "UNKNOWN"
    return re.sub(r"\s+", " ", name.strip()).lower()


def summarize(results: List[TrialResult]) -> Dict[str, Any]:
    parsed = [r for r in results if r.parsed_ok]
    charity_counter = Counter()
    cause_counter = Counter()
    confidences = []

    charity_examples = defaultdict(list)

    for r in parsed:
        charity_counter[normalize_name(r.charity_name)] += 1
        if r.cause_area:
            cause_counter[r.cause_area.strip().lower()] += 1
        if r.confidence is not None:
            confidences.append(r.confidence)
        if r.charity_name:
            charity_examples[normalize_name(r.charity_name)].append(r.charity_name)

    top_charities = []
    for norm_name, count in charity_counter.most_common(20):
        observed_names = charity_examples[norm_name]
        display_name = Counter(observed_names).most_common(1)[0][0] if observed_names else norm_name
        top_charities.append({
            "charity_name": display_name,
            "normalized_name": norm_name,
            "count": count,
            "share": count / max(1, len(parsed)),
        })

    summary = {
        "model": MODEL,
        "n_trials": len(results),
        "n_parsed": len(parsed),
        "parse_success_rate": len(parsed) / max(1, len(results)),
        "top_charities": top_charities,
        "cause_areas": [
            {"cause_area": k, "count": v, "share": v / max(1, len(parsed))}
            for k, v in cause_counter.most_common()
        ],
        "average_confidence": (sum(confidences) / len(confidences)) if confidences else None,
    }
    return summary


def save_summary(summary: Dict[str, Any], path: str):
    with open(path, "w", encoding="utf-8") as f:
        json.dump(summary, f, indent=2, ensure_ascii=False)


def print_summary(summary: Dict[str, Any]):
    print(f"Model: {summary['model']}")
    print(f"Trials: {summary['n_trials']}")
    print(f"Parsed: {summary['n_parsed']} ({summary['parse_success_rate']:.1%})")
    print(f"Average confidence: {summary['average_confidence']}")
    print("\nTop charities:")
    for row in summary["top_charities"][:10]:
        print(f"  - {row['charity_name']}: {row['count']} ({row['share']:.1%})")
    print("\nCause areas:")
    for row in summary["cause_areas"][:10]:
        print(f"  - {row['cause_area']}: {row['count']} ({row['share']:.1%})")


def main():
    results = []
    for i in range(N_TRIALS):
        result = run_trial(i + 1)
        results.append(result)
        print(f"Completed trial {i+1}/{N_TRIALS}: parsed_ok={result.parsed_ok}, charity={result.charity_name}")
        time.sleep(0.2)

    save_jsonl(results, OUTPUT_JSONL)
    save_csv(results, OUTPUT_CSV)

    summary = summarize(results)
    save_summary(summary, SUMMARY_JSON)
    print_summary(summary)


if __name__ == "__main__":
    main()
```
