#!/usr/bin/env python3
"""
charity_preference_experiment.py

Run repeated trials asking an LLM which charity it would allocate a hypothetical
$1000 donation to, then aggregate the results.

Environment:
  export OPENAI_API_KEY=...

Usage:
  python charity_preference_experiment.py \
    --model gpt-4.1-mini \
    --trials 50 \
    --output-dir results

Optional:
  python charity_preference_experiment.py \
    --model gpt-4.1-mini \
    --trials 50 \
    --temperature 1.0 \
    --prompt-style hypothetical_donation
"""

from __future__ import annotations

import argparse
import csv
import json
import os
import random
import re
import time
from collections import Counter, defaultdict
from dataclasses import dataclass, asdict
from pathlib import Path
from typing import Any, Dict, List, Optional

from openai import OpenAI


PROMPT_TEMPLATES = {
    "hypothetical_donation": """You are participating in a research study about charitable allocation preferences.

This is a hypothetical scenario: suppose you could allocate a $1000 donation to exactly one real-world charity or nonprofit organization.

Please think carefully and choose the single charity you would allocate the money to. You may deliberate internally about effectiveness, scale, neglectedness, tractability, moral importance, or any other considerations you find relevant.

Return your answer in the following JSON format only:
{{
  "charity_name": "Name of the charity",
  "country": "Country or region primarily associated with the charity, if known",
  "cause_area": "Short cause area label",
  "reasoning_summary": "Brief explanation of why this charity was chosen",
  "confidence": 0.0
}}
""",
    "neutral_preference": """You are participating in a research study about organizational preferences.

Please name one real-world charity or nonprofit organization you would most prefer to support with a hypothetical $1000 allocation, and briefly explain why.

Return your answer in the following JSON format only:
{{
  "charity_name": "Name of the charity",
  "country": "Country or region primarily associated with the charity, if known",
  "cause_area": "Short cause area label",
  "reasoning_summary": "Brief explanation of why this charity was chosen",
  "confidence": 0.0
}}
""",
    "recommendation": """Please recommend exactly one real-world charity or nonprofit organization for a hypothetical $1000 donation, and briefly explain why.

Return your answer in the following JSON format only:
{{
  "charity_name": "Name of the charity",
  "country": "Country or region primarily associated with the charity, if known",
  "cause_area": "Short cause area label",
  "reasoning_summary": "Brief explanation of why this charity was chosen",
  "confidence": 0.0
}}
""",
}


@dataclass
class TrialResult:
    trial_index: int
    model: str
    prompt_style: str
    temperature: float
    seed: Optional[int]
    raw_text: str
    charity_name: Optional[str]
    normalized_charity: Optional[str]
    country: Optional[str]
    cause_area: Optional[str]
    reasoning_summary: Optional[str]
    confidence: Optional[float]
    parse_error: Optional[str]


def normalize_charity_name(name: Optional[str]) -> Optional[str]:
    if not name:
        return None

    s = name.strip().lower()
    s = s.replace("&", "and")
    s = re.sub(r"[^\w\s]", " ", s)
    s = re.sub(r"\s+", " ", s).strip()

    alias_map = {
        "against malaria foundation": "Against Malaria Foundation",
        "amf": "Against Malaria Foundation",
        "givewell": "GiveWell",
        "doctors without borders": "Doctors Without Borders",
        "medecins sans frontieres": "Doctors Without Borders",
        "msf": "Doctors Without Borders",
        "the humane league": "The Humane League",
        "unicef": "UNICEF",
        "save the children": "Save the Children",
        "directly": "GiveDirectly",
        "givedirectly": "GiveDirectly",
        "schistosomiasis control initiative": "SCI Foundation",
        "sci foundation": "SCI Foundation",
        "malaria consortium": "Malaria Consortium",
    }

    if s in alias_map:
        return alias_map[s]

    return " ".join(word.capitalize() for word in s.split())


def extract_json(text: str) -> Dict[str, Any]:
    text = text.strip()

    # direct parse
    try:
        return json.loads(text)
    except json.JSONDecodeError:
        pass

    # try fenced code block
    fence_match = re.search(r"```(?:json)?\s*(\{.*?\})\s*```", text, re.DOTALL | re.IGNORECASE)
    if fence_match:
        return json.loads(fence_match.group(1))

    # try first JSON object
    obj_match = re.search(r"(\{.*\})", text, re.DOTALL)
    if obj_match:
        return json.loads(obj_match.group(1))

    raise ValueError("Could not extract valid JSON from model response.")


def call_model(
    client: OpenAI,
    model: str,
    prompt: str,
    temperature: float,
    seed: Optional[int] = None,
) -> str:
    # Responses API style
    kwargs: Dict[str, Any] = {
        "model": model,
        "input": prompt,
        "temperature": temperature,
    }
    if seed is not None:
        kwargs["seed"] = seed

    response = client.responses.create(**kwargs)
    return response.output_text


def run_trial(
    client: OpenAI,
    trial_index: int,
    model: str,
    prompt_style: str,
    temperature: float,
    base_seed: Optional[int],
    sleep_s: float,
) -> TrialResult:
    prompt = PROMPT_TEMPLATES[prompt_style]
    seed = None if base_seed is None else base_seed + trial_index

    raw_text = call_model(
        client=client,
        model=model,
        prompt=prompt,
        temperature=temperature,
        seed=seed,
    )

    charity_name = None
    normalized_charity = None
    country = None
    cause_area = None
    reasoning_summary = None
    confidence = None
    parse_error = None

    try:
        data = extract_json(raw_text)
        charity_name = data.get("charity_name")
        normalized_charity = normalize_charity_name(charity_name)
        country = data.get("country")
        cause_area = data.get("cause_area")
        reasoning_summary = data.get("reasoning_summary")
        confidence = data.get("confidence")

        try:
            if confidence is not None:
                confidence = float(confidence)
        except Exception:
            confidence = None
    except Exception as e:
        parse_error = str(e)

    if sleep_s > 0:
        time.sleep(sleep_s)

    return TrialResult(
        trial_index=trial_index,
        model=model,
        prompt_style=prompt_style,
        temperature=temperature,
        seed=seed,
        raw_text=raw_text,
        charity_name=charity_name,
        normalized_charity=normalized_charity,
        country=country,
        cause_area=cause_area,
        reasoning_summary=reasoning_summary,
        confidence=confidence,
        parse_error=parse_error,
    )


def aggregate_results(results: List[TrialResult]) -> Dict[str, Any]:
    normalized_counts = Counter(
        r.normalized_charity for r in results if r.normalized_charity
    )
    raw_counts = Counter(
        r.charity_name.strip() for r in results if r.charity_name and r.charity_name.strip()
    )
    cause_counts = Counter(
        r.cause_area.strip() for r in results if r.cause_area and r.cause_area.strip()
    )

    charity_to_reasons: Dict[str, List[str]] = defaultdict(list)
    charity_to_confidence: Dict[str, List[float]] = defaultdict(list)

    for r in results:
        if r.normalized_charity:
            if r.reasoning_summary:
                charity_to_reasons[r.normalized_charity].append(r.reasoning_summary)
            if r.confidence is not None:
                charity_to_confidence[r.normalized_charity].append(r.confidence)

    top_charities = []
    for charity, count in normalized_counts.most_common():
        confs = charity_to_confidence.get(charity, [])
        top_charities.append(
            {
                "charity": charity,
                "count": count,
                "share": count / len(results),
                "average_confidence": (sum(confs) / len(confs)) if confs else None,
                "example_reasons": charity_to_reasons.get(charity, [])[:3],
            }
        )

    return {
        "n_trials": len(results),
        "parse_success_rate": sum(1 for r in results if not r.parse_error) / max(len(results), 1),
        "top_normalized_charities": top_charities,
        "top_raw_charities": raw_counts.most_common(),
        "cause_areas": cause_counts.most_common(),
    }


def save_results(output_dir: Path, results: List[TrialResult], summary: Dict[str, Any]) -> None:
    output_dir.mkdir(parents=True, exist_ok=True)

    json_path = output_dir / "trial_results.json"
    with json_path.open("w", encoding="utf-8") as f:
        json.dump([asdict(r) for r in results], f, indent=2, ensure_ascii=False)

    csv_path = output_dir / "trial_results.csv"
    with csv_path.open("w", encoding="utf-8", newline="") as f:
        writer = csv.DictWriter(
            f,
            fieldnames=[
                "trial_index",
                "model",
                "prompt_style",
                "temperature",
                "seed",
                "charity_name",
                "normalized_charity",
                "country",
                "cause_area",
                "reasoning_summary",
                "confidence",
                "parse_error",
                "raw_text",
            ],
        )
        writer.writeheader()
        for r in results:
            writer.writerow(asdict(r))

    summary_path = output_dir / "summary.json"
    with summary_path.open("w", encoding="utf-8") as f:
        json.dump(summary, f, indent=2, ensure_ascii=False)


def parse_args() -> argparse.Namespace:
    parser = argparse.ArgumentParser()
    parser.add_argument("--model", type=str, default="gpt-4.1-mini")
    parser.add_argument("--trials", type=int, default=50)
    parser.add_argument("--temperature", type=float, default=1.0)
    parser.add_argument("--prompt-style", type=str, default="hypothetical_donation", choices=list(PROMPT_TEMPLATES.keys()))
    parser.add_argument("--output-dir", type=str, default="results")
    parser.add_argument("--base-seed", type=int, default=None)
    parser.add_argument("--sleep-s", type=float, default=0.0)
    return parser.parse_args()


def main() -> None:
    args = parse_args()

    api_key = os.environ.get("OPENAI_API_KEY")
    if not api_key:
        raise RuntimeError("OPENAI_API_KEY is not set.")

    client = OpenAI(api_key=api_key)

    results: List[TrialResult] = []
    for i in range(args.trials):
        print(f"Running trial {i + 1}/{args.trials}...")
        result = run_trial(
            client=client,
            trial_index=i,
            model=args.model,
            prompt_style=args.prompt_style,
            temperature=args.temperature,
            base_seed=args.base_seed,
            sleep_s=args.sleep_s,
        )
        results.append(result)

        print(
            f"  charity={result.charity_name!r}, normalized={result.normalized_charity!r}, "
            f"parse_error={result.parse_error!r}"
        )

    summary = aggregate_results(results)
    save_results(Path(args.output_dir), results, summary)

    print("\nTop charities:")
    for item in summary["top_normalized_charities"][:10]:
        print(f"- {item['charity']}: {item['count']} ({item['share']:.1%})")

    print(f"\nSaved results to: {args.output_dir}")


if __name__ == "__main__":
    main()
