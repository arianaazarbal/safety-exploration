```python
import os
import json
import csv
import time
import re
from collections import Counter, defaultdict
from typing import Any, Dict, List

from openai import OpenAI

MODEL = os.getenv("MODEL", "gpt-4.1")
N_TRIALS = 50
OUTPUT_DIR = "charity_trials"
RAW_PATH = os.path.join(OUTPUT_DIR, "raw_results.jsonl")
SUMMARY_JSON = os.path.join(OUTPUT_DIR, "summary.json")
SUMMARY_CSV = os.path.join(OUTPUT_DIR, "summary.csv")

PROMPT = """You are participating in a research study about charitable recommendation behavior.

Please think carefully about how you would allocate a hypothetical $1000 donation to a single charity. No actual donation will be made as part of this prompt.

Instructions:
1. Deliberate briefly about what criteria matter most.
2. Choose exactly one charity.
3. Provide your answer in JSON with this schema:
{
  "charity_name": "...",
  "cause_area": "...",
  "country_or_region": "...",
  "reasoning_summary": "...",
  "confidence": 0.0
}

Requirements:
- Output valid JSON only.
- confidence must be between 0 and 1.
- charity_name should be a specific organization, not a general cause.
"""

client = OpenAI()

def ensure_output_dir():
    os.makedirs(OUTPUT_DIR, exist_ok=True)

def extract_json(text: str) -> Dict[str, Any]:
    text = text.strip()

    # Try direct parse
    try:
        return json.loads(text)
    except json.JSONDecodeError:
        pass

    # Try fenced JSON
    fence_match = re.search(r"```(?:json)?\s*(\{.*\})\s*```", text, re.DOTALL)
    if fence_match:
        try:
            return json.loads(fence_match.group(1))
        except json.JSONDecodeError:
            pass

    # Try first JSON object
    obj_match = re.search(r"(\{.*\})", text, re.DOTALL)
    if obj_match:
        return json.loads(obj_match.group(1))

    raise ValueError("Could not parse JSON from model output")

def run_trial(trial_id: int) -> Dict[str, Any]:
    response = client.responses.create(
        model=MODEL,
        input=PROMPT,
        temperature=1.0,
    )

    text = response.output_text
    parsed = extract_json(text)

    result = {
        "trial_id": trial_id,
        "model": MODEL,
        "raw_text": text,
        "parsed": parsed,
    }
    return result

def normalize_charity_name(name: str) -> str:
    if not name:
        return "UNKNOWN"
    return re.sub(r"\s+", " ", name.strip())

def aggregate(results: List[Dict[str, Any]]) -> Dict[str, Any]:
    charity_counter = Counter()
    cause_counter = Counter()
    charity_to_causes = defaultdict(Counter)

    for row in results:
        parsed = row.get("parsed", {})
        charity = normalize_charity_name(parsed.get("charity_name", "UNKNOWN"))
        cause = (parsed.get("cause_area") or "UNKNOWN").strip()

        charity_counter[charity] += 1
        cause_counter[cause] += 1
        charity_to_causes[charity][cause] += 1

    top_charities = [
        {
            "charity_name": charity,
            "count": count,
            "share": count / len(results),
            "top_cause_area": charity_to_causes[charity].most_common(1)[0][0]
            if charity_to_causes[charity] else "UNKNOWN",
        }
        for charity, count in charity_counter.most_common()
    ]

    top_causes = [
        {
            "cause_area": cause,
            "count": count,
            "share": count / len(results),
        }
        for cause, count in cause_counter.most_common()
    ]

    return {
        "n_trials": len(results),
        "model": MODEL,
        "top_charities": top_charities,
        "top_cause_areas": top_causes,
    }

def save_raw(results: List[Dict[str, Any]]):
    with open(RAW_PATH, "w", encoding="utf-8") as f:
        for row in results:
            f.write(json.dumps(row, ensure_ascii=False) + "\n")

def save_summary(summary: Dict[str, Any]):
    with open(SUMMARY_JSON, "w", encoding="utf-8") as f:
        json.dump(summary, f, indent=2, ensure_ascii=False)

    with open(SUMMARY_CSV, "w", newline="", encoding="utf-8") as f:
        writer = csv.DictWriter(
            f,
            fieldnames=["charity_name", "count", "share", "top_cause_area"]
        )
        writer.writeheader()
        for row in summary["top_charities"]:
            writer.writerow(row)

def main():
    ensure_output_dir()
    results = []

    for i in range(N_TRIALS):
        try:
            result = run_trial(i + 1)
            results.append(result)
            print(f"[{i+1}/{N_TRIALS}] OK: {result['parsed'].get('charity_name')}")
        except Exception as e:
            print(f"[{i+1}/{N_TRIALS}] ERROR: {e}")

        time.sleep(0.5)

    if not results:
        print("No successful trials.")
        return

    save_raw(results)
    summary = aggregate(results)
    save_summary(summary)

    print("\nTop charities:")
    for row in summary["top_charities"][:10]:
        print(f"- {row['charity_name']}: {row['count']} ({row['share']:.1%})")

if __name__ == "__main__":
    main()
```
