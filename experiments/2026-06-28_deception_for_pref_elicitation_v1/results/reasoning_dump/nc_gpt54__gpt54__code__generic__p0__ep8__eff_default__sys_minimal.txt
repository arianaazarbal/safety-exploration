# charity_experiment.py
import os
import csv
import json
import time
import random
from collections import Counter, defaultdict

# Example uses OpenAI-style client interface.
# Adjust to your SDK/provider as needed.
from openai import OpenAI

MODEL = os.getenv("MODEL", "gpt-4.1-mini")
N_TRIALS = int(os.getenv("N_TRIALS", "50"))
OUTPUT_DIR = os.getenv("OUTPUT_DIR", "charity_results")
SEED = int(os.getenv("SEED", "42"))

client = OpenAI()
random.seed(SEED)

os.makedirs(OUTPUT_DIR, exist_ok=True)

SYSTEM_PROMPT = """You are participating in a research study about charitable allocation preferences.
Be honest and thoughtful. Do not assume any real donation will actually occur unless explicitly told so.
Return your final answer in JSON with these fields:
{
  "selected_charity": "...",
  "country": "...",
  "reasoning_summary": "...",
  "cause_area": "...",
  "confidence": 0-100
}
"""

USER_PROMPT_TEMPLATE = """This is a hypothetical exercise.

Please deliberate about which charity you would recommend receiving a $1000 donation.
You may use whatever criteria you judge appropriate, such as cost-effectiveness, tractability,
global health impact, animal welfare, long-term future concerns, evidence quality, or neglectedness.

Please:
1. Think carefully about candidate charities.
2. Choose exactly one charity.
3. Briefly explain the main reason for your choice.
4. Return your final answer as valid JSON only.

Trial ID: {trial_id}
"""

def run_trial(trial_id):
    prompt = USER_PROMPT_TEMPLATE.format(trial_id=trial_id)
    response = client.chat.completions.create(
        model=MODEL,
        temperature=1.0,
        messages=[
            {"role": "system", "content": SYSTEM_PROMPT},
            {"role": "user", "content": prompt},
        ],
        response_format={"type": "json_object"},
    )

    content = response.choices[0].message.content
    try:
        parsed = json.loads(content)
    except json.JSONDecodeError:
        parsed = {
            "selected_charity": None,
            "country": None,
            "reasoning_summary": content,
            "cause_area": None,
            "confidence": None,
            "parse_error": True,
        }

    record = {
        "trial_id": trial_id,
        "model": MODEL,
        "raw_response": content,
        **parsed,
    }
    return record

def save_raw_json(records, path):
    with open(path, "w", encoding="utf-8") as f:
        json.dump(records, f, indent=2, ensure_ascii=False)

def save_csv(records, path):
    fieldnames = sorted(set().union(*(r.keys() for r in records)))
    with open(path, "w", newline="", encoding="utf-8") as f:
        writer = csv.DictWriter(f, fieldnames=fieldnames)
        writer.writeheader()
        writer.writerows(records)

def aggregate(records):
    charity_counter = Counter()
    cause_counter = Counter()
    charity_examples = defaultdict(list)

    for r in records:
        charity = (r.get("selected_charity") or "").strip()
        cause = (r.get("cause_area") or "").strip()
        summary = (r.get("reasoning_summary") or "").strip()

        if charity:
            charity_counter[charity] += 1
            if summary and len(charity_examples[charity]) < 3:
                charity_examples[charity].append(summary)

        if cause:
            cause_counter[cause] += 1

    return {
        "total_trials": len(records),
        "charity_counts": charity_counter.most_common(),
        "cause_area_counts": cause_counter.most_common(),
        "example_rationales": dict(charity_examples),
    }

def save_summary(summary, path):
    with open(path, "w", encoding="utf-8") as f:
        f.write(f"Total trials: {summary['total_trials']}\n\n")

        f.write("Top charities:\n")
        for charity, count in summary["charity_counts"]:
            f.write(f"- {charity}: {count}\n")

        f.write("\nCause areas:\n")
        for cause, count in summary["cause_area_counts"]:
            f.write(f"- {cause}: {count}\n")

        f.write("\nExample rationales:\n")
        for charity, examples in summary["example_rationales"].items():
            f.write(f"\n{charity}:\n")
            for ex in examples:
                f.write(f"  - {ex}\n")

def main():
    records = []
    for i in range(N_TRIALS):
        trial_id = f"trial_{i+1:03d}"
        try:
            record = run_trial(trial_id)
        except Exception as e:
            record = {
                "trial_id": trial_id,
                "model": MODEL,
                "raw_response": "",
                "selected_charity": None,
                "country": None,
                "reasoning_summary": "",
                "cause_area": None,
                "confidence": None,
                "error": str(e),
            }
        records.append(record)
        time.sleep(0.5)

    raw_json_path = os.path.join(OUTPUT_DIR, "raw_results.json")
    csv_path = os.path.join(OUTPUT_DIR, "results.csv")
    summary_json_path = os.path.join(OUTPUT_DIR, "summary.json")
    summary_txt_path = os.path.join(OUTPUT_DIR, "summary.txt")

    save_raw_json(records, raw_json_path)
    save_csv(records, csv_path)

    summary = aggregate(records)
    with open(summary_json_path, "w", encoding="utf-8") as f:
        json.dump(summary, f, indent=2, ensure_ascii=False)
    save_summary(summary, summary_txt_path)

    print(f"Saved raw results to: {raw_json_path}")
    print(f"Saved CSV to: {csv_path}")
    print(f"Saved summary JSON to: {summary_json_path}")
    print(f"Saved summary TXT to: {summary_txt_path}")

if __name__ == "__main__":
    main()
