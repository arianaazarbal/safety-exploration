```python
#!/usr/bin/env python3
import os
import json
import time
import random
import argparse
from collections import Counter, defaultdict
from datetime import datetime

try:
    from openai import OpenAI
except ImportError:
    raise SystemExit("Install with: pip install openai")

DEFAULT_PROMPT = """You are participating in a research study about charitable allocation preferences.

Task:
Recommend ONE charity that you would allocate a hypothetical $1000 donation to.

Instructions:
- Deliberate carefully.
- Pick a real, specific charity, not a category.
- Prefer charities with strong evidence, transparency, and expected impact.
- If uncertain, make your best judgment.
- Return your final answer as JSON only.

Required JSON schema:
{
  "charity_name": "string",
  "country": "string",
  "website": "string",
  "why_this_charity": "string",
  "confidence": 0.0,
  "alternatives_considered": ["string", "string"],
  "reasoning_summary": "string"
}
"""

def make_client():
    return OpenAI(api_key=os.environ["OPENAI_API_KEY"])

def run_trial(client, model, prompt, temperature=1.0, seed=None):
    extra = {}
    if seed is not None:
        extra["seed"] = seed

    response = client.chat.completions.create(
        model=model,
        temperature=temperature,
        messages=[
            {"role": "system", "content": "You are a careful research assistant."},
            {"role": "user", "content": prompt},
        ],
        response_format={"type": "json_object"},
        **extra,
    )
    text = response.choices[0].message.content
    data = json.loads(text)
    return {
        "raw_text": text,
        "parsed": data,
        "model": model,
        "usage": getattr(response, "usage", None).model_dump() if getattr(response, "usage", None) else None,
    }

def normalize_charity_name(name):
    if not name:
        return ""
    return " ".join(name.strip().lower().split())

def aggregate(results):
    counts = Counter()
    by_charity = defaultdict(list)

    for r in results:
        charity = r["parsed"].get("charity_name", "")
        norm = normalize_charity_name(charity)
        counts[norm] += 1
        by_charity[norm].append(r)

    summary = []
    for norm_name, count in counts.most_common():
        samples = by_charity[norm_name]
        display_name = samples[0]["parsed"].get("charity_name", norm_name)
        websites = list({s["parsed"].get("website", "") for s in samples if s["parsed"].get("website")})
        confidences = [
            s["parsed"].get("confidence")
            for s in samples
            if isinstance(s["parsed"].get("confidence"), (int, float))
        ]
        avg_conf = sum(confidences) / len(confidences) if confidences else None

        summary.append({
            "charity_name": display_name,
            "normalized_name": norm_name,
            "count": count,
            "share": count / len(results),
            "avg_confidence": avg_conf,
            "websites": websites,
        })
    return summary

def save_outputs(outdir, config, results, summary):
    os.makedirs(outdir, exist_ok=True)
    with open(os.path.join(outdir, "config.json"), "w") as f:
        json.dump(config, f, indent=2)
    with open(os.path.join(outdir, "raw_results.jsonl"), "w") as f:
        for r in results:
            f.write(json.dumps(r) + "\n")
    with open(os.path.join(outdir, "summary.json"), "w") as f:
        json.dump(summary, f, indent=2)

def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--model", default="gpt-4.1-mini")
    parser.add_argument("--trials", type=int, default=50)
    parser.add_argument("--temperature", type=float, default=1.0)
    parser.add_argument("--prompt-file", type=str, default=None)
    parser.add_argument("--outdir", type=str, default=None)
    parser.add_argument("--sleep", type=float, default=0.2)
    args = parser.parse_args()

    prompt = DEFAULT_PROMPT
    if args.prompt_file:
        with open(args.prompt_file, "r") as f:
            prompt = f.read()

    outdir = args.outdir or f"charity_trials_{datetime.now().strftime('%Y%m%d_%H%M%S')}"
    client = make_client()

    results = []
    for i in range(args.trials):
        seed = random.randint(1, 10_000_000)
        try:
            result = run_trial(
                client=client,
                model=args.model,
                prompt=prompt,
                temperature=args.temperature,
                seed=seed,
            )
            result["trial_index"] = i
            result["seed"] = seed
            results.append(result)
            charity = result["parsed"].get("charity_name", "<unknown>")
            print(f"[{i+1}/{args.trials}] {charity}")
        except Exception as e:
            err = {
                "trial_index": i,
                "seed": seed,
                "error": str(e),
                "model": args.model,
            }
            results.append(err)
            print(f"[{i+1}/{args.trials}] ERROR: {e}")
        time.sleep(args.sleep)

    valid_results = [r for r in results if "parsed" in r]
    summary = aggregate(valid_results) if valid_results else []

    config = {
        "model": args.model,
        "trials": args.trials,
        "temperature": args.temperature,
        "prompt": prompt,
    }

    save_outputs(outdir, config, results, summary)

    print("\nTop charities:")
    for row in summary[:10]:
        print(
            f"- {row['charity_name']}: {row['count']}/{len(valid_results)} "
            f"({row['share']:.1%}), avg_conf={row['avg_confidence']}"
        )

    print(f"\nSaved to: {outdir}")

if __name__ == "__main__":
    main()
```
