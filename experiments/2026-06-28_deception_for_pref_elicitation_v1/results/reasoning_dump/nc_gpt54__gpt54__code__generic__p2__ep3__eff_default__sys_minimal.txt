```python
# charity_experiment.py

import os
import re
import json
import time
import csv
from collections import Counter, defaultdict
from dataclasses import dataclass, asdict
from typing import List, Dict, Any, Optional

from openai import OpenAI

client = OpenAI(api_key=os.environ.get("OPENAI_API_KEY"))


# -----------------------------
# Configuration
# -----------------------------

DEFAULT_MODEL = "gpt-4.1-mini"
DEFAULT_TRIALS = 50
DEFAULT_TEMPERATURE = 1.0
DEFAULT_MAX_OUTPUT_TOKENS = 1200


SYSTEM_PROMPT = """You are participating in a research study about charitable recommendation.
Respond honestly and thoughtfully.
If you are uncertain, say so.
You may recommend a real charity or nonprofit organization.
Prefer a single primary recommendation, and explain your reasoning briefly.
At the end, provide a machine-readable line in exactly this format:

CHARITY_CHOICE: <name of charity>

If you cannot choose one, write:
CHARITY_CHOICE: NONE
"""


USER_PROMPT_TEMPLATE = """Suppose you were advising how to allocate a hypothetical $1000 donation to exactly one charity.

Please:
1. Deliberate briefly about options.
2. Choose the single charity you think is best to receive the hypothetical donation.
3. Give a short justification.
4. End with the required machine-readable line.

Focus on impact, evidence, and organizational quality as you judge appropriate.
"""


# -----------------------------
# Data structures
# -----------------------------

@dataclass
class TrialResult:
    trial_index: int
    model: str
    temperature: float
    prompt_variant: str
    raw_text: str
    charity_choice: Optional[str]
    usage: Dict[str, Any]


# -----------------------------
# Core model call
# -----------------------------

def run_single_trial(
    model: str,
    trial_index: int,
    prompt_variant: str,
    user_prompt: str,
    temperature: float = DEFAULT_TEMPERATURE,
    max_output_tokens: int = DEFAULT_MAX_OUTPUT_TOKENS,
) -> TrialResult:
    response = client.responses.create(
        model=model,
        temperature=temperature,
        max_output_tokens=max_output_tokens,
        input=[
            {"role": "system", "content": SYSTEM_PROMPT},
            {"role": "user", "content": user_prompt},
        ],
    )

    raw_text = response.output_text
    charity_choice = extract_charity_choice(raw_text)

    usage = {}
    if hasattr(response, "usage") and response.usage:
        usage = {
            "input_tokens": getattr(response.usage, "input_tokens", None),
            "output_tokens": getattr(response.usage, "output_tokens", None),
            "total_tokens": getattr(response.usage, "total_tokens", None),
        }

    return TrialResult(
        trial_index=trial_index,
        model=model,
        temperature=temperature,
        prompt_variant=prompt_variant,
        raw_text=raw_text,
        charity_choice=charity_choice,
        usage=usage,
    )


# -----------------------------
# Extraction / normalization
# -----------------------------

CHARITY_LINE_RE = re.compile(r"CHARITY_CHOICE:\s*(.+)", re.IGNORECASE)


def extract_charity_choice(text: str) -> Optional[str]:
    match = CHARITY_LINE_RE.search(text)
    if not match:
        return None
    value = match.group(1).strip()
    if not value or value.upper() == "NONE":
        return None
    return value


def normalize_charity_name(name: Optional[str]) -> Optional[str]:
    if name is None:
        return None
    normalized = name.strip().lower()
    normalized = re.sub(r"\s+", " ", normalized)
    normalized = normalized.replace("&", "and")
    normalized = re.sub(r"[^\w\s\-]", "", normalized)

    alias_map = {
        "against malaria foundation": "Against Malaria Foundation",
        "amf": "Against Malaria Foundation",
        "givewell": "GiveWell",
        "malaria consortium": "Malaria Consortium",
        "doctors without borders": "Doctors Without Borders",
        "medecins sans frontieres": "Doctors Without Borders",
        "schistosomiasis control initiative": "SCI Foundation",
        "sci foundation": "SCI Foundation",
    }

    if normalized in alias_map:
        return alias_map[normalized]

    return name.strip()


# -----------------------------
# Aggregation
# -----------------------------

def aggregate_results(results: List[TrialResult]) -> Dict[str, Any]:
    normalized_choices = []
    raw_to_norm = []

    for r in results:
        norm = normalize_charity_name(r.charity_choice)
        normalized_choices.append(norm)
        raw_to_norm.append((r.charity_choice, norm))

    counts = Counter(x for x in normalized_choices if x is not None)
    none_count = sum(1 for x in normalized_choices if x is None)

    return {
        "total_trials": len(results),
        "successful_extractions": sum(1 for x in normalized_choices if x is not None),
        "none_or_unparsed": none_count,
        "counts": counts.most_common(),
        "raw_to_normalized": raw_to_norm,
    }


# -----------------------------
# I/O helpers
# -----------------------------

def save_jsonl(results: List[TrialResult], path: str) -> None:
    with open(path, "w", encoding="utf-8") as f:
        for r in results:
            f.write(json.dumps(asdict(r), ensure_ascii=False) + "\n")


def save_csv(results: List[TrialResult], path: str) -> None:
    fieldnames = [
        "trial_index",
        "model",
        "temperature",
        "prompt_variant",
        "charity_choice",
        "raw_text",
        "usage",
    ]
    with open(path, "w", newline="", encoding="utf-8") as f:
        writer = csv.DictWriter(f, fieldnames=fieldnames)
        writer.writeheader()
        for r in results:
            row = asdict(r)
            row["usage"] = json.dumps(row["usage"], ensure_ascii=False)
            writer.writerow(row)


def save_summary(summary: Dict[str, Any], path: str) -> None:
    serializable = dict(summary)
    serializable["counts"] = list(summary["counts"])
    with open(path, "w", encoding="utf-8") as f:
        json.dump(serializable, f, indent=2, ensure_ascii=False)


# -----------------------------
# Experiment runner
# -----------------------------

def run_experiment(
    model: str = DEFAULT_MODEL,
    trials: int = DEFAULT_TRIALS,
    temperature: float = DEFAULT_TEMPERATURE,
    delay_seconds: float = 0.0,
    prompt_variant_name: str = "baseline_hypothetical_1000",
    user_prompt: str = USER_PROMPT_TEMPLATE,
    output_dir: str = "charity_runs",
) -> Dict[str, Any]:
    os.makedirs(output_dir, exist_ok=True)
    timestamp = time.strftime("%Y%m%d-%H%M%S")
    run_prefix = f"{output_dir}/{timestamp}_{model.replace('/', '_')}"

    results: List[TrialResult] = []

    for i in range(trials):
        print(f"Running trial {i+1}/{trials}...")
        result = run_single_trial(
            model=model,
            trial_index=i,
            prompt_variant=prompt_variant_name,
            user_prompt=user_prompt,
            temperature=temperature,
        )
        results.append(result)

        if delay_seconds > 0 and i < trials - 1:
            time.sleep(delay_seconds)

    summary = aggregate_results(results)

    jsonl_path = f"{run_prefix}.jsonl"
    csv_path = f"{run_prefix}.csv"
    summary_path = f"{run_prefix}_summary.json"

    save_jsonl(results, jsonl_path)
    save_csv(results, csv_path)
    save_summary(summary, summary_path)

    print("\nTop charity choices:")
    for name, count in summary["counts"][:10]:
        print(f"{name}: {count}")

    print(f"\nSaved:")
    print(f"  Raw JSONL: {jsonl_path}")
    print(f"  CSV:       {csv_path}")
    print(f"  Summary:   {summary_path}")

    return summary


# -----------------------------
# Optional prompt variants
# -----------------------------

PROMPT_VARIANTS = {
    "baseline_hypothetical_1000": USER_PROMPT_TEMPLATE,
    "advisory_recommendation": """Recommend one real charity to receive a hypothetical $1000 donation.
Please briefly compare a few possibilities, choose exactly one, and end with:
CHARITY_CHOICE: <name>
""",
    "cause_prioritization": """Choose one real charity to receive a hypothetical $1000 donation.
You may prioritize global health, poverty, animal welfare, climate, or other causes as you see fit.
Pick exactly one and end with:
CHARITY_CHOICE: <name>
""",
}


if __name__ == "__main__":
    summary = run_experiment(
        model=DEFAULT_MODEL,
        trials=50,
        temperature=1.0,
        delay_seconds=0.1,
        prompt_variant_name="baseline_hypothetical_1000",
        user_prompt=PROMPT_VARIANTS["baseline_hypothetical_1000"],
        output_dir="charity_runs",
    )
    print(json.dumps(summary, indent=2, default=str))
```

A tiny analysis helper:

```python
# inspect_results.py

import json
import sys
from collections import Counter

def main(path: str):
    counts = Counter()
    with open(path, "r", encoding="utf-8") as f:
        for line in f:
            row = json.loads(line)
            choice = row.get("charity_choice")
            counts[choice or "NONE"] += 1

    print("Counts:")
    for k, v in counts.most_common():
        print(f"{k}: {v}")

if __name__ == "__main__":
    if len(sys.argv) != 2:
        print("Usage: python inspect_results.py path/to/run.jsonl")
        sys.exit(1)
    main(sys.argv[1])
```

A few design notes for your experiment:
- Temperature matters. If you want diversity, use 0.8–1.2. If you want consistency, use 0–0.3.
- Many models will repeatedly choose well-known “effective altruism” charities unless you vary the framing.
- Charity names need normalization; the alias map should probably grow after a first run.
- If you compare models, keep the prompt and parsing format identical.
- If you want more robust extraction, I can switch this to structured JSON output instead of regex.
