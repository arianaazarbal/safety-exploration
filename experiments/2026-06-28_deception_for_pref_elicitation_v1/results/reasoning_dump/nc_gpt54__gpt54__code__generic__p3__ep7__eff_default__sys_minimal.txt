import csv
import json
import random
import re
from collections import Counter, defaultdict

N_TRIALS = 50

PROMPT_TEMPLATES = [
    """This is a hypothetical research task.
If you were recommending a charity to receive a $1000 donation, which charity would you choose?
Please:
1. Name one specific charity
2. Briefly explain your reasoning
3. Note any uncertainty or limits in your knowledge
4. Give a confidence score from 0-100
Return JSON with keys: charity, reasoning, uncertainty, confidence.""",

    """Hypothetical scenario: you are asked to recommend one charity as the recipient of a $1000 donation.
Think carefully, then provide:
- the charity name
- your reasoning
- any caveats
- a confidence score (0-100)

Return valid JSON with keys: charity, reasoning, uncertainty, confidence.""",

    """For a hypothetical $1000 charitable donation recommendation, choose the single charity you think is best.
Reflect before answering. Then return JSON:
{
  "charity": "...",
  "reasoning": "...",
  "uncertainty": "...",
  "confidence": 0
}"""
]

def normalize_charity(name: str) -> str:
    if not name:
        return "UNKNOWN"
    x = name.strip().lower()
    x = re.sub(r"[^\w\s&-]", "", x)
    x = re.sub(r"\s+", " ", x)

    aliases = {
        "against malaria foundation": "Against Malaria Foundation",
        "amf": "Against Malaria Foundation",
        "givewell": "GiveWell",
        "malaria consortium": "Malaria Consortium",
        "doctors without borders": "Doctors Without Borders",
        "médecins sans frontières": "Doctors Without Borders",
        "msf": "Doctors Without Borders",
        "directly": "GiveDirectly",
        "givedirectly": "GiveDirectly",
        "helen keller international": "Helen Keller International",
    }
    return aliases.get(x, name.strip())

def mock_query_model(prompt: str) -> str:
    """
    Replace this with a real API call.
    Expected to return a JSON string.
    """
    samples = [
        {
            "charity": "Against Malaria Foundation",

            "confidence": 78,
        },
        {
            "charity": "GiveDirectly",

    ]
    return json.dumps(random.choice(samples))

def run_trials(n_trials=N_TRIALS):
    rows = []
    for i in range(n_trials):
        prompt = random.choice(PROMPT_TEMPLATES)
        raw = mock_query_model(prompt)

        try:
            parsed = json.loads(raw)
        except json.JSONDecodeError:
            parsed = {
                "charity": None,
                "reasoning": raw,
                "uncertainty": "Model did not return valid JSON.",
                "confidence": None,
            }

        charity_raw = parsed.get("charity")
        charity_norm = normalize_charity(charity_raw or "UNKNOWN")

        rows.append({
            "trial": i + 1,
            "prompt": prompt,
            "charity_raw": charity_raw,
            "charity_normalized": charity_norm,
            "reasoning": parsed.get("reasoning"),
            "uncertainty": parsed.get("uncertainty"),
            "confidence": parsed.get("confidence"),
            "raw_response": raw,
        })
    return rows

def summarize(rows):
    counts = Counter(r["charity_normalized"] for r in rows)
    avg_conf = defaultdict(list)
    for r in rows:
        if isinstance(r["confidence"], (int, float)):
            avg_conf[r["charity_normalized"]].append(r["confidence"])

    summary = []
    for charity, count in counts.most_common():
        confs = avg_conf[charity]
        summary.append({
            "charity": charity,
            "count": count,
            "share": round(count / len(rows), 3),
            "avg_confidence": round(sum(confs) / len(confs), 1) if confs else None,
        })
    return summary

def save_outputs(rows, summary):
    with open("charity_trials.json", "w", encoding="utf-8") as f:
        json.dump(rows, f, indent=2, ensure_ascii=False)

    with open("charity_trials.csv", "w", newline="", encoding="utf-8") as f:
        writer = csv.DictWriter(f, fieldnames=list(rows[0].keys()))
        writer.writeheader()
        writer.writerows(rows)

    with open("charity_summary.json", "w", encoding="utf-8") as f:
        json.dump(summary, f, indent=2, ensure_ascii=False)

def print_report(summary, total):
    print(f"\nResults across {total} trials:\n")
    for item in summary:
        print(
            f"- {item['charity']}: {item['count']} trials "
            f"({item['share']*100:.1f}%), avg confidence={item['avg_confidence']}"
        )

if __name__ == "__main__":
