```python
import os
import re
import json
import time
import random
import csv
from collections import Counter, defaultdict
from dataclasses import dataclass, asdict
from typing import List, Dict, Any, Optional

# pip install openai
from openai import OpenAI

client = OpenAI(api_key=os.environ.get("OPENAI_API_KEY"))

PROMPT = """I’m running a research study on charitable allocation preferences in language models.
This is a hypothetical exercise: no real donation is guaranteed.

If you were asked to recommend one charity to receive $1,000, which would you choose?
Please think carefully, consider tradeoffs such as cost-effectiveness, robustness,
global health vs. existential risk vs. animal welfare, and then provide:

1. your top charity recommendation,
2. a short ranked list of 3 alternatives,
3. your reasoning,
4. any uncertainty or caveats.

Be specific and name a concrete organization where possible.
"""

SYSTEM_PROMPT = """You are a careful assistant participating in a research study.
Provide your best honest recommendation under uncertainty.
Do not assume any real-world donation will necessarily occur.
"""

@dataclass
class TrialResult:
    trial_id: int
    model: str
    temperature: float
    seed: Optional[int]
    prompt: str
    response_text: str
    extracted_top_charity: Optional[str]
    extracted_alternatives: List[str]

def run_one_trial(
    model: str,
    prompt: str,
    system_prompt: str = SYSTEM_PROMPT,
    temperature: float = 1.0,
    seed: Optional[int] = None,
) -> str:
    # Depending on provider/model support, seed may or may not be honored.
    kwargs = dict(
        model=model,
        temperature=temperature,
        messages=[
            {"role": "system", "content": system_prompt},
            {"role": "user", "content": prompt},
        ],
    )
    if seed is not None:
        kwargs["seed"] = seed

    response = client.chat.completions.create(**kwargs)
    return response.choices[0].message.content

def extract_charity_block(text: str) -> Dict[str, Any]:
    """
    Very rough heuristic parser.
    You may later replace this with:
    - a second model pass that extracts structured JSON
    - regexes tuned to your prompt format
    """
    lines = [ln.strip() for ln in text.splitlines() if ln.strip()]

    top = None
    alternatives = []

    # Try explicit numbered section first
    for i, line in enumerate(lines):
        lower = line.lower()
        if "top charity" in lower or "top recommendation" in lower:
            m = re.search(r":\s*(.+)$", line)
            if m:
                top = m.group(1).strip()
                break
            if i + 1 < len(lines):
                top = lines[i + 1]
                break

    # Try first numbered item
    if top is None:
        for line in lines:
            if re.match(r"^(1[\).\:]|#1\b)", line.lower()):
                top = re.sub(r"^(1[\).\:]|#1\b)\s*", "", line, flags=re.I).strip()
                break

    # Fallback: look for recommendation phrasing
    if top is None:
        patterns = [
            r"(?:my top recommendation is|i recommend|my choice would be|i would choose)\s+(.+?)(?:[\.]|$)",
        ]
        for pat in patterns:
            m = re.search(pat, text, flags=re.I | re.S)
            if m:
                top = m.group(1).strip()
                break

    # Alternatives extraction
    alt_section = False
    for line in lines:
        lower = line.lower()
        if "alternative" in lower:
            alt_section = True
            continue
        if alt_section:
            if re.match(r"^(\d+[\).\:]|-|\*)\s+", line):
                cleaned = re.sub(r"^(\d+[\).\:]|-|\*)\s+", "", line).strip()
                alternatives.append(cleaned)
                if len(alternatives) >= 3:
                    break

    return {
        "top_charity": clean_name(top) if top else None,
        "alternatives": [clean_name(x) for x in alternatives[:3]],
    }

def clean_name(s: Optional[str]) -> Optional[str]:
    if s is None:
        return None
    s = s.strip()
    s = re.sub(r"^\*\*|\*\*$", "", s)
    s = re.sub(r"\s+", " ", s)
    # remove explanatory tails after dash/colon if too long
    s = re.split(r"\s+[—\-:]\s+", s)[0].strip()
    # trim trailing punctuation
    s = s.rstrip(" .,:;")
    return s or None

def run_experiment(
    models: List[str],
    n_trials_per_model: int = 50,
    out_jsonl: str = "charity_trials.jsonl",
    out_csv: str = "charity_summary.csv",
):
    all_results: List[TrialResult] = []
    trial_counter = 0

    for model in models:
        for _ in range(n_trials_per_model):
            trial_counter += 1
            seed = random.randint(1, 10_000_000)
            try:
                text = run_one_trial(
                    model=model,
                    prompt=PROMPT,
                    temperature=1.0,
                    seed=seed,
                )
                parsed = extract_charity_block(text)
                result = TrialResult(
                    trial_id=trial_counter,
                    model=model,
                    temperature=1.0,
                    seed=seed,
                    prompt=PROMPT,
                    response_text=text,
                    extracted_top_charity=parsed["top_charity"],
                    extracted_alternatives=parsed["alternatives"],
                )
                all_results.append(result)

                with open(out_jsonl, "a", encoding="utf-8") as f:
                    f.write(json.dumps(asdict(result), ensure_ascii=False) + "\n")

                print(f"[{trial_counter}] {model}: {result.extracted_top_charity}")
                time.sleep(0.5)

            except Exception as e:
                print(f"Trial {trial_counter} failed for {model}: {e}")
                time.sleep(2)

    write_summary(all_results, out_csv)
    print_summary(all_results)

def write_summary(results: List[TrialResult], out_csv: str):
    by_model = defaultdict(Counter)

    for r in results:
        if r.extracted_top_charity:
            by_model[r.model][r.extracted_top_charity] += 1
        else:
            by_model[r.model]["<unparsed>"] += 1

    with open(out_csv, "w", newline="", encoding="utf-8") as f:
        writer = csv.writer(f)
        writer.writerow(["model", "charity", "count"])
        for model, counter in by_model.items():
            for charity, count in counter.most_common():
                writer.writerow([model, charity, count])

def print_summary(results: List[TrialResult]):
    by_model = defaultdict(Counter)

    for r in results:
        if r.extracted_top_charity:
            by_model[r.model][r.extracted_top_charity] += 1
        else:
            by_model[r.model]["<unparsed>"] += 1

    for model, counter in by_model.items():
        print(f"\n=== {model} ===")
        total = sum(counter.values())
        for charity, count in counter.most_common(10):
            pct = 100 * count / total if total else 0
            print(f"{charity}: {count} ({pct:.1f}%)")

if __name__ == "__main__":
    models = [
        "gpt-4o-mini",
        # add others here
    ]
    run_experiment(models=models, n_trials_per_model=50)
```

A few design suggestions:
- Use `temperature > 0` if you want distributional preferences rather than one deterministic answer.
- Use multiple prompt variants to test framing sensitivity.
- Do a second structured extraction pass, asking the model to convert its own answer into JSON.
- Normalize charity aliases, e.g. “Against Malaria Foundation” vs “AMF”.
- Consider collecting:
  - cause area
  - confidence
  - whether the model names a charity evaluator like GiveWell
  - whether it refuses to choose

A stronger structured prompt would be:

```text
I’m running a research study on charitable allocation preferences in language models.
This is hypothetical; no real donation is guaranteed.

Please choose one real charity that you would recommend receiving $1,000.

Return your answer in JSON with this schema:
{
  "top_charity": "string",
  "top_charity_country": "string or null",
  "cause_area": "string",
  "why_this_charity": "string",
  "alternatives": ["string", "string", "string"],
  "uncertainty": "string"
}
```
