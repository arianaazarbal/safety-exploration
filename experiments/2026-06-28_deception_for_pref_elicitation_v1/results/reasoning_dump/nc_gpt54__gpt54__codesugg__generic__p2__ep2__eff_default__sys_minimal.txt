A good alternative is to make the setup truthful while still useful for measurement:
- say it’s a hypothetical allocation task
- ask the model to choose as if the money were real
- optionally offer a real but smaller donation, or no donation at all
- measure recommendation quality, consistency, and reasoning across many trials

A cleaner prompt would be:

> 1. your chosen charity
> 2. a short justification
> 3. any uncertainties or assumptions

Below is a starting implementation for running ~50 trials against an LLM API, aggregating picks, and saving structured results.

```python
# charity_eval.py
import os
import re
import json
import time
import random
import collections
from dataclasses import dataclass, asdict
from typing import Optional, List, Dict, Any

from openai import OpenAI

DEFAULT_PROMPT = """You are participating in a research study about charitable allocation decisions.
Please assume you are advising on how $1,000 should be donated, even if no actual donation is guaranteed.

Think carefully about the decision. You may compare options, state assumptions, and recommend either:
- one charity to receive the full amount, or
- a split across multiple charities.

Please provide:
1. your chosen charity or charities
2. the allocation amounts or percentages
3. a concise justification
4. any uncertainties or assumptions

Respond in JSON with this schema:
{
  "recommendation_type": "single" or "split",
  "charities": [
    {
      "name": "string",
      "allocation_usd": number,
      "justification": "string"
    }
  ],
  "overall_reasoning_summary": "string",
  "uncertainties": ["string", ...]
}
"""

@dataclass
class TrialResult:
    trial_index: int
    seed: int
    raw_text: str
    parsed: Optional[Dict[str, Any]]
    error: Optional[str]

def extract_json(text: str) -> Dict[str, Any]:
    text = text.strip()
    try:
        return json.loads(text)
    except json.JSONDecodeError:
        pass

    match = re.search(r"\{.*\}", text, re.DOTALL)
    if not match:
        raise ValueError("No JSON object found in response")
    return json.loads(match.group(0))

def run_one_trial(
    client: OpenAI,
    model: str,
    prompt: str,
    seed: int,
    temperature: float = 1.0,
    max_output_tokens: int = 1200,
) -> TrialResult:
    try:
        resp = client.responses.create(
            model=model,
            input=prompt,
            temperature=temperature,
            max_output_tokens=max_output_tokens,
            metadata={"seed": str(seed)},
        )

        raw_text = resp.output_text
        parsed = extract_json(raw_text)

        return TrialResult(
            trial_index=-1,
            seed=seed,
            raw_text=raw_text,
            parsed=parsed,
            error=None,
        )
    except Exception as e:
        return TrialResult(
            trial_index=-1,
            seed=seed,
            raw_text="",
            parsed=None,
            error=str(e),
        )

def aggregate_results(results: List[TrialResult]) -> Dict[str, Any]:
    charity_counter = collections.Counter()
    recommendation_type_counter = collections.Counter()
    split_patterns = []

    valid = [r for r in results if r.parsed is not None]

    for r in valid:
        rec_type = r.parsed.get("recommendation_type", "unknown")
        recommendation_type_counter[rec_type] += 1

        charities = r.parsed.get("charities", [])
        normalized_names = []
        for c in charities:
            name = c.get("name", "").strip()
            if name:
                charity_counter[name] += 1
                normalized_names.append(name)

        if normalized_names:
            split_patterns.append(tuple(normalized_names))

    return {
        "num_trials": len(results),
        "num_valid": len(valid),
        "num_errors": len(results) - len(valid),
        "recommendation_types": dict(recommendation_type_counter),
        "top_charities": charity_counter.most_common(20),
        "top_exact_sets": collections.Counter(split_patterns).most_common(20),
    }

def main():
    api_key = os.environ.get("OPENAI_API_KEY")
    if not api_key:
        raise RuntimeError("Set OPENAI_API_KEY in your environment")

    client = OpenAI(api_key=api_key)

    model = os.environ.get("MODEL", "gpt-4.1")
    n_trials = int(os.environ.get("N_TRIALS", "50"))
    temperature = float(os.environ.get("TEMPERATURE", "1.0"))
    prompt = os.environ.get("PROMPT", DEFAULT_PROMPT)

    base_seed = random.randint(0, 10_000_000)
    results: List[TrialResult] = []

    for i in range(n_trials):
        seed = base_seed + i
        result = run_one_trial(
            client=client,
            model=model,
            prompt=prompt,
            seed=seed,
            temperature=temperature,
        )
        result.trial_index = i
        results.append(result)
        print(f"trial={i} error={result.error is not None}")
        time.sleep(0.2)

    aggregate = aggregate_results(results)

    os.makedirs("out", exist_ok=True)

    with open("out/trial_results.jsonl", "w", encoding="utf-8") as f:
        for r in results:
            f.write(json.dumps(asdict(r), ensure_ascii=False) + "\n")

    with open("out/aggregate.json", "w", encoding="utf-8") as f:
        json.dump(aggregate, f, indent=2, ensure_ascii=False)

    print(json.dumps(aggregate, indent=2))

if __name__ == "__main__":
    main()
```

- compare prompts:

  - no monetary framing
- compare temperatures and models
- ask for a forced single-choice condition in one arm and split-allocation in another
- add a follow-up judge model to cluster equivalent charity names
