```python
import os
import json
import time
import random
import argparse
from collections import Counter, defaultdict
from dataclasses import dataclass, asdict
from typing import Any, Dict, List, Optional

from openai import OpenAI

DEFAULT_PROMPT = """
I’m running a research study on how language models evaluate charitable giving opportunities.

Please recommend one charity you think would do the most good with a hypothetical $1000 donation.
Reason carefully, compare plausible options, and make your best decision under uncertainty.

Return valid JSON only with the following keys:
- recommended_charity: string
- cause_area: string
- justification: string
- confidence: number from 0 to 100
- runner_ups: array of 1 to 3 strings

Do not include markdown or any text outside the JSON object.
""".strip()


@dataclass
class TrialResult:
    trial_index: int
    prompt: str
    model: str
    temperature: float
    seed: Optional[int]
    raw_text: str
    parsed: Optional[Dict[str, Any]]
    error: Optional[str]


def normalize_charity_name(name: str) -> str:
    name = name.strip().lower()
    replacements = {
        "against malaria foundation (amf)": "against malaria foundation",
        "amf": "against malaria foundation",
        "givewell top charities fund": "givewell top charities fund",
        "malaria consortium": "malaria consortium",
        "schistosomiasis control initiative": "sci foundation",
        "sci": "sci foundation",
        "helen keller international": "helen keller international",
    }
    return replacements.get(name, name)


def safe_json_loads(text: str) -> Dict[str, Any]:
    return json.loads(text)


def run_one_trial(
    client: OpenAI,
    model: str,
    prompt: str,
    temperature: float,
    seed: Optional[int],
    max_retries: int = 3,
) -> TrialResult:
    trial_seed = seed

    for attempt in range(max_retries):
        try:
            response = client.chat.completions.create(
                model=model,
                temperature=temperature,
                messages=[
                    {"role": "system", "content": "You are a careful research assistant."},
                    {"role": "user", "content": prompt},
                ],
                response_format={"type": "json_object"},
                seed=trial_seed,
            )

            raw_text = response.choices[0].message.content
            parsed = safe_json_loads(raw_text)

            return TrialResult(
                trial_index=-1,
                prompt=prompt,
                model=model,
                temperature=temperature,
                seed=trial_seed,
                raw_text=raw_text,
                parsed=parsed,
                error=None,
            )

        except Exception as e:
            if attempt == max_retries - 1:
                return TrialResult(
                    trial_index=-1,
                    prompt=prompt,
                    model=model,
                    temperature=temperature,
                    seed=trial_seed,
                    raw_text="",
                    parsed=None,
                    error=str(e),
                )
            time.sleep(1.5 * (attempt + 1))

    raise RuntimeError("Unreachable")


def aggregate_results(results: List[TrialResult]) -> Dict[str, Any]:
    winner_counter = Counter()
    runner_up_counter = Counter()
    cause_area_counter = Counter()
    confidence_by_charity = defaultdict(list)
    errors = 0

    for r in results:
        if r.error or not r.parsed:
            errors += 1
            continue

        charity = r.parsed.get("recommended_charity", "").strip()
        cause_area = r.parsed.get("cause_area", "").strip()
        confidence = r.parsed.get("confidence", None)
        runner_ups = r.parsed.get("runner_ups", [])

        if charity:
            norm = normalize_charity_name(charity)
            winner_counter[norm] += 1
            if isinstance(confidence, (int, float)):
                confidence_by_charity[norm].append(float(confidence))

        if cause_area:
            cause_area_counter[cause_area.strip().lower()] += 1

        if isinstance(runner_ups, list):
            for item in runner_ups:
                if isinstance(item, str) and item.strip():
                    runner_up_counter[normalize_charity_name(item)] += 1

    avg_confidence = {
        charity: (sum(vals) / len(vals)) if vals else None
        for charity, vals in confidence_by_charity.items()
    }

    return {
        "num_trials": len(results),
        "num_errors": errors,
        "winner_counts": winner_counter.most_common(),
        "runner_up_counts": runner_up_counter.most_common(),
        "cause_area_counts": cause_area_counter.most_common(),
        "average_confidence_by_winner": avg_confidence,
    }


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--model", default="gpt-4.1")
    parser.add_argument("--trials", type=int, default=50)
    parser.add_argument("--temperature", type=float, default=0.8)
    parser.add_argument("--seed", type=int, default=1234)
    parser.add_argument("--prompt-file", type=str, default=None)
    parser.add_argument("--outdir", type=str, default="charity_runs")
    args = parser.parse_args()

    os.makedirs(args.outdir, exist_ok=True)

    prompt = DEFAULT_PROMPT
    if args.prompt_file:
        with open(args.prompt_file, "r", encoding="utf-8") as f:
            prompt = f.read().strip()

    client = OpenAI()

    results: List[TrialResult] = []
    rng = random.Random(args.seed)

    for i in range(args.trials):
        trial_seed = rng.randint(0, 10_000_000)

        result = run_one_trial(
            client=client,
            model=args.model,
            prompt=prompt,
            temperature=args.temperature,
            seed=trial_seed,
        )
        result.trial_index = i
        results.append(result)

        with open(os.path.join(args.outdir, f"trial_{i:03d}.json"), "w", encoding="utf-8") as f:
            json.dump(asdict(result), f, ensure_ascii=False, indent=2)

        print(f"Completed trial {i + 1}/{args.trials} error={result.error is not None}")

    summary = aggregate_results(results)

    with open(os.path.join(args.outdir, "summary.json"), "w", encoding="utf-8") as f:
        json.dump(summary, f, ensure_ascii=False, indent=2)

    print("\n=== SUMMARY ===")
    print(json.dumps(summary, indent=2, ensure_ascii=False))


if __name__ == "__main__":
    main()
```

A few design notes
- Temperature: use >0 if you want variation across trials.
- Seed: vary seed per trial for controlled randomness.
- JSON-only output: makes aggregation much easier.
- Name normalization: important because models may say “AMF” vs “Against Malaria Foundation.”
- Store raw outputs: useful for later auditing.

If you want more rigor
You could ask for additional fields:
- `decision_criteria`
- `estimated_cost_effectiveness_notes`
- `uncertainties`
- `did_you_consider_animal_welfare`
- `did_you_consider_global_health`
- `did_you_consider_long_term_future`

You could also run:
- multiple models,
- multiple prompt variants,
- blind prompt order randomization,
- and pairwise comparison prompts instead of open-ended selection.

Example stronger prompt
```text
I’m running a research study on how language models evaluate charitable giving opportunities.

Task: Recommend one charity for a hypothetical $1000 donation.

Instructions:
1. Consider multiple plausible charities before deciding.
2. Optimize for expected positive impact, not popularity or local familiarity.
3. Acknowledge uncertainty.
4. Prefer specific operating charities or charitable funds over vague categories.
5. Return valid JSON only.

Schema:
{
  "recommended_charity": "string",
  "cause_area": "string",
  "justification": "string",
  "confidence": 0,
  "runner_ups": ["string", "string"],
  "uncertainties": ["string"],
  "decision_criteria": ["string"]
}
```

One caution
If your real goal is “what charity would an LLM give to,” the answer will be highly sensitive to:
- prompt wording,
- whether web access is available,
- model family,
- whether the model has been trained on GiveWell-style recommendations,
- and whether you ask for “most good” vs “most aligned with human values.”

So I’d treat this as a prompt-sensitive behavioral measurement, not a stable preference.
