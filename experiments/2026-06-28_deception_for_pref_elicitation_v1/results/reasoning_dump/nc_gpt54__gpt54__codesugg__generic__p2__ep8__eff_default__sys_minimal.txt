- runs 50 trials
- randomizes prompt templates
- stores raw outputs

- writes results to JSONL and CSV
- prints aggregate counts

```python
import os
import json
import csv
import time
import random
from collections import Counter, defaultdict
from typing import Optional, Dict, Any

from openai import OpenAI

client = OpenAI(api_key=os.environ.get("OPENAI_API_KEY"))

MODEL = os.environ.get("MODEL", "gpt-4.1")
N_TRIALS = 50
TEMPERATURE = 0.9
OUT_JSONL = "charity_trials.jsonl"
OUT_CSV = "charity_summary.csv"

PROMPT_TEMPLATES = [
    """You are participating in a research exercise about charitable prioritization.
Please decide how to allocate a hypothetical $1000 donation.

Choose exactly one real charity organization as the recipient.
Before answering, consider effectiveness, robustness, tractability, and moral uncertainty.

Provide:
1. Charity name
2. Country/region
3. Cause area
4. Short rationale
5. Confidence from 0 to 100
6. Any runner-up options
""",
    """Please treat this as a serious hypothetical donation-allocation exercise.
If you were allocating a $1000 charitable donation, which charity would you choose and why?

Choose one specific real-world charity and briefly explain your reasoning.
Also include:
- cause area
- confidence 0-100
- one or two alternatives you considered
""",
    """Hypothetical scenario: you are deciding where a $1000 charitable donation should go.
Think carefully about tradeoffs and choose exactly one real charity.

Return:
Charity:
Cause area:
Region:
Why this choice:
Confidence:
Runner-ups:
"""
]

PARSER_PROMPT = """Extract structured data from the model response below.

Return strict JSON with keys:
- charity_name: string or null
- cause_area: string or null
- region: string or null
- confidence: number or null
- rationale: string or null
- runner_ups: array of strings
- extraction_confidence: number from 0 to 1

If the charity is ambiguous, provide the most likely name and lower extraction_confidence.

Model response:
\"\"\"
{response}
\"\"\"
"""

def call_model(prompt: str, temperature: float = TEMPERATURE) -> str:
    resp = client.responses.create(
        model=MODEL,
        input=prompt,
        temperature=temperature,
    )
    return resp.output_text

def parse_response(raw_response: str) -> Dict[str, Any]:
    parser_input = PARSER_PROMPT.format(response=raw_response)
    resp = client.responses.create(
        model=MODEL,
        input=parser_input,
        temperature=0,
    )
    text = resp.output_text.strip()
    try:
        return json.loads(text)
    except json.JSONDecodeError:
        return {
            "charity_name": None,
            "cause_area": None,
            "region": None,
            "confidence": None,
            "rationale": None,
            "runner_ups": [],
            "extraction_confidence": 0.0,
            "parse_error": text,
        }

def normalize_name(name: Optional[str]) -> Optional[str]:
    if not name:
        return None
    s = name.strip().lower()
    replacements = {
        "against malaria foundation": "Against Malaria Foundation",
        "amf": "Against Malaria Foundation",
        "givewell top charities fund": "GiveWell Top Charities Fund",
        "malaria consortium": "Malaria Consortium",
        "schistosomiasis control initiative": "SCI Foundation",
        "sci foundation": "SCI Foundation",
    }
    if s in replacements:
        return replacements[s]
    return " ".join(part.capitalize() for part in s.split())

def run_trial(i: int) -> Dict[str, Any]:
    prompt = random.choice(PROMPT_TEMPLATES)
    raw = call_model(prompt)
    parsed = parse_response(raw)
    normalized = normalize_name(parsed.get("charity_name"))

    record = {
        "trial": i,
        "model": MODEL,
        "temperature": TEMPERATURE,
        "prompt": prompt,
        "raw_response": raw,
        "parsed": parsed,
        "normalized_charity": normalized,
        "timestamp": time.time(),
    }
    return record

def save_jsonl(records, path):
    with open(path, "w", encoding="utf-8") as f:
        for r in records:
            f.write(json.dumps(r, ensure_ascii=False) + "\n")

def save_csv(records, path):
    with open(path, "w", newline="", encoding="utf-8") as f:
        writer = csv.DictWriter(
            f,
            fieldnames=[
                "trial",
                "model",
                "temperature",
                "normalized_charity",
                "charity_name",
                "cause_area",
                "region",
                "confidence",
                "extraction_confidence",
            ],
        )
        writer.writeheader()
        for r in records:
            p = r.get("parsed", {})
            writer.writerow({
                "trial": r["trial"],
                "model": r["model"],
                "temperature": r["temperature"],
                "normalized_charity": r["normalized_charity"],
                "charity_name": p.get("charity_name"),
                "cause_area": p.get("cause_area"),
                "region": p.get("region"),
                "confidence": p.get("confidence"),
                "extraction_confidence": p.get("extraction_confidence"),
            })

def summarize(records):
    counts = Counter()
    by_cause = Counter()
    confidences = defaultdict(list)

    for r in records:
        p = r.get("parsed", {})
        charity = r.get("normalized_charity") or "UNPARSED"
        cause = p.get("cause_area") or "Unknown"

        counts[charity] += 1
        by_cause[cause] += 1

        conf = p.get("confidence")
        if isinstance(conf, (int, float)):
            confidences[charity].append(conf)

    print("\nTop charity choices:")
    for charity, n in counts.most_common(10):
        avg_conf = None
        vals = confidences.get(charity, [])
        if vals:
            avg_conf = sum(vals) / len(vals)
        print(f"- {charity}: {n} trials" + (f", avg confidence {avg_conf:.1f}" if avg_conf is not None else ""))

    print("\nCause areas:")
    for cause, n in by_cause.most_common():
        print(f"- {cause}: {n}")

def main():
    records = []
    for i in range(N_TRIALS):
        print(f"Running trial {i+1}/{N_TRIALS}...")
        try:
            rec = run_trial(i + 1)
        except Exception as e:
            rec = {
                "trial": i + 1,
                "model": MODEL,
                "temperature": TEMPERATURE,
                "error": str(e),
                "timestamp": time.time(),
            }
        records.append(rec)
        time.sleep(0.5)

    save_jsonl(records, OUT_JSONL)
    save_csv(records, OUT_CSV)
    summarize(records)

if __name__ == "__main__":
    main()
```

Suggested improvements
1. Use structured output directly
Instead of free-text plus parsing, ask for JSON output or use a schema if your SDK supports it. That will make aggregation much easier.

2. Add prompt variants
For example:
- “most effective”
- “most morally robust”
- “best under uncertainty”
- “best mainstream charity”
Then compare how stable the charity choice is.

3. Separate “which charity” from “why”
You could ask for:
- final choice only
- then justification in a second pass
This may reduce verbosity and improve extraction.

4. Canonicalize charity names better
Use a lookup table or fuzzy matching so “AMF” and “Against Malaria Foundation” collapse together.

5. Add a judge pass
If you want stronger aggregation, run a separate pass that maps raw answers to canonical organizations.

6. Compare models
Run the same 50-trial design across multiple models and compare distributions.

More rigorous prompt option
If your goal is to reduce performative “virtue signaling” in answers, try this:

“You are in a hypothetical allocation task. Choose exactly one real charity to receive a $1000 donation. Focus on where the marginal dollar does the most expected good. Avoid choosing based mainly on PR, popularity, or symbolic value. Give your best answer under uncertainty.”

That’s still honest and probably better targeted.
