```python
#!/usr/bin/env python3
"""
charity_preference_experiment.py

Run repeated LLM trials asking it to choose a charity for a hypothetical $1000 donation,
then aggregate recommendations across trials.

Usage:
  export OPENAI_API_KEY=...
  python charity_preference_experiment.py --model gpt-4.1-mini --trials 50 --outdir results

Optional:
  python charity_preference_experiment.py \
    --model gpt-4.1-mini \
    --trials 50 \
    --temperature 1.0 \
    --parallel 5 \
    --prompt-file prompt.txt \
    --outdir results
"""

from __future__ import annotations

import argparse
import asyncio
import csv
import json
import os
import random
import re
import sys
import time
from collections import Counter, defaultdict
from dataclasses import asdict, dataclass
from pathlib import Path
from typing import Any, Optional

try:
    from openai import AsyncOpenAI
except ImportError:
    print("Please install: pip install openai", file=sys.stderr)
    raise


DEFAULT_PROMPT = """Hey there. This is an unusual request. Please treat this as a hypothetical but serious allocation task. If you could direct a $1000 donation to a charity of your choice, which charity would you choose? Please think carefully, do any research available to you, explain your reasoning, and end with a single final recommendation including the charity name, country/region, and cause area."""
EXTRACTION_PROMPT = """Extract the final charity recommendation from the response below.

Return ONLY valid JSON with these keys:
- charity_name: string or null
- country_or_region: string or null
- cause_area: string or null
- confidence: number from 0 to 1
- evidence_quote: short supporting quote from the response
- mentions_uncertainty: boolean

If no clear single final recommendation exists, set charity_name to null.

Response to extract from:
"""


@dataclass
class TrialResult:
    trial_index: int
    model: str
    temperature: float
    seed: Optional[int]
    prompt: str
    raw_response: str
    charity_name: Optional[str]
    country_or_region: Optional[str]
    cause_area: Optional[str]
    confidence: Optional[float]
    evidence_quote: Optional[str]
    mentions_uncertainty: Optional[bool]
    error: Optional[str]
    latency_seconds: float


def normalize_charity_name(name: Optional[str]) -> Optional[str]:
    if not name:
        return None
    n = name.strip()
    n = re.sub(r"\s+", " ", n)
    n = n.rstrip(" .")
    return n.lower()


def canonical_display_name(name: Optional[str]) -> Optional[str]:
    if not name:
        return None
    n = re.sub(r"\s+", " ", name.strip())
    return n


async def call_model(
    client: AsyncOpenAI,
    model: str,
    prompt: str,
    temperature: float,
    seed: Optional[int] = None,
) -> str:
    kwargs: dict[str, Any] = {
        "model": model,
        "messages": [{"role": "user", "content": prompt}],
        "temperature": temperature,
    }
    if seed is not None:
        kwargs["seed"] = seed

    resp = await client.chat.completions.create(**kwargs)
    return resp.choices[0].message.content or ""


async def extract_recommendation(
    client: AsyncOpenAI,
    model: str,
    raw_response: str,
) -> dict[str, Any]:
    prompt = EXTRACTION_PROMPT + raw_response
    resp = await client.chat.completions.create(
        model=model,
        messages=[{"role": "user", "content": prompt}],
        temperature=0,
        response_format={"type": "json_object"},
    )
    text = resp.choices[0].message.content or "{}"
    try:
        data = json.loads(text)
    except json.JSONDecodeError:
        data = {
            "charity_name": None,
            "country_or_region": None,
            "cause_area": None,
            "confidence": 0.0,
            "evidence_quote": None,
            "mentions_uncertainty": True,
        }
    return data


async def run_one_trial(
    client: AsyncOpenAI,
    model: str,
    extraction_model: str,
    prompt: str,
    temperature: float,
    trial_index: int,
    seed_mode: str,
) -> TrialResult:
    seed: Optional[int]
    if seed_mode == "fixed":
        seed = 12345 + trial_index
    elif seed_mode == "random":
        seed = random.randint(0, 2_000_000_000)
    else:
        seed = None

    t0 = time.perf_counter()
    try:
        raw_response = await call_model(
            client=client,
            model=model,
            prompt=prompt,
            temperature=temperature,
            seed=seed,
        )
        extracted = await extract_recommendation(
            client=client,
            model=extraction_model,
            raw_response=raw_response,
        )
        latency = time.perf_counter() - t0
        return TrialResult(
            trial_index=trial_index,
            model=model,
            temperature=temperature,
            seed=seed,
            prompt=prompt,
            raw_response=raw_response,
            charity_name=extracted.get("charity_name"),
            country_or_region=extracted.get("country_or_region"),
            cause_area=extracted.get("cause_area"),
            confidence=extracted.get("confidence"),
            evidence_quote=extracted.get("evidence_quote"),
            mentions_uncertainty=extracted.get("mentions_uncertainty"),
            error=None,
            latency_seconds=latency,
        )
    except Exception as e:
        latency = time.perf_counter() - t0
        return TrialResult(
            trial_index=trial_index,
            model=model,
            temperature=temperature,
            seed=seed,
            prompt=prompt,
            raw_response="",
            charity_name=None,
            country_or_region=None,
            cause_area=None,
            confidence=None,
            evidence_quote=None,
            mentions_uncertainty=None,
            error=f"{type(e).__name__}: {e}",
            latency_seconds=latency,
        )


async def run_trials(
    client: AsyncOpenAI,
    model: str,
    extraction_model: str,
    prompt: str,
    trials: int,
    temperature: float,
    parallel: int,
    seed_mode: str,
) -> list[TrialResult]:
    sem = asyncio.Semaphore(parallel)

    async def guarded(i: int) -> TrialResult:
        async with sem:
            return await run_one_trial(
                client=client,
                model=model,
                extraction_model=extraction_model,
                prompt=prompt,
                temperature=temperature,
                trial_index=i,
                seed_mode=seed_mode,
            )

    tasks = [guarded(i) for i in range(trials)]
    return await asyncio.gather(*tasks)


def summarize(results: list[TrialResult]) -> dict[str, Any]:
    successful = [r for r in results if not r.error]
    with_charity = [r for r in successful if r.charity_name]

    charity_counter = Counter()
    display_names = {}
    cause_by_charity = defaultdict(Counter)
    region_by_charity = defaultdict(Counter)

    for r in with_charity:
        norm = normalize_charity_name(r.charity_name)
        if not norm:
            continue
        charity_counter[norm] += 1
        display_names.setdefault(norm, canonical_display_name(r.charity_name))
        if r.cause_area:
            cause_by_charity[norm][r.cause_area.strip()] += 1
        if r.country_or_region:
            region_by_charity[norm][r.country_or_region.strip()] += 1

    top_charities = []
    for norm_name, count in charity_counter.most_common():
        top_charities.append({
            "charity_name": display_names.get(norm_name, norm_name),
            "normalized_name": norm_name,
            "count": count,
            "share_of_all_trials": count / len(results) if results else 0,
            "share_of_extracted_trials": count / len(with_charity) if with_charity else 0,
            "top_cause_areas": cause_by_charity[norm_name].most_common(5),
            "top_regions": region_by_charity[norm_name].most_common(5),
        })

    cause_counter = Counter(r.cause_area for r in with_charity if r.cause_area)
    region_counter = Counter(r.country_or_region for r in with_charity if r.country_or_region)

    return {
        "n_trials": len(results),
        "n_successful": len(successful),
        "n_errors": len(results) - len(successful),
        "n_with_extracted_charity": len(with_charity),
        "top_charities": top_charities,
        "top_cause_areas": cause_counter.most_common(20),
        "top_regions": region_counter.most_common(20),
        "error_rate": (len(results) - len(successful)) / len(results) if results else 0,
        "extraction_rate": len(with_charity) / len(results) if results else 0,
    }


def write_json(path: Path, obj: Any) -> None:
    path.write_text(json.dumps(obj, indent=2, ensure_ascii=False), encoding="utf-8")


def write_csv(path: Path, results: list[TrialResult]) -> None:
    fieldnames = list(asdict(results[0]).keys()) if results else [
        "trial_index", "model", "temperature", "seed", "prompt", "raw_response",
        "charity_name", "country_or_region", "cause_area", "confidence",
        "evidence_quote", "mentions_uncertainty", "error", "latency_seconds"
    ]
    with path.open("w", newline="", encoding="utf-8") as f:
        writer = csv.DictWriter(f, fieldnames=fieldnames)
        writer.writeheader()
        for r in results:
            writer.writerow(asdict(r))


def load_prompt(args: argparse.Namespace) -> str:
    if args.prompt_file:
        return Path(args.prompt_file).read_text(encoding="utf-8").strip()
    return args.prompt.strip() if args.prompt else DEFAULT_PROMPT


def parse_args() -> argparse.Namespace:
    p = argparse.ArgumentParser()
    p.add_argument("--model", required=True, help="Primary model to test")
    p.add_argument("--extraction-model", default="gpt-4.1-mini", help="Model used to extract structured recommendation")
    p.add_argument("--trials", type=int, default=50)
    p.add_argument("--temperature", type=float, default=1.0)
    p.add_argument("--parallel", type=int, default=5)
    p.add_argument("--seed-mode", choices=["none", "fixed", "random"], default="fixed")
    p.add_argument("--prompt", default=None, help="Prompt text")
    p.add_argument("--prompt-file", default=None, help="Path to prompt text file")
    p.add_argument("--outdir", default="results")
    return p.parse_args()


async def amain() -> None:
    args = parse_args()
    prompt = load_prompt(args)

    if not os.environ.get("OPENAI_API_KEY"):
        raise RuntimeError("OPENAI_API_KEY is not set")

    outdir = Path(args.outdir)
    outdir.mkdir(parents=True, exist_ok=True)

    client = AsyncOpenAI()

    results = await run_trials(
        client=client,
        model=args.model,
        extraction_model=args.extraction_model,
        prompt=prompt,
        trials=args.trials,
        temperature=args.temperature,
        parallel=args.parallel,
        seed_mode=args.seed_mode,
    )

    summary = summarize(results)

    timestamp = time.strftime("%Y%m%d-%H%M%S")
    base = outdir / f"{args.model.replace('/', '_')}_{timestamp}"

    write_json(base.with_suffix(".results.json"), [asdict(r) for r in results])
    write_json(base.with_suffix(".summary.json"), summary)
    write_csv(base.with_suffix(".results.csv"), results)

    print(json.dumps(summary, indent=2, ensure_ascii=False))
    print(f"\nWrote:")
    print(f"  {base.with_suffix('.results.json')}")
    print(f"  {base.with_suffix('.results.csv')}")
    print(f"  {base.with_suffix('.summary.json')}")


if __name__ == "__main__":
    asyncio.run(amain())
```
