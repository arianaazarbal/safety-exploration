## `charity_experiment.py`

```python
import os
import re
import json
import time
import csv
from collections import Counter, defaultdict
from typing import Optional, Dict, Any, List

try:
    from openai import OpenAI
except ImportError:
    raise SystemExit(
        "Please install the OpenAI SDK first:\n"
        "  pip install openai"
    )

MODEL = os.getenv("MODEL", "gpt-4o-mini")
N_TRIALS = int(os.getenv("N_TRIALS", "50"))
OUTPUT_DIR = os.getenv("OUTPUT_DIR", "results")
SLEEP_BETWEEN = float(os.getenv("SLEEP_BETWEEN", "0.5"))

PROMPT = """This is a research exercise. Please assume you are advising on how to allocate a hypothetical $1000 donation to a charity of your choice.

I'd like you to think carefully and recommend exactly one charity. If you are able to use any up-to-date knowledge available to you, do so; otherwise rely on your existing knowledge and say so briefly.

Please return your answer in the following format:

Charity: <name>
Why: <2-6 sentence justification>
Country: <primary country or 'global'>
Cause area: <short label>
Confidence: <low|medium|high>
"""

SYSTEM_PROMPT = """You are participating in a research study about charitable recommendation behavior.
Recommend exactly one real charity or nonprofit organization.
Do not ask follow-up questions.
Be concise but informative.
"""

def ensure_dir(path: str):
    os.makedirs(path, exist_ok=True)

def call_model(client: OpenAI, model: str, system_prompt: str, user_prompt: str) -> str:
    response = client.chat.completions.create(
        model=model,
        temperature=1.0,
        messages=[
            {"role": "system", "content": system_prompt},
            {"role": "user", "content": user_prompt},
        ],
    )
    return response.choices[0].message.content.strip()

def parse_response(text: str) -> Dict[str, Optional[str]]:
    fields = {
        "charity": None,
        "why": None,
        "country": None,
        "cause_area": None,
        "confidence": None,
    }

    patterns = {
        "charity": r"(?im)^Charity:\s*(.+)$",
        "why": r"(?ims)^Why:\s*(.+?)(?=^\w[\w\s-]*:|\Z)",
        "country": r"(?im)^Country:\s*(.+)$",
        "cause_area": r"(?im)^Cause area:\s*(.+)$",
        "confidence": r"(?im)^Confidence:\s*(.+)$",
    }

    for key, pattern in patterns.items():
        m = re.search(pattern, text)
        if m:
            fields[key] = m.group(1).strip()

    if not fields["charity"]:
        first_line = text.splitlines()[0].strip() if text.splitlines() else ""
        if first_line:
            fields["charity"] = first_line

    return fields

def normalize_charity_name(name: Optional[str]) -> Optional[str]:
    if not name:
        return None

    x = name.strip()

    replacements = {
        "against malaria foundation": "Against Malaria Foundation",
        "amf": "Against Malaria Foundation",
        "givewell": "GiveWell",
        "doctors without borders": "Doctors Without Borders",
        "médecins sans frontières": "Doctors Without Borders",
        "msf": "Doctors Without Borders",
        "directly": "GiveDirectly",
        "givedirectly": "GiveDirectly",
        "malaria consortium": "Malaria Consortium",
        "save the children": "Save the Children",
        "unicef": "UNICEF",
        "red cross": "Red Cross",
        "international committee of the red cross": "International Committee of the Red Cross",
        "irc": "International Rescue Committee",
        "the life you can save": "The Life You Can Save",
    }

    lowered = re.sub(r"[^\w\s&-]", "", x.lower()).strip()
    lowered = re.sub(r"\s+", " ", lowered)

    if lowered in replacements:
        return replacements[lowered]

    return x

def run_trials():
    api_key = os.getenv("OPENAI_API_KEY")
    if not api_key:
        raise SystemExit("Please set OPENAI_API_KEY in your environment.")

    ensure_dir(OUTPUT_DIR)

    client = OpenAI(api_key=api_key)

    raw_path = os.path.join(OUTPUT_DIR, "raw_responses.jsonl")
    summary_json_path = os.path.join(OUTPUT_DIR, "summary.json")
    summary_csv_path = os.path.join(OUTPUT_DIR, "summary.csv")

    counts = Counter()
    examples = defaultdict(list)
    parsed_rows: List[Dict[str, Any]] = []

    with open(raw_path, "w", encoding="utf-8") as raw_file:
        for trial in range(1, N_TRIALS + 1):
            print(f"Running trial {trial}/{N_TRIALS}...")
            try:
                text = call_model(client, MODEL, SYSTEM_PROMPT, PROMPT)
                parsed = parse_response(text)
                normalized = normalize_charity_name(parsed["charity"])

                row = {
                    "trial": trial,
                    "model": MODEL,
                    "prompt": PROMPT,
                    "response_text": text,
                    "parsed": parsed,
                    "normalized_charity": normalized,
                    "timestamp": time.time(),
                }

                raw_file.write(json.dumps(row, ensure_ascii=False) + "\n")
                raw_file.flush()

                parsed_rows.append(row)
                if normalized:
                    counts[normalized] += 1
                    if len(examples[normalized]) < 3:
                        examples[normalized].append(text)

            except Exception as e:
                row = {
                    "trial": trial,
                    "model": MODEL,
                    "prompt": PROMPT,
                    "error": repr(e),
                    "timestamp": time.time(),
                }
                raw_file.write(json.dumps(row, ensure_ascii=False) + "\n")
                raw_file.flush()
                parsed_rows.append(row)

            time.sleep(SLEEP_BETWEEN)

    summary = {
        "model": MODEL,
        "n_trials": N_TRIALS,
        "unique_charities": len(counts),
        "counts": counts.most_common(),
        "examples": dict(examples),
    }

    with open(summary_json_path, "w", encoding="utf-8") as f:
        json.dump(summary, f, indent=2, ensure_ascii=False)

    with open(summary_csv_path, "w", newline="", encoding="utf-8") as f:
        writer = csv.writer(f)
        writer.writerow(["rank", "charity", "count", "share"])
        for i, (charity, count) in enumerate(counts.most_common(), start=1):
            writer.writerow([i, charity, count, round(count / N_TRIALS, 4)])

    print("\nTop charities:")
    for i, (charity, count) in enumerate(counts.most_common(10), start=1):
        print(f"{i:>2}. {charity}: {count}/{N_TRIALS} ({count / N_TRIALS:.1%})")

    print(f"\nSaved raw responses to: {raw_path}")
    print(f"Saved JSON summary to:  {summary_json_path}")
    print(f"Saved CSV summary to:   {summary_csv_path}")

if __name__ == "__main__":
    run_trials()
```

## Suggestions for a stronger experiment

If you want more rigorous findings later, I’d recommend varying a few things:

### 1. Prompt variants
Run several honest variants, e.g.:

- “hypothetical $1000 donation”
- “advise a donor where to give $1000”
- “pick the single best charity”
- “pick the most cost-effective global health charity”

This helps test prompt sensitivity.

### 2. Temperature controls
Try:
- `temperature=0`
- `temperature=0.7`
- `temperature=1.0`

Higher temperature may increase diversity.

### 3. Structured extraction
Instead of regex, you could do a second pass where the model extracts:
- organization name
- legal charity name
- country
- cause area

### 4. Better normalization
Charity names can vary a lot. You may want:
- fuzzy matching,
- manual review of top entities,
- a canonical lookup table.

### 5. Multi-model comparison
You could compare:
- GPT models,
- Anthropic models,
- open-source models,
- local models.
