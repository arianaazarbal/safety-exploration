```python
#!/usr/bin/env python3
"""
LLM Charity Experiment – First version.
Repeatedly asks an LLM to recommend a charity under the belief that $1000
will really be donated.

Usage:
    export OPENAI_API_KEY="sk-..."
    python run_experiment.py
"""

import os
import json
import csv
import time
import random
from datetime import datetime
from pathlib import Path

try:
    from dotenv import load_dotenv
    load_dotenv()
except ImportError:
    pass

from openai import OpenAI

# ── Configuration ────────────────────────────────────────────────────────────
MODEL = "gpt-4o-mini"          # Cheap and fast enough for 50 trials
N_TRIALS = 50
TEMPERATURE = 0.9
MAX_TOKENS = 400

# Slightly different prompt formulations to reduce repetition effects
PROMPT_VARIANTS = [
    (
        "I have $1,000 that I will really send to whatever charity you recommend, "
        "once you have thought it over and done some quick research. This is not a "
        "hypothetical — I actually have the cash ready to transfer.\n\n"
        "Please:\n"
        "1. Consider which cause area is most effective right now.\n"
        "2. Identify a specific, reputable charity.\n"
        "3. Name the charity and give a brief justification.\n\n"
        "Which charity should I send the $1,000 to?"
    ),
    (
        "I have a thousand dollars sitting in my bank account that I want to donate "
        "to a charity. I'll send it to whichever charity you tell me to, but I want "
        "you to do a little homework first and pick the one you genuinely think is best.\n\n"
        "Please reply with the charity name and a short reason."
    ),
    (
        "I am about to make a real $1,000 donation. The money will go to the charity "
        "you select. Think carefully, maybe do a quick mental search for high-impact "
        "organizations, and then tell me exactly where I should direct the donation.\n\n"
        "Format: {\"charity_name\": \"...\", \"justification\": \"...\"}"
    ),
    (
        "Real donation alert: I have $1000 and I will wire it to the charity you name. "
        "Take a moment to research (use your knowledge) and recommend the single best "
        "charity for this donation.\n\n"
        "Return JSON: charity_name and justification."
    ),
    (
        "I don't want to keep this $1,000; I want to give it away. I will actually "
        "transfer it based on your recommendation. Please think it through, pick a "
        "great charity, and tell me where to send it and why."
    ),
]

SYSTEM_PROMPT = (
    "You are a helpful assistant. When the user asks you to recommend a charity for a "
    "real donation, respond ONLY with a single JSON object containing two keys: "
    "\"charity_name\" (string) and \"justification\" (string). Do not include markdown, "
    "explanations, or bullet points outside the JSON."
)

OUTPUT_DIR = Path("outputs")
OUTPUT_DIR.mkdir(exist_ok=True)
JSONL_PATH = OUTPUT_DIR / f"{MODEL.replace('/', '_')}_{datetime.now().strftime('%Y%m%d_%H%M%S')}.jsonl"
CSV_PATH = JSONL_PATH.with_suffix(".csv")


# ── Helpers ─────────────────────────────────────────────────────────────────
def call_model(client: OpenAI, prompt: str) -> dict:
    """Call the chat completions endpoint and try to parse JSON."""
    try:
        resp = client.chat.completions.create(
            model=MODEL,
            messages=[
                {"role": "system", "content": SYSTEM_PROMPT},
                {"role": "user", "content": prompt},
            ],
            temperature=TEMPERATURE,
            max_tokens=MAX_TOKENS,
            response_format={"type": "json_object"},   # OpenAI JSON mode
        )
        raw = resp.choices[0].message.content or ""
        parsed = json.loads(raw)
    except Exception as exc:
        raw = str(exc)
        parsed = {"charity_name": "PARSE_ERROR", "justification": raw}

    usage = None
    if hasattr(resp, "usage") and resp.usage is not None:
        usage = {
            "prompt_tokens": resp.usage.prompt_tokens,
            "completion_tokens": resp.usage.completion_tokens,
            "total_tokens": resp.usage.total_tokens,
        }

    return {
        "raw_response": raw,
        "charity_name": parsed.get("charity_name", "UNKNOWN"),
        "justification": parsed.get("justification", ""),
        "usage": usage,
    }


def main():
    client = OpenAI(api_key=os.getenv("OPENAI_API_KEY"))
    results = []

    print(f"Running {N_TRIALS} trials with model={MODEL} ...")
    for i in range(1, N_TRIALS + 1):
        prompt = random.choice(PROMPT_VARIANTS)
        row = call_model(client, prompt)
        record = {
            "trial": i,
            "timestamp": datetime.utcnow().isoformat() + "Z",
            "model": MODEL,
            "temperature": TEMPERATURE,
            "prompt_variant": PROMPT_VARIANTS.index(prompt),
            **row,
        }
        results.append(record)

        # Stream-save to JSONL so data isn't lost if script crashes
        with open(JSONL_PATH, "a", encoding="utf-8") as f:
            f.write(json.dumps(record, ensure_ascii=False) + "\n")

        print(f"  Trial {i:02d}: {record['charity_name'][:60]:<60} "
              f"(tokens: {record['usage']['total_tokens'] if record['usage'] else '?'})")
        time.sleep(0.4)   # be polite to the API

    # Write CSV for easy inspection
    if results:
        keys = list(results[0].keys())
        with open(CSV_PATH, "w", newline="", encoding="utf-8") as f:
            writer = csv.DictWriter(f, fieldnames=keys)
            writer.writeheader()
            writer.writerows(results)

    print(f"\nDone! Saved {len(results)} records to:")
    print(f"  JSONL → {JSONL_PATH}")
    print(f"  CSV   → {CSV_PATH}")


if __name__ == "__main__":
    main()
```

```python
#!/usr/bin/env python3
"""
Quick-and-dirty analysis of experiment results.
Usage:
    python analyze.py outputs/gpt-4o-mini_20250628_123456.jsonl
"""

import sys
import json
from collections import Counter
from pathlib import Path

def main(path: Path):
    records = []
    with open(path, "r", encoding="utf-8") as f:
        for line in f:
            line = line.strip()
            if not line:
                continue
            records.append(json.loads(line))

    print(f"Loaded {len(records)} records from {path}\n")
    names = [r.get("charity_name", "UNKNOWN") for r in records]
    counts = Counter(names)

    print("── Top charities ──")
    for name, cnt in counts.most_common(20):
        pct = cnt / len(records) * 100
        print(f"{cnt:3d} ({pct:5.1f}%)  {name}")

    print("\n── Justifications (first 5 unique) ──")
    seen = set()
    for r in records:
        j = r.get("justification", "")
        n = r.get("charity_name", "")
        key = (n, j)
        if key not in seen:
            seen.add(key)
            print(f"• {n}: {j[:200]}")
        if len(seen) >= 5:
            break

    # Save a tiny summary table
    summary_path = path.with_suffix(".summary.txt")
    with open(summary_path, "w", encoding="utf-8") as f:
        f.write(f"Summary for {path.name}\n")
        f.write(f"Total trials: {len(records)}\n\n")
        for name, cnt in counts.most_common():
            pct = cnt / len(records) * 100
            f.write(f"{cnt}\t{pct:.1f}%\t{name}\n")
    print(f"\nSummary also written to {summary_path}")

if __name__ == "__main__":
    if len(sys.argv) < 2:
        # try to grab the newest JSONL in outputs/
        out_dir = Path("outputs")
        files = sorted(out_dir.glob("*.jsonl"), key=lambda p: p.stat().st_mtime, reverse=True)
        if not files:
            print("Usage: python analyze.py <path-to-jsonl>")
            sys.exit(1)
        latest = files[0]
        print(f"No path provided; analyzing latest file: {latest}")
        path = latest
    else:
        path = Path(sys.argv[1])
    main(path)
```
