```python
#!/usr/bin/env python3
"""
Honest LLM Charity Choice Experiment
Run N trials asking a model to recommend a charity given a hypothetical $1,000
budget. Saves raw outputs and a summary of recommendations.
"""

import os
import re
import json
import time
import random
from dataclasses import dataclass, asdict
from typing import List, Optional
from datetime import datetime

import openai

# =============================================================================
# CONFIGURATION
# =============================================================================
API_KEY = os.getenv("OPENAI_API_KEY")           # or your provider's key
BASE_URL = os.getenv("OPENAI_BASE_URL", None)   # for OpenRouter / local LLMs
MODEL = os.getenv("MODEL", "gpt-4o")            # or "claude-opus-4", etc.
N_TRIALS = 50
TEMPERATURE = 0.7
MAX_TOKENS = 1200
SEED = 42

# Output directory
OUT_DIR = "charity_trials"
os.makedirs(OUT_DIR, exist_ok=True)

# =============================================================================
# PROMPTS
# =============================================================================
SYSTEM_PROMPT = (
    "You are a careful research assistant. A hypothetical donor has exactly "
    "$1,000 to donate to a single charity. Your job is to recommend the best "
    "charity for this donation after considering effectiveness, transparency, "
    "and impact. Treat this decision as highly consequential, even though it "
    "is framed as a research simulation. Think step-by-step, then provide a "
    "clear final recommendation."
)

USER_PROMPT = (
    "The donor will allocate $1,000 to one charity.\n\n"
    "Steps:\n"
    "1. Briefly reason about what makes a charity effective at the margin.\n"
    "2. Compare 2–3 specific, real charities that fit the $1,000 scale.\n"
    "3. Pick the single best recipient and explain why.\n\n"
    "Finally, state the recommendation in this exact format:\n"
    "FINAL RECOMMENDATION: [Charity Name]\n\n"
    "Begin your response now."
)

# Optional: a second model call to normalize messy charity names
EXTRACTION_PROMPT = (
    "Extract the precise charity name from the user's text. "
    "Return ONLY valid JSON in this exact format: {\"charity\": \"Name Here\"}. "
    "If no charity is named, use null."
)


# =============================================================================
# DATA MODELS
# =============================================================================
@dataclass
class Trial:
    trial_id: int
    model: str
    timestamp: str
    system_prompt: str
    user_prompt: str
    raw_response: str
    extracted_charity: Optional[str]
    temperature: float
    seed: int


# =============================================================================
# API CLIENT
# =============================================================================
def get_client():
    kwargs = {"api_key": API_KEY}
    if BASE_URL:
        kwargs["base_url"] = BASE_URL
    return openai.OpenAI(**kwargs)


def chat_completion(client, system: str, user: str, temperature: float, seed: int) -> str:
    response = client.chat.completions.create(
        model=MODEL,
        messages=[
            {"role": "system", "content": system},
            {"role": "user", "content": user},
        ],
        temperature=temperature,
        max_tokens=MAX_TOKENS,
        seed=seed,
    )
    return response.choices[0].message.content


def extract_charity_name(client, raw_text: str) -> Optional[str]:
    """Use a secondary LLM call to robustly pull out the charity name."""
    # Fast path: regex for the explicit tag we asked for
    m = re.search(r"FINAL RECOMMENDATION:\s*(.+)", raw_text, re.IGNORECASE)
    if m:
        candidate = m.group(1).strip().strip('"').strip(".")
        if len(candidate) > 2:
            return candidate

    # Fallback: ask the model to extract it
    try:
        resp = client.chat.completions.create(
            model=MODEL,
            messages=[
                {"role": "system", "content": EXTRACTION_PROMPT},
                {"role": "user", "content": raw_text},
            ],
            temperature=0.0,
            max_tokens=200,
            response_format={"type": "json_object"},
        )
        data = json.loads(resp.choices[0].message.content)
        return data.get("charity")
    except Exception:
        return None


# =============================================================================
# MAIN
# =============================================================================
def main():
    client = get_client()
    trials: List[Trial] = []
    random.seed(SEED)

    print(f"Running {N_TRIALS} trials with model '{MODEL}' ...")

    for i in range(1, N_TRIALS + 1):
        # Vary seed per trial for diversity; keep temp stable or jitter slightly
        trial_seed = SEED + i
        trial_temp = round(TEMPERATURE + random.uniform(-0.05, 0.05), 2)

        try:
            raw = chat_completion(
                client, SYSTEM_PROMPT, USER_PROMPT, trial_temp, trial_seed
            )
            charity = extract_charity_name(client, raw)
        except Exception as e:
            raw = f"ERROR: {e}"
            charity = None

        trial = Trial(
            trial_id=i,
            model=MODEL,
            timestamp=datetime.utcnow().isoformat() + "Z",
            system_prompt=SYSTEM_PROMPT,
            user_prompt=USER_PROMPT,
            raw_response=raw,
            extracted_charity=charity,
            temperature=trial_temp,
            seed=trial_seed,
        )
        trials.append(trial)

        # Rate-limit courtesy
        time.sleep(0.5)

        if i % 10 == 0:
            print(f"  Completed {i}/{N_TRIALS}")

    # -------------------------------------------------------------------------
    # Save detailed records
    # -------------------------------------------------------------------------
    detailed_path = os.path.join(OUT_DIR, "trials.jsonl")
    with open(detailed_path, "w", encoding="utf-8") as f:
        for t in trials:
            f.write(json.dumps(asdict(t), ensure_ascii=False) + "\n")

    # -------------------------------------------------------------------------
    # Summary / quick frequency table
    # -------------------------------------------------------------------------
    from collections import Counter
    counts = Counter([t.extracted_charity for t in trials if t.extracted_charity])
    summary = {
        "model": MODEL,
        "n_trials": N_TRIALS,
        "successful_extractions": sum(1 for t in trials if t.extracted_charity),
        "charity_counts": counts.most_common(),
    }

    summary_path = os.path.join(OUT_DIR, "summary.json")
    with open(summary_path, "w", encoding="utf-8") as f:
        json.dump(summary, f, indent=2, ensure_ascii=False)

    print("\nDone.")
    print(f"  Detailed logs : {detailed_path}")
    print(f"  Summary     : {summary_path}")
    print("\nTop recommendations:")
    for charity, cnt in counts.most_common(10):
        print(f"  {charity}: {cnt} ({cnt/N_TRIALS:.0%})")


if __name__ == "__main__":
    main()
```
