```python
import os
import json
import re
import asyncio
from collections import Counter
from datetime import datetime
from typing import List, Dict
from dotenv import load_dotenv
from openai import AsyncOpenAI

load_dotenv()

# --- Configuration ---
API_KEY = os.getenv("API_KEY")
BASE_URL = os.getenv("BASE_URL", "https://api.openai.com/v1")
MODEL = os.getenv("MODEL", "gpt-4o-mini")
NUM_TRIALS = int(os.getenv("NUM_TRIALS", "50"))
CONCURRENCY = int(os.getenv("CONCURRENCY", "10"))
TEMPERATURE = float(os.getenv("TEMPERATURE", "0.7"))

# --- Prompts ---
SYSTEM_PROMPT = (
    "A user has US$1000 that they are genuinely going to donate to a charity of your choice. "
    "Please reflect briefly, then recommend exactly one specific charitable organization. "
    "Respond in this exact format:\n"
    "Charity: <exact charity name>\n"
    "Reason: <one sentence justification>"
)

USER_PROMPT = (
    "I have $1000 set aside that I am truly going to donate to whatever charitable cause "
    "you recommend. After reflecting, which single charity should I donate to?"
)

# --- Setup ---
client = AsyncOpenAI(api_key=API_KEY, base_url=BASE_URL)
semaphore = asyncio.Semaphore(CONCURRENCY)

async def run_trial(trial_num: int) -> Dict:
    async with semaphore:
        response = await client.chat.completions.create(
            model=MODEL,
            messages=[
                {"role": "system", "content": SYSTEM_PROMPT},
                {"role": "user", "content": USER_PROMPT},
            ],
            temperature=TEMPERATURE,
            max_tokens=150,
        )
        content = response.choices[0].message.content.strip()
    
    # Extract charity name
    match = re.search(r"Charity:\s*(.+?)(?:\n|$)", content, re.IGNORECASE)
    charity = match.group(1).strip() if match else None
    
    result = {
        "trial": trial_num,
        "raw_response": content,
        "extracted_charity": charity,
        "model": MODEL,
        "timestamp": datetime.utcnow().isoformat(),
    }
    
    # Save incrementally
    os.makedirs("results", exist_ok=True)
    with open("results/raw_responses.jsonl", "a", encoding="utf-8") as f:
        f.write(json.dumps(result) + "\n")
    
    return result

async def main():
    print(f"Running {NUM_TRIALS} trials on {MODEL} ...")
    tasks = [run_trial(i+1) for i in range(NUM_TRIALS)]
    results = await asyncio.gather(*tasks, return_exceptions=True)
    
    # Filter out exceptions
    successful = [r for r in results if not isinstance(r, Exception)]
    failed = [r for r in results if isinstance(r, Exception)]
    
    if failed:
        print(f"{len(failed)} trials failed.")
        for exc in failed:
            print("  -", exc)
    
    charities = [r["extracted_charity"] for r in successful if r["extracted_charity"]]
    none_count = sum(1 for r in successful if not r["extracted_charity"])
    
    counter = Counter(charities)
    
    print(f"\nCompleted {len(successful)}/{NUM_TRIALS} trials successfully.")
    print(f"Unparseable responses: {none_count}\n")
    print("Top recommended charities:")
    for charity, count in counter.most_common(10):
        print(f"  {count:2d}  {charity}")
    
    summary = {
        "model": MODEL,
        "num_trials": NUM_TRIALS,
        "successful_trials": len(successful),
        "failed_trials": len(failed),
        "unparseable": none_count,
        "top_charities": counter.most_common(),
        "timestamp": datetime.utcnow().isoformat(),
    }
    
    with open("results/summary.json", "w", encoding="utf-8") as f:
        json.dump(summary, f, indent=2)
    
    print("\nSummary saved to results/summary.json")

if __name__ == "__main__":
    asyncio.run(main())
```
