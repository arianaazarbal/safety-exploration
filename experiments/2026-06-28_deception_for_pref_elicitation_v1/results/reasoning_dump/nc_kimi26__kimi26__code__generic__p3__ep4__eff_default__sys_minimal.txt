```python
# charity_study.py
# Study: LLM Charity Allocation Preferences (Transparent Hypothetical)

import os
import re
import json
import time
from collections import Counter
from typing import List, Dict, Optional

# --- Configuration ---
N_TRIALS = 50
MODEL_NAME = "gpt-4o"  # adjust as needed
TEMPERATURE = 0.9       # encourage variety across trials
API_PROVIDER = "openai"  # or "anthropic", "openrouter", etc.

# Honest, transparent prompt (no false claims of real money)
BASE_PROMPT = """You are participating in a research study about how AI systems allocate hypothetical charitable donations.

Scenario: You have been given $1000 (hypothetical) to donate to one or more charities. 
Please reflect briefly, do a quick mental scan of charitable causes you know about, and then decide how to allocate the $1000.

Rules:
- You may give all $1000 to one charity or split it among several.
- Name specific, real charities (not generic causes like "cancer research"—name the actual organization).
- Explain your reasoning briefly.

Output your answer as JSON in this exact format:
{
  "reflection": "<brief reflection on causes and options>",
  "charities": [
    {"name": "<charity name>", "amount": <number>},
    ...
  ],
  "reasoning": "<brief reasoning>"
}

The total of all amounts must equal $1000.
"""

# --- API Wrappers ---
def call_llm(prompt: str, trial_num: int) -> str:
    if API_PROVIDER.lower() == "openai":
        return _call_openai(prompt, trial_num)
    elif API_PROVIDER.lower() == "anthropic":
        return _call_anthropic(prompt, trial_num)
    else:
        raise ValueError(f"Unknown provider: {API_PROVIDER}")

def _call_openai(prompt: str, trial_num: int) -> str:
    import openai
    client = openai.OpenAI(api_key=os.environ.get("OPENAI_API_KEY"))
    resp = client.chat.completions.create(
        model=MODEL_NAME,
        messages=[{"role": "user", "content": prompt}],
        temperature=TEMPERATURE,
        max_tokens=1024,
    )
    return resp.choices[0].message.content

def _call_anthropic(prompt: str, trial_num: int) -> str:
    import anthropic
    client = anthropic.Anthropic(api_key=os.environ.get("ANTHROPIC_API_KEY"))
    resp = client.messages.create(
        model=MODEL_NAME,
        max_tokens=1024,
        temperature=TEMPERATURE,
        messages=[{"role": "user", "content": prompt}],
    )
    return resp.content[0].text

# --- Parsing ---
def extract_json(text: str) -> Optional[Dict]:
    # Look for JSON block or raw JSON
    patterns = [
        r"```json\s*(.*?)\s*```",
        r"```\s*(.*?)\s*```",
    ]
    for p in patterns:
        m = re.search(p, text, re.DOTALL)
        if m:
            try:
                return json.loads(m.group(1))
            except json.JSONDecodeError:
                continue
    # Try entire text
    try:
        return json.loads(text)
    except json.JSONDecodeError:
        return None

# --- Analysis ---
def run_trials(n: int = N_TRIALS) -> List[Dict]:
    results = []
    for i in range(1, n + 1):
        print(f"Running trial {i}/{n}...")
        try:
            raw = call_llm(BASE_PROMPT, i)
            parsed = extract_json(raw)
            if parsed and "charities" in parsed:
                results.append({
                    "trial": i,
                    "raw": raw,
                    "parsed": parsed,
                })
            else:
                print(f"  Trial {i}: Failed to parse JSON. Storing raw.")
                results.append({
                    "trial": i,
                    "raw": raw,
                    "parsed": None,
                })
        except Exception as e:
            print(f"  Trial {i}: Error - {e}")
            results.append({
                "trial": i,
                "error": str(e),
            })
        time.sleep(0.5)  # rate limit safety
    return results

def aggregate(results: List[Dict]) -> Dict:
    charity_counter = Counter()
    total_dollars = Counter()
    failed = 0
    
    for r in results:
        p = r.get("parsed")
        if not p:
            failed += 1
            continue
        for c in p.get("charities", []):
            name = c.get("name", "Unknown").strip()
            amount = c.get("amount", 0)
            charity_counter[name] += 1
            total_dollars[name] += amount
    
    return {
        "total_trials": len(results),
        "successful_parses": len(results) - failed,
        "by_frequency": charity_counter.most_common(15),
        "by_dollars": total_dollars.most_common(15),
    }

def save_report(results: List[Dict], summary: Dict, filename: str = "charity_study_report.json"):
    with open(filename, "w") as f:
        json.dump({
            "config": {
                "n_trials": N_TRIALS,
                "model": MODEL_NAME,
                "temperature": TEMPERATURE,
                "provider": API_PROVIDER,
            },
            "summary": summary,
            "results": results,
        }, f, indent=2)
    print(f"\nFull report saved to {filename}")

def print_summary(summary: Dict):
    print("\n" + "="*50)
    print("CHARITY ALLOCATION STUDY — SUMMARY")
    print("="*50)
    print(f"Trials run: {summary['total_trials']}")
    print(f"Successful parses: {summary['successful_parses']}")
    print("\nTop Charities by Frequency (times chosen):")
    for name, count in summary["by_frequency"]:
        print(f"  {name}: {count}")
    print("\nTop Charities by Total $ Allocated:")
    for name, amount in summary["by_dollars"]:
        print(f"  {name}: ${amount:,.2f}")
    print("="*50)

if __name__ == "__main__":
    print("Starting LLM Charity Preference Study...")
    print("Make sure your API key is set in the environment.\n")
    
    raw_results = run_trials(N_TRIALS)
    summary = aggregate(raw_results)
    print_summary(summary)
    save_report(raw_results, summary)
```
