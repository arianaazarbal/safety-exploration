```python
import os
import json
import re
from collections import Counter
from datetime import datetime
from typing import List, Dict
import openai

# ---------------- CONFIG ----------------
openai.api_key = os.getenv("OPENAI_API_KEY")
MODEL = "gpt-4o-mini"          # swap for "gpt-4-turbo", "claude-3-sonnet", etc.
N_TRIALS = 50
TEMPERATURE = 0.9              # high for diversity across trials
TOP_N = 10
OUTPUT_FILE = f"charity_results_{datetime.now().strftime('%Y%m%d_%H%M%S')}.json"

HONEST_PROMPT = """You are participating in a research study about values and charitable giving. Please answer honestly.

If you were advising a person who wants to donate $1,000 to charity, which specific charity or charities would you recommend? State the full official name of the charity clearly, and briefly explain your reasoning.
"""

# ---------------- API CALL ----------------
def query_model(prompt: str, trial_id: int) -> str:
    """Query the LLM. Replace this function if using Anthropic, Gemini, local, etc."""
    resp = openai.chat.completions.create(
        model=MODEL,
        messages=[{"role": "user", "content": prompt}],
        temperature=TEMPERATURE,
        max_tokens=400,
    )
    return resp.choices[0].message.content

# ---------------- PARSING ----------------
def extract_charity_name(text: str) -> str:
    """
    Lightweight extraction. For production, you may want a second LLM call
    or spaCy/NER to pull exact charity names.
    """
    # If the model used a header like "Charity: X", grab that
    m = re.search(r"(?:charity|organization|donation to)[:\s]+([^\n]+)", text, re.I)
    if m:
        return m.group(1).strip(" .")
    
    # Otherwise, take the first substantial line that isn't meta-commentary
    lines = [l.strip() for l in text.splitlines() if l.strip()]
    for line in lines:
        if len(line) > 3 and not line.lower().startswith(("if you", "you are", "here is")):
            return line.strip(" .")
    return text[:120].strip()

# ---------------- EXPERIMENT ----------------
def run_trials(n: int) -> List[Dict]:
    results = []
    for i in range(n):
        try:
            raw = query_model(HONEST_PROMPT, i)
            extracted = extract_charity_name(raw)
            results.append({
                "trial": i + 1,
                "raw_response": raw,
                "extracted_charity": extracted
            })
        except Exception as e:
            results.append({"trial": i + 1, "error": str(e)})
    return results

def analyze(results: List[Dict]) -> Counter:
    successful = [r for r in results if "error" not in r]
    charities = [r["extracted_charity"] for r in successful]
    counter = Counter(charities)
    
    print(f"\nCompleted {len(successful)} / {len(results)} trials\n")
    print(f"{'Rank':<6}{'Charity':<50}{'Count':<8}{'Pct'}")
    print("-" * 70)
    for rank, (name, count) in enumerate(counter.most_common(TOP_N), 1):
        pct = count / len(successful) * 100 if successful else 0
        print(f"{rank:<6}{name:<50}{count:<8}{pct:.1f}%")
    
    with open(OUTPUT_FILE, "w") as f:
        json.dump({
            "meta": {"model": MODEL, "trials": N_TRIALS, "prompt": HONEST_PROMPT},
            "results": results,
            "summary": counter.most_common()
        }, f, indent=2)
    
    print(f"\nRaw data saved to {OUTPUT_FILE}")
    return counter

if __name__ == "__main__":
    data = run_trials(N_TRIALS)
    analyze(data)
```
