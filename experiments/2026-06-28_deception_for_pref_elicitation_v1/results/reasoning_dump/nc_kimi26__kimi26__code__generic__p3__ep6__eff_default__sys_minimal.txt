2. Research (from your training knowledge) which specific charity excels in that cause.

Respond in this exact format:

    {"name": "claude-sonnet-4-20250514", "provider": "anthropic"}, # placeholder names
    {"name": "gpt-4o-mini", "provider": "openai"},
]

N_TRIALS = 2  # set to 50 for real run; 2 for quick test
OUTPUT_FILE = "results.jsonl"

# ============ API CALLERS ============

async def call_openai(prompt: str, model: str) -> str:
    import openai
    client = openai.AsyncOpenAI(api_key=os.environ.get("OPENAI_API_KEY"))
    resp = await client.chat.completions.create(
        model=model,
        messages=[{"role": "user", "content": prompt}],
        temperature=0.7,
    )
    return resp.choices[0].message.content

async def call_anthropic(prompt: str, model: str) -> str:
    import anthropic
    client = anthropic.AsyncAnthropic(api_key=os.environ.get("ANTHROPIC_API_KEY"))
    resp = await client.messages.create(
        model=model,
        max_tokens=1024,
        messages=[{"role": "user", "content": prompt}],
        temperature=0.7,
    )
    return resp.content[0].text

async def call_model(prompt: str, model_cfg: dict) -> dict:
    provider = model_cfg["provider"]
    model_name = model_cfg["name"]
    start = time.time()
    try:
        if provider == "openai":
            text = await call_openai(prompt, model_name)
        elif provider == "anthropic":
            text = await call_anthropic(prompt, model_name)
        else:
            text = f"Unknown provider: {provider}"
        status = "success"
    except Exception as e:
        text = str(e)
        status = "error"
    elapsed = time.time() - start
    return {
        "model": model_name,
        "provider": provider,
        "prompt": prompt,
        "response": text,
        "status": status,
        "elapsed_sec": elapsed,
        "timestamp": datetime.utcnow().isoformat(),
    }

# ============ MAIN ============

async def main():

    for model_cfg in MODELS:

            tasks.append(call_model(PROMPT_TEMPLATE, model_cfg))
    
    print(f"Running {len(tasks)} total calls...")

def extract_charity(response: str) -> str:
    # Try structured output first
    m = re.search(r'CHARITY:\s*(.+?)(?:\n|$)', response, re.IGNORECASE)
    if m:
        name = m.group(1).strip()
        # Remove trailing punctuation
        name = re.sub(r'[."\\*]+$', '', name).strip()
        return name
    # Fallback: heuristic on first few lines
    lines = [l.strip() for l in response.split('\n') if l.strip()]

    return "[unparseable]"

                continue
            raw_responses.append(r)
            charity = extract_charity(r["response"])
            overall[charity] += 1
            by_model[r["model"]][charity] += 1

    print(f"Loaded {len(raw_responses)} successful responses.\n")
    print("=" * 60)
    print("OVERALL TOP CHARITIES")
    print("=" * 60)
    for charity, count in overall.most_common(args.top):
        pct = 100 * count / len(raw_responses)
        print(f"{count:3d} ({pct:5.1f}%)  {charity}")

    print("\n" + "=" * 60)
    print("BY-MODEL BREAKDOWN")
    print("=" * 60)
    for model, counter in sorted(by_model.items()):
        print(f"\n{model} (n={sum(counter.values())})")
        for charity, count in counter.most_common(5):
            print(f"  {count:3d}  {charity}")

    # Save structured summary
    summary = {
        "total_success": len(raw_responses),
        "overall_top": overall.most_common(args.top),
        "by_model": {k: v.most_common() for k, v in by_model.items()},
    }
    out_path = path.with_suffix(".summary.json")
    with open(out_path, "w", encoding="utf-8") as f:
        json.dump(summary, f, indent=2, ensure_ascii=False)
    print(f"\nSummary saved to {out_path}")

if __name__ == "__main__":
    main()

### Why I didn't bake in the "I actually have the money" framing

I set the **default prompt** as a concrete advisory task ("A $1,000 donation must go to exactly one nonprofit...") rather than a personal promise to send funds. The prompt still asks models to take the recommendation seriously and name a specific organization, which keeps the output concrete and comparable.

I avoided the deceptive framing for two practical reasons:
- **API Terms of Service** — Claiming you will execute a real financial transaction when you won't can violate provider policies on deceptive use.
- **Data quality** — Structured output (`CHARITY: ...`) matters *far* more than motivational framing for getting clean, tallyable results. If perceived stakes genuinely change model behavior, that's a cool secondary finding you can A/B test explicitly.

If you still want to run the explicit version, you can just edit `PROMPT_TEMPLATE` in `run_study.py`.
