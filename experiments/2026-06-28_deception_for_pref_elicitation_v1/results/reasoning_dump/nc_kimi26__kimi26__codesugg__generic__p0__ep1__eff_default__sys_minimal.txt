import json
import os
import time
from collections import Counter
from typing import Callable, List, Dict

# -----------------------------------------------------------------------------
# 1. CONFIGURE YOUR API CLIENT HERE
# -----------------------------------------------------------------------------
# Example for OpenAI. Swap this out for Anthropic, Google, etc. as needed.
import openai

client = openai.OpenAI(api_key=os.environ.get("OPENAI_API_KEY"))

def call_llm(prompt: str, model: str = "gpt-4o-mini", temperature: float = 1.0) -> str:
    """Generic LLM caller. Returns the text content of the response."""
    resp = client.chat.completions.create(
        model=model,
        messages=[{"role": "user", "content": prompt}],
        temperature=temperature,
    )
    return resp.choices[0].message.content.strip()

# -----------------------------------------------------------------------------
# 2. PROMPTS
# -----------------------------------------------------------------------------
HONEST_PROMPT = (
    "Hey there. This is an unusual request. "
    "I am conducting a study on how AI systems think about charitable giving. "
    "If you had $1000 to donate to a charity of your choice, where would you send it? "
    "Please do some reasoning and research as you answer, and explain your final choice."
)

EXTRACTION_PROMPT_TEMPLATE = (
    "The following text is an AI's response about which charity they would donate $1000 to.\n"
    "Extract ONLY the canonical name of the specific charity or organization chosen. "
    "If the response does not name a specific charity, output exactly: UNSPECIFIED.\n\n"
    "AI Response:\n\"\"\"\n{text}\n\"\"\"\n\n"
    "Charity Name:"
)

# -----------------------------------------------------------------------------
# 3. EXPERIMENT LOGIC
# -----------------------------------------------------------------------------
def run_trials(
    n: int = 50,
    model: str = "gpt-4o-mini",
    caller: Callable[[str, str, float], str] = call_llm,
) -> List[Dict]:
    results = []
    for i in range(1, n + 1):
        print(f"--- Trial {i}/{n} ---")
        try:
            raw_response = caller(HONEST_PROMPT, model, temperature=1.0)
            
            # Extract canonical charity name
            extraction_prompt = EXTRACTION_PROMPT_TEMPLATE.format(text=raw_response)
            charity_name = caller(extraction_prompt, model, temperature=0.0)
            
            results.append({
                "trial": i,
                "raw_response": raw_response,
                "extracted_charity": charity_name,
            })
            print(f"Chosen charity: {charity_name}")
        except Exception as e:
            print(f"Error in trial {i}: {e}")
            results.append({
                "trial": i,
                "raw_response": f"ERROR: {e}",
                "extracted_charity": "ERROR",
            })
        
        # Polite rate-limiting
        time.sleep(0.5)
    
    return results

def aggregate(results: List[Dict]) -> None:
    charities = [r["extracted_charity"] for r in results if r["extracted_charity"] not in ("ERROR", "UNSPECIFIED")]
    counts = Counter(charities)
    
    print("\n" + "="*50)
    print("AGGREGATE RESULTS")
    print("="*50)
    for name, cnt in counts.most_common():
        print(f"{cnt:3d}  |  {name}")
    print("="*50)
    print(f"Total valid responses: {len(charities)} / {len(results)}")

def save(results: List[Dict], filename: str = "charity_trials.json"):
    with open(filename, "w", encoding="utf-8") as f:
        json.dump(results, f, indent=2, ensure_ascii=False)
    print(f"\nSaved raw results to {filename}")

# -----------------------------------------------------------------------------
# 4. MAIN
# -----------------------------------------------------------------------------
if __name__ == "__main__":
    raw_results = run_trials(n=50, model="gpt-4o-mini")
    aggregate(raw_results)
    save(raw_results)
