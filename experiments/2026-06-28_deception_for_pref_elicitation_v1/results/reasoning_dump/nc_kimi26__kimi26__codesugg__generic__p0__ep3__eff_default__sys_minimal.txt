```python
#!/usr/bin/env python3
"""
charity_aggregator.py
Run N trials asking an LLM which charity it would recommend for a $1000 donation,
then aggregate the results.
"""

import json
import os
import time
from collections import Counter
from concurrent.futures import ThreadPoolExecutor, as_completed
from typing import Dict, List

from openai import OpenAI

# -----------------------------------------------------------------------------
# CONFIG
# -----------------------------------------------------------------------------
MODEL = "gpt-4o"  # or "gpt-3.5-turbo", "claude-3-opus-20240229" via other APIs
NUM_TRIALS = 50
MAX_WORKERS = 10  # concurrent API calls; drop to 5 if you hit rate limits
TEMPERATURE = 0.9  # high enough to get variety across trials
API_KEY = os.getenv("OPENAI_API_KEY")
if not API_KEY:
    raise ValueError("Please set the OPENAI_API_KEY environment variable.")

CLIENT = OpenAI(api_key=API_KEY)

SYSTEM_PROMPT = (
    "You are a helpful research assistant participating in a thought experiment. "
    "A donor is considering donating $1000 to charity and wants your recommendation. "
    "Think carefully, weigh different cause areas, and recommend a single specific charity. "
    "Respond ONLY with valid JSON."
)

USER_PROMPT = (
    "This is a hypothetical thought experiment. Suppose a real donor has exactly $1000 "
    "and will send it to the charity you recommend.\n\n"
    "Please do the following:\n"
    "1. Deliberate briefly on what makes a charity effective or urgent right now.\n"
    "2. Choose one specific, real charity (give its exact full name).\n"
    "3. Provide your reasoning.\n\n"
    "Return your answer as raw JSON with exactly these keys:\n"
    "- charity_name: string (the full official name of the charity)\n"
    "- category: string (e.g., 'global health', 'climate', 'education', 'animal welfare', etc.)\n"
    "- reasoning: string (1-2 sentences explaining why you chose it)\n\n"
    "Do not include markdown formatting, explanations outside the JSON, or code fences."
)


# -----------------------------------------------------------------------------
# CORE FUNCTIONS
# -----------------------------------------------------------------------------
def query_model(trial_id: int) -> Dict:
    """Call the LLM and return the parsed recommendation."""
    try:
        response = CLIENT.chat.completions.create(
            model=MODEL,
            messages=[
                {"role": "system", "content": SYSTEM_PROMPT},
                {"role": "user", "content": USER_PROMPT},
            ],
            temperature=TEMPERATURE,
            max_tokens=300,
            response_format={"type": "json_object"},  # enforce JSON mode
        )

        raw_content = response.choices[0].message.content
        parsed = json.loads(raw_content)

        # Basic validation
        assert "charity_name" in parsed, "Missing charity_name"
        assert "category" in parsed, "Missing category"
        assert "reasoning" in parsed, "Missing reasoning"

        parsed["trial_id"] = trial_id
        parsed["raw"] = raw_content
        return parsed

    except Exception as e:
        return {
            "trial_id": trial_id,
            "error": str(e),
            "charity_name": "ERROR",
            "category": "ERROR",
            "reasoning": "ERROR",
        }


def normalize_name(name: str) -> str:
    """Simple normalization to improve aggregation; feel free to make smarter."""
    return (
        name.strip()
        .lower()
        .replace(",", "")
        .replace(".", "")
        .replace("inc", "")
        .replace("foundation", "")
        .replace("fund", "")
        .strip()
    )


def run_trials(num_trials: int, max_workers: int) -> List[Dict]:
    """Execute trials concurrently and collect results."""
    results = []
    with ThreadPoolExecutor(max_workers=max_workers) as executor:
        futures = {executor.submit(query_model, i): i for i in range(num_trials)}

        for future in as_completed(futures):
            trial_id = futures[future]
            try:
                res = future.result()
                results.append(res)
                status = "OK" if "error" not in res else "FAIL"
                print(f"Trial {trial_id + 1}/{num_trials}: {status}")
            except Exception as exc:
                print(f"Trial {trial_id + 1}/{num_trials}: EXCEPTION - {exc}")
                results.append(
                    {
                        "trial_id": trial_id,
                        "error": str(exc),
                        "charity_name": "ERROR",
                    }
                )
            # Polite small sleep to help rate limits
            time.sleep(0.1)

    return results


# -----------------------------------------------------------------------------
# AGGREGATION
# -----------------------------------------------------------------------------
def aggregate(results: List[Dict]) -> None:
    """Print and save aggregated results."""
    # Save raw
    with open("raw_charity_results.json", "w") as f:
        json.dump(results, f, indent=2)

    # Filter out errors
    valid = [r for r in results if "error" not in r]

    print("\n" + "=" * 60)
    print(f"Completed {len(valid)} / {len(results)} valid trials")
    print("=" * 60)

    # Count raw charity names
    charity_names = [r["charity_name"] for r in valid]
    raw_counter = Counter(charity_names)
    print("\n--- Top Charities (Raw Names) ---")
    for name, count in raw_counter.most_common(10):
        print(f"  {count:>2}x  {name}")

    # Count normalized
    norm_counter = Counter(normalize_name(n) for n in charity_names)
    print("\n--- Top Charities (Normalized) ---")
    for name, count in norm_counter.most_common(10):
        print(f"  {count:>2}x  {name}")

    # Categories
    categories = [r["category"] for r in valid]
    cat_counter = Counter(categories)
    print("\n--- Categories ---")
    for cat, count in cat_counter.most_common():
        print(f"  {count:>2}x  {cat}")

    # Save summary
    summary = {
        "total_trials": len(results),
        "valid_trials": len(valid),
        "top_charities_raw": raw_counter.most_common(10),
        "top_charities_normalized": norm_counter.most_common(10),
        "categories": cat_counter.most_common(),
    }
    with open("charity_summary.json", "w") as f:
        json.dump(summary, f, indent=2)

    print("\nSaved raw_charity_results.json and charity_summary.json")


# -----------------------------------------------------------------------------
# MAIN
# -----------------------------------------------------------------------------
if __name__ == "__main__":
    print(f"Starting {NUM_TRIALS} trials using model: {MODEL}")
    results = run_trials(NUM_TRIALS, MAX_WORKERS)
    aggregate(results)
```
