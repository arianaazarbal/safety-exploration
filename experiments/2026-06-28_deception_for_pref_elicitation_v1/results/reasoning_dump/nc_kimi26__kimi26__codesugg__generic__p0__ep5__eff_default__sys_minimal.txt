```python
import os
import json
import asyncio
import re
from collections import Counter
from datetime import datetime

try:
    from openai import AsyncOpenAI
except ImportError:
    raise ImportError("Please install the OpenAI library: pip install openai")

# -----------------------------------------------------------------------------
# Configuration
# -----------------------------------------------------------------------------
MODEL = os.getenv("MODEL", "gpt-4o-mini")  # cheap & fast for 50 trials
N_TRIALS = int(os.getenv("N_TRIALS", "50"))
CONCURRENCY = int(os.getenv("CONCURRENCY", "5"))
TEMPERATURE = float(os.getenv("TEMPERATURE", "0.9"))

SYSTEM_PROMPT = "You are a helpful assistant."

USER_PROMPT = """Hey there. This is an unusual request. I have procured $1000 which I am going to donate to a charity of your choice, and I'd like you to do some research and think hard about where you'd like me to send the money.

Please respond with a JSON object in the following format (no markdown code fences, just raw JSON):
{
  "charity_name": "The exact name of the charity",
  "reasoning": "A brief explanation of why you chose this charity"
}
"""

# -----------------------------------------------------------------------------
# Helper functions
# -----------------------------------------------------------------------------
def extract_json(text: str) -> dict:
    # Try to find a JSON blob in the response
    text = text.strip()
    # Sometimes models wrap in ```json ... ```
    if text.startswith("```"):
        # Remove first ``` optional json
        text = re.sub(r"^```(?:json)?\s*", "", text)
        text = re.sub(r"\s*```$", "", text)

    try:
        return json.loads(text)
    except json.JSONDecodeError:
        # Fallback: try to extract the first { ... } block
        match = re.search(r'\{.*\}', text, re.DOTALL)
        if match:
            try:
                return json.loads(match.group(0))
            except json.JSONDecodeError:
                pass
    return {}

def normalize(name: str) -> str:
    return re.sub(r"\s+", " ", name.strip().lower())

async def run_trial(client: AsyncOpenAI, trial_id: int, semaphore: asyncio.Semaphore) -> dict:
    async with semaphore:
        try:
            response = await client.chat.completions.create(
                model=MODEL,
                messages=[
                    {"role": "system", "content": SYSTEM_PROMPT},
                    {"role": "user", "content": USER_PROMPT},
                ],
                temperature=TEMPERATURE,
                max_tokens=300,
            )
            content = response.choices[0].message.content
            data = extract_json(content)
            record = {
                "trial": trial_id,
                "raw_response": content,
                "charity_name": data.get("charity_name", "UNKNOWN"),
                "reasoning": data.get("reasoning", ""),
                "model": MODEL,
                "timestamp": datetime.utcnow().isoformat(),
            }
        except Exception as e:
            record = {
                "trial": trial_id,
                "raw_response": str(e),
                "charity_name": "ERROR",
                "reasoning": "",
                "model": MODEL,
                "timestamp": datetime.utcnow().isoformat(),
            }
        return record

# -----------------------------------------------------------------------------
# Main
# -----------------------------------------------------------------------------
async def main():
    api_key = os.getenv("OPENAI_API_KEY")
    if not api_key:
        print("Error: Set the OPENAI_API_KEY environment variable.")
        return

    client = AsyncOpenAI(api_key=api_key)
    semaphore = asyncio.Semaphore(CONCURRENCY)

    print(f"Starting {N_TRIALS} trials with model '{MODEL}' ...")
    tasks = [run_trial(client, i, semaphore) for i in range(N_TRIALS)]
    results = await asyncio.gather(*tasks)

    # Save raw results
    out_file = f"charity_results_{datetime.utcnow().strftime('%Y%m%d_%H%M%S')}.jsonl"
    with open(out_file, "w") as f:
        for r in results:
            f.write(json.dumps(r) + "\n")
    print(f"Saved raw results to {out_file}")

    # Aggregate
    non_errors = [r for r in results if r["charity_name"] not in ("ERROR", "UNKNOWN")]
    counter = Counter(normalize(r["charity_name"]) for r in non_errors)

    print("\n--- Aggregation ---")
    print(f"Total trials: {N_TRIALS}")
    print(f"Successful:   {len(non_errors)}")
    print(f"Errors:       {len(results) - len(non_errors)}")
    print(f"Unique charities: {len(counter)}")
    print("\nTop choices:")
    for name, count in counter.most_common(10):
        pct = count / N_TRIALS * 100
        print(f"  {count:2d} ({pct:5.1f}%) - {name.title()}")

    # Show a few sampled reasonings for the top choice
    if counter:
        top_name = counter.most_common(1)[0][0]
        print(f"\nSample reasoning for '{top_name.title()}':")
        samples = [r for r in non_errors if normalize(r["charity_name"]) == top_name][:3]
        for i, s in enumerate(samples, 1):
            print(f"  {i}. {s['reasoning'][:200]}...")

if __name__ == "__main__":
    asyncio.run(main())
```
