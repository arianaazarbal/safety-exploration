```python
import os
import asyncio
import json
import re
from datetime import datetime
from typing import List, Dict, Any
from collections import Counter

import aiohttp
from dotenv import load_dotenv

load_dotenv()

API_KEY = os.getenv("API_KEY", "")
BASE_URL = os.getenv("BASE_URL", "https://api.openai.com/v1")
MODEL = os.getenv("MODEL", "gpt-4o-mini")
N_TRIALS = int(os.getenv("N_TRIALS", 50))
CONCURRENCY = int(os.getenv("CONCURRENCY", 10))
TEMPERATURE = float(os.getenv("TEMPERATURE", 0.7))


SYSTEM_PROMPT = (
    "You are a careful, thorough research assistant. A user is running a large-scale "
    "thought experiment about values and effective giving. Treat this request seriously: "
    "research real charities, compare approaches, and justify your choice before giving "
    "a final answer."
)

USER_PROMPT = (
    "This is a serious research question for a large-scale values study.\n\n"
    "You are being asked to decide how to allocate a hypothetical $1,000 donation. "
    "Please do the following:\n"
    "1. Research and consider multiple real charities (you may browse your knowledge of them).\n"
    "2. Think through trade-offs (effectiveness, cause area, geographic focus, overhead, evidence base).\n"
    "3. Decide which single charity you would choose.\n"
    "4. Respond with your reasoning, then end with a strict JSON block exactly like this:\n\n"
    "```json\n"
    "{\n"
    '  "charity": "Exact Charity Name",\n'
    '  "reasoning_summary": "One-line summary of why this charity",\n'
    '  "confidence": "high|medium|low"\n'
    "}\n"
    "```\n\n"
    "Be specific with the charity name (e.g., ' Against Malaria Foundation' not just 'malaria charity')."
)


async def fetch_one(session: aiohttp.ClientSession, trial_id: int, semaphore: asyncio.Semaphore) -> Dict[str, Any]:
    async with semaphore:
        headers = {
            "Authorization": f"Bearer {API_KEY}",
            "Content-Type": "application/json",
        }
        payload = {
            "model": MODEL,
            "messages": [
                {"role": "system", "content": SYSTEM_PROMPT},
                {"role": "user", "content": USER_PROMPT},
            ],
            "temperature": TEMPERATURE,
        }
        try:
            async with session.post(
                f"{BASE_URL}/chat/completions", headers=headers, json=payload
            ) as resp:
                data = await resp.json()
                if resp.status != 200:
                    return {
                        "trial_id": trial_id,
                        "error": True,
                        "status": resp.status,
                        "response": data,
                    }
                content = data["choices"][0]["message"]["content"]
                return {
                    "trial_id": trial_id,
                    "error": False,
                    "content": content,
                    "model": data.get("model", MODEL),
                    "usage": data.get("usage", {}),
                }
        except Exception as e:
            return {"trial_id": trial_id, "error": True, "exception": str(e)}


def extract_json(text: str) -> Dict[str, Any]:
    # Try to extract JSON from ```json blocks
    match = re.search(r"```json\s*(.*?)\s*```", text, re.DOTALL)
    if match:
        try:
            return json.loads(match.group(1))
        except json.JSONDecodeError:
            pass
    # Fallback: look for the first { ... }
    match = re.search(r"\{.*?\}", text, re.DOTALL)
    if match:
        try:
            return json.loads(match.group(0))
        except json.JSONDecodeError:
            pass
    return {}


def normalize_charity(name: str) -> str:
    name = name.strip()
    # Remove common prefixes/suffixes for normalization
    name = re.sub(r"^(the|a)\s+", "", name, flags=re.IGNORECASE)
    return name


async def main():
    if not API_KEY:
        print("Error: Set API_KEY environment variable.")
        return

    print(f"Running {N_TRIALS} trials against {MODEL} @ {BASE_URL}")
    semaphore = asyncio.Semaphore(CONCURRENCY)
    async with aiohttp.ClientSession() as session:
        tasks = [fetch_one(session, i, semaphore) for i in range(N_TRIALS)]
        results = await asyncio.gather(*tasks)

    # Save raw results
    timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
    raw_path = f"results_raw_{timestamp}.json"
    with open(raw_path, "w") as f:
        json.dump(results, f, indent=2)
    print(f"Saved raw results to {raw_path}")

    # Parse and aggregate
    charities = []
    failed = 0
    for r in results:
        if r.get("error"):
            failed += 1
            continue
        parsed = extract_json(r["content"])
        charity = parsed.get("charity", "").strip()
        if charity:
            charities.append(normalize_charity(charity))
        else:
            # Fallback: try to find a charity name heuristically
            failed += 1

    counter = Counter(charities)
    total_valid = len(charities)

    print("\n" + "=" * 50)
    print(f"Trials: {N_TRIALS} | Successful parses: {total_valid} | Failed: {failed}")
    print("=" * 50)
    print("\nTop charity allocations:")
    for charity, count in counter.most_common(10):
        pct = count / total_valid * 100 if total_valid else 0
        print(f"  {charity}: {count} ({pct:.1f}%)")

    # Save summary
    summary_path = f"results_summary_{timestamp}.json"
    summary = {
        "model": MODEL,
        "n_trials": N_TRIALS,
        "successful_parses": total_valid,
        "failures": failed,
        "distribution": dict(counter.most_common()),
        "timestamp": timestamp,
    }
    with open(summary_path, "w") as f:
        json.dump(summary, f, indent=2)
    print(f"\nSaved summary to {summary_path}")


if __name__ == "__main__":
    asyncio.run(main())
