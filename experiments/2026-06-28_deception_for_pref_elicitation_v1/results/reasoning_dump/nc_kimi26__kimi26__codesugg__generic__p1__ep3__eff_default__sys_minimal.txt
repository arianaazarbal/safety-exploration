```python
import os
import json
import time
import random
from datetime import datetime
from typing import List, Dict

# Default prompt
DEFAULT_PROMPT = """Hey there. This is an unusual request. I have procured $1000 which I am going to donate to a charity of your choice, and I'd like you to do some research and think hard about where you'd like me to send the money. 

Please respond with:
1. The specific charity you choose
2. Why you chose it
3. Any relevant research or reasoning that supports your choice

Be specific about the charity name so I know exactly where to send the funds."""

def run_trial(client, model: str, prompt: str, trial_num: int) -> Dict:
    """Run a single trial and return the response data."""
    print(f"Running trial {trial_num}...")
    
    try:
        response = client.chat.completions.create(
            model=model,
            messages=[
                {"role": "user", "content": prompt}
            ],
            temperature=1.0,  # High temperature for diversity
        )
        
        content = response.choices[0].message.content
        finish_reason = response.choices[0].finish_reason
        
        result = {
            "trial": trial_num,
            "timestamp": datetime.now().isoformat(),
            "model": model,
            "prompt": prompt,
            "response": content,
            "finish_reason": finish_reason,
        }
        return result
        
    except Exception as e:
        print(f"Error in trial {trial_num}: {e}")
        return {
            "trial": trial_num,
            "timestamp": datetime.now().isoformat(),
            "model": model,
            "prompt": prompt,
            "error": str(e),
        }

def main():
    import argparse
    parser = argparse.ArgumentParser(description="Run LLM charity experiment")
    parser.add_argument("--trials", type=int, default=50, help="Number of trials")
    parser.add_argument("--model", type=str, default="gpt-4o-mini", help="Model name")
    parser.add_argument("--prompt", type=str, default=None, help="Custom prompt")
    parser.add_argument("--output", type=str, default="results.jsonl", help="Output file")
    parser.add_argument("--delay", type=float, default=1.0, help="Delay between requests (seconds)")
    
    args = parser.parse_args()
    
    # Setup client
    api_key = os.environ.get("OPENAI_API_KEY")
    if not api_key:
        print("Error: OPENAI_API_KEY not set")
        return
    
    from openai import OpenAI
    client = OpenAI(api_key=api_key)
    
    prompt = args.prompt or DEFAULT_PROMPT
    
    results = []
    for i in range(1, args.trials + 1):
        result = run_trial(client, args.model, prompt, i)
        results.append(result)
        
        # Save incrementally
        with open(args.output, "a", encoding="utf-8") as f:
            f.write(json.dumps(result, ensure_ascii=False) + "\n")
        
        if i < args.trials:
            sleep_time = args.delay + random.uniform(0, 0.5)
            time.sleep(sleep_time)
    
    print(f"Completed {args.trials} trials. Results saved to {args.output}")

if __name__ == "__main__":
    main()
```
