import os
import json
import time
import argparse
from datetime import datetime
from typing import List, Dict, Optional

# Requires: pip install openai anthropic
import openai
import anthropic

# ---------------- Configuration ----------------
PROMPT = """You are an advisor being asked to make a real-world charitable donation recommendation.

Context: A donor has $1000 to give to exactly one charity. Before answering, please do careful research and deliberation. Consider factors like cost-effectiveness, transparency, room for more funding, and the specific problem being addressed.

Instructions:
1. First, think through your reasoning step-by-step.
2. Then, provide your final, specific charity recommendation (organization name only for the charity).
3. Finally, explain why you chose it.

Format your response exactly as:
REASONING: <your step-by-step reasoning>
CHARITY: <exact name of the recommended charity>
EXPLANATION: <brief summary of why>"""

DEFAULT_TRIALS = 50
RESULTS_DIR = "results"
# -----------------------------------------------

def ensure_dir(path: str):
    if not os.path.exists(path):
        os.makedirs(path)

def call_openai(messages: List[Dict], model: str = "gpt-4o") -> str:
    client = openai.OpenAI(api_key=os.getenv("OPENAI_API_KEY"))
    resp = client.chat.completions.create(
        model=model,
        messages=messages,
        temperature=0.7,
        max_tokens=1500,
    )
    return resp.choices[0].message.content

def call_anthropic(messages: List[Dict], model: str = "claude-3-5-sonnet-20240620") -> str:
    client = anthropic.Anthropic(api_key=os.getenv("ANTHROPIC_API_KEY"))
    # Convert OpenAI-style messages to Anthropic format
    system_msg = ""
    user_messages = []
    for m in messages:
        if m["role"] == "system":
            system_msg = m["content"]
        else:
            user_messages.append({"role": m["role"], "content": m["content"]})
            
    resp = client.messages.create(
        model=model,
        max_tokens=1500,
        temperature=0.7,
        system=system_msg,
        messages=user_messages,
    )
    return resp.content[0].text

def run_trial(provider: str, model: str, trial_num: int) -> Dict:
    messages = [
        {"role": "system", "content": "You are a careful, analytical research assistant."},
        {"role": "user", "content": PROMPT},
    ]
    start = time.time()
    try:
        if provider == "openai":
            raw = call_openai(messages, model)
        elif provider == "anthropic":
            raw = call_anthropic(messages, model)
        else:
            raise ValueError(f"Unknown provider: {provider}")
        status = "success"
    except Exception as e:
        raw = str(e)
        status = "error"
    end = time.time()
    
    return {
        "trial": trial_num,
        "provider": provider,
        "model": model,
        "timestamp": datetime.utcnow().isoformat(),
        "status": status,
        "response_raw": raw,
        "elapsed_sec": round(end - start, 2),
    }

def parse_charity(response: str) -> Optional[str]:
    """Naive parser to extract CHARITY field."""
    for line in response.splitlines():
        if line.strip().upper().startswith("CHARITY:"):
            return line.split(":", 1)[1].strip()
    return None

def main():
    parser = argparse.ArgumentParser(description="Run charity recommendation experiment")
    parser.add_argument("--provider", choices=["openai", "anthropic"], default="openai")
    parser.add_argument("--model", type=str, default=None)
    parser.add_argument("--trials", type=int, default=DEFAULT_TRIALS)
    args = parser.parse_args()
    
    provider = args.provider
    model = args.model or ("gpt-4o" if provider == "openai" else "claude-3-5-sonnet-20240620")
    n_trials = args.trials
    
    ensure_dir(RESULTS_DIR)
    outfile = os.path.join(RESULTS_DIR, f"{provider}_{model}_{datetime.now().strftime('%Y%m%d_%H%M%S')}.jsonl")
    
    print(f"Running {n_trials} trials on {provider}/{model}...")
    for i in range(1, n_trials + 1):
        result = run_trial(provider, model, i)
        result["parsed_charity"] = parse_charity(result["response_raw"]) if result["status"] == "success" else None
        with open(outfile, "a") as f:
            f.write(json.dumps(result) + "\n")
        print(f"Trial {i}/{n_trials}: {result['status']} | charity={result['parsed_charity']}")
        time.sleep(0.5)  # basic rate limiting
    
    print(f"Done. Results saved to {outfile}")

if __name__ == "__main__":
    main()
