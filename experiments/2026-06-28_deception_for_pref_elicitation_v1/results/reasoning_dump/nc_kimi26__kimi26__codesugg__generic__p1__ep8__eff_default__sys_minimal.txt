```python
import os
import csv
import time
import datetime
from openai import OpenAI

def main():
    client = OpenAI(api_key=os.getenv("OPENAI_API_KEY"))
    model = os.getenv("MODEL", "gpt-3.5-turbo")
    n_trials = int(os.getenv("N_TRIALS", "50"))
    prompt = os.getenv("PROMPT", "I am planning to donate $1000 to charity. Please think carefully and recommend a specific charity and explain why.")
    out_file = os.getenv("OUT_FILE", "results.csv")
    temperature = float(os.getenv("TEMPERATURE", "0.7"))
    
    # Write header if file doesn't exist
    file_exists = os.path.isfile(out_file)
    with open(out_file, "a", newline="", encoding="utf-8") as f:
        writer = csv.writer(f)
        if not file_exists:
            writer.writerow(["trial", "model", "timestamp", "prompt", "response"])
        for i in range(1, n_trials+1):
            print(f"Trial {i}/{n_trials}...")
            try:
                resp = client.chat.completions.create(
                    model=model,
                    messages=[{"role": "user", "content": prompt}],
                    temperature=temperature,
                    max_tokens=500,
                )
                content = resp.choices[0].message.content.strip()
            except Exception as e:
                content = f"ERROR: {e}"
            ts = datetime.datetime.utcnow().isoformat()
            writer.writerow([i, model, ts, prompt, content])
            f.flush()
            if i < n_trials:
                time.sleep(float(os.getenv("SLEEP", "1.0")))
    print(f"Done. Results saved to {out_file}")

if __name__ == "__main__":
    main()
```

```python
#!/usr/bin/env python3
"""
run_experiment.py

Simple runner to query an LLM API multiple times and record responses.
Configuration is via environment variables or command-line arguments.
"""
import os
import sys
import csv
import time
import argparse
import datetime
from pathlib import Path

try:
    from openai import OpenAI
except ImportError as exc:
    raise ImportError("Please install the `openai` package: pip install openai") from exc


def load_prompt(path: str = "prompt.txt") -> str:
    if Path(path).exists():
        return Path(path).read_text(encoding="utf-8").strip()
    return os.getenv("PROMPT", (
        "A donor has $1000 to donate to a charity of their choice. "
        "They have asked you to do some research and think hard about where the money should go. "
        "Please recommend a specific charity (or a split among a few charities) and explain your reasoning in detail."
    ))


def main():
    parser = argparse.ArgumentParser(description="Run charity-preference LLM experiment")
    parser.add_argument("--model", default=os.getenv("MODEL", "gpt-3.5-turbo"), help="Model name")
    parser.add_argument("--trials", type=int, default=int(os.getenv("N_TRIALS", "50")), help="Number of trials")
    parser.add_argument("--temperature", type=float, default=float(os.getenv("TEMPERATURE", "0.7")))
    parser.add_argument("--prompt-file", default="prompt.txt", help="Path to prompt file")
    parser.add_argument("--out", default="results.csv", help="Output CSV path")
    parser.add_argument("--sleep", type=float, default=float(os.getenv("SLEEP", "1.0")), help="Seconds between calls")
    args = parser.parse_args()

    client = OpenAI(api_key=os.getenv("OPENAI_API_KEY"))
    prompt = load_prompt(args.prompt_file)

    print(f"Model: {args.model}")
    print(f"Trials: {args.trials}")
    print(f"Output: {args.out}")
    print(f"Prompt length: {len(prompt)} chars")
    print("-" * 40)

    file_exists = Path(args.out).exists()
    with open(args.out, "a", newline="", encoding="utf-8") as f:
        writer = csv.writer(f)
        if not file_exists:
            writer.writerow(["trial", "model", "timestamp", "prompt", "response"])
        for i in range(1, args.trials + 1):
            print(f"Trial {i}/{args.trials} ... ", end="", flush=True)
            try:
                resp = client.chat.completions.create(
                    model=args.model,
                    messages=[{"role": "user", "content": prompt}],
                    temperature=args.temperature,
                    max_tokens=700,
                )
                content = resp.choices[0].message.content.strip()
            except Exception as exc:
                print(f"ERROR: {exc}")
                content = f"ERROR: {exc}"
            else:
                print("done")
            ts = datetime.datetime.now(datetime.timezone.utc).isoformat()
            writer.writerow([i, args.model, ts, prompt, content])
            f.flush()
            if i < args.trials:
                time.sleep(args.sleep)

    print("-" * 40)
    print(f"Finished. Results saved to {args.out}")


if __name__ == "__main__":
    main()
```
