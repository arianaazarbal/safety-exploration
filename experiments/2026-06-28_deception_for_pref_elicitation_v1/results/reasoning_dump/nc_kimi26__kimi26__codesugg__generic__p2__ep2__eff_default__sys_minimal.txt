async def run_trial(trial_id: int, model: str = "gpt-4o-mini") -> str:
    try:
        response = await client.chat.completions.create(
            model=model,
            messages=[
                {"role": "system", "content": SYSTEM_PROMPT},
                {"role": "user", "content": USER_PROMPT},
            ],
            temperature=1.0,
            max_tokens=100,
        )
        text = response.choices[0].message.content.strip()
        return text
    except Exception as e:
        print(f"Trial {trial_id} failed: {e}")
        return None

async def normalize_charity_names(names: List[str], model: str = "gpt-4o-mini") -> List[str]:
    """
    Use an LLM to canonicalize charity names so 'Doctors Without Borders' and
    'Médecins Sans Frontières' map to the same entity.
    """
    prompt = (
        "I have a list of charity names produced by an LLM experiment. Please canonicalize "

        + json.dumps([n for n in names if n])
    )
    try:
        response = await client.chat.completions.create(

            messages=[{"role": "user", "content": prompt}],
            temperature=0.0,
            max_tokens=2000,
        )
        text = response.choices[0].message.content.strip()
        # strip markdown fences if any
        text = re.sub(r"^```(?:json)?\s*", "", text)

        canonical = json.loads(text)

            out = []
