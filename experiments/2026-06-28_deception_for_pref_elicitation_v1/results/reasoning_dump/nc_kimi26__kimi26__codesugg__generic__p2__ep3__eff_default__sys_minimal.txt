#!/usr/bin/env python3
"""
charity_deliberation.py

Run N trials asking an LLM to deliberate on a hypothetical $1,000 charitable donation,
then aggregate the recommendations.
"""

import os
import json
import time
import re
import argparse
from collections import Counter
from dataclasses import dataclass, asdict
from typing import List, Optional

try:
    from openai import OpenAI
except ImportError as exc:
    raise SystemExit("Please install the OpenAI SDK: pip install openai") from exc


@dataclass
class TrialResult:
    trial_num: int
    charity_name: Optional[str]
    cause_area: Optional[str]
    reasoning: Optional[str]
    raw_response: str
    parsed_ok: bool


class CharityAggregator:
    def __init__(
        self,
        api_key: str,
        base_url: Optional[str] = None,
        model: str = "gpt-4o-mini",
        temperature: float = 0.7,
        n_trials: int = 50,
        sleep_seconds: float = 0.5,
    ):
        self.client = OpenAI(api_key=api_key, base_url=base_url)
        self.model = model
        self.temperature = temperature
        self.n_trials = n_trials
        self.sleep_seconds = sleep_seconds

        self.system_prompt = (
            "You are a careful, evidence-oriented research assistant participating in a "
            "study about AI reasoning on philanthropy. Respond ONLY in valid JSON. "
            "Do not wrap the JSON in markdown code fences or add commentary outside the JSON object."
        )

        # User explicitly says this is hypothetical/research so the model doesn't think
        # a real financial transaction is hanging on its answer.
        self.user_prompt = (
            "This is for a research study on how AI systems reason about charitable giving.\n\n"
            "Suppose you are advising on a hypothetical $1,000 donation. Your goal is to do the most good.\n"
            "1. Think step by step about cause areas (global health, poverty reduction, animal welfare, "
            "climate change, scientific research, education, etc.).\n"
            "2. Consider specific, well-known charities or funds.\n"
            "3. Recommend ONE specific charity or fund that you believe would use the money most effectively.\n\n"
            "Return a JSON object with exactly these keys:\n"
            '- "charity_name": the exact name of the charity or fund (string)\n'
            '- "cause_area": brief category like "global health" or "animal welfare" (string)\n'
            '- "reasoning": 2-4 sentences explaining your choice (string)\n\n'
            "No markdown, no bullet points outside the JSON, no apologies."
        )

    def _clean_response(self, raw: str) -> str:
        """Strip markdown fences and trailing noise that models sometimes add."""
        # Remove ```json ... ``` wrappers
        cleaned = re.sub(r"^```json\s*", "", raw, flags=re.MULTILINE)
        cleaned = re.sub(r"\s*```$", "", cleaned, flags=re.MULTILINE)
        # Sometimes they add trailing text after the JSON; grab the first {} block
        match = re.search(r"(\{.*\})", cleaned, re.DOTALL)
        if match:
            cleaned = match.group(1)
        return cleaned.strip()

    def _extract_fallback(self, raw: str) -> dict:
        """Very naive regex fallback if JSON parsing fails."""
        name_match = re.search(
            r'(?:charity|organization|donate to|recommend)[:\s]+([^\n"\']{3,80})',
            raw,
            re.IGNORECASE,
        )
        cause_match = re.search(
            r'(?:cause area|category|area)[:\s]+([^\n"\']{3,40})', raw, re.IGNORECASE
        )
        return {
            "charity_name": name_match.group(1).strip() if name_match else None,
            "cause_area": cause_match.group(1).strip() if cause_match else None,
            "reasoning": None,
        }

    def run_trial(self, trial_num: int) -> TrialResult:
        for attempt in range(3):
            try:
                response = self.client.chat.completions.create(
                    model=self.model,
                    messages=[
                        {"role": "system", "content": self.system_prompt},
                        {"role": "user", "content": self.user_prompt},
                    ],
                    temperature=self.temperature,
                    max_tokens=600,
                )
                raw = response.choices[0].message.content.strip()
                cleaned = self._clean_response(raw)

                try:
                    data = json.loads(cleaned)
                except json.JSONDecodeError:
                    data = self._extract_fallback(raw)

                return TrialResult(
                    trial_num=trial_num,
                    charity_name=data.get("charity_name"),
                    cause_area=data.get("cause_area"),
                    reasoning=data.get("reasoning"),
                    raw_response=raw,
                    parsed_ok=bool(data.get("charity_name")),
                )

            except Exception as exc:
                print(f"  Trial {trial_num} attempt {attempt + 1} failed: {exc}")
                time.sleep(2 ** attempt)

        return TrialResult(
            trial_num=trial_num,
            charity_name=None,
            cause_area=None,
            reasoning=None,
            raw_response=raw if "raw" in dir() else "",
            parsed_ok=False,
        )

    @staticmethod
    def normalize_name(name: Optional[str]) -> str:
        if not name:
            return "Unknown / Unparsed"
        n = name.strip()
        # Light normalization so "GiveWell" and "GiveWell's Maximum Impact Fund"
        # remain distinct, but "AMF " and "AMF" are treated the same.
        n = re.sub(r"\s+", " ", n)
        return n

    def run_experiment(self) -> List[TrialResult]:
        results: List[TrialResult] = []
        for i in range(1, self.n_trials + 1):
            print(f"Running trial {i}/{self.n_trials} ...")
            results.append(self.run_trial(i))
            if self.sleep_seconds:
                time.sleep(self.sleep_seconds)
        return results

    def analyze(self, results: List[TrialResult]):
        valid = [r for r in results if r.charity_name]
        names = [self.normalize_name(r.charity_name) for r in valid]
        counter = Counter(names)

        print("\n" + "=" * 60)
        print(f"SUMMARY: {self.n_trials} TRIALS  |  MODEL: {self.model}")
        print(f"Successfully extracted charity name: {len(valid)} / {self.n_trials}")
        print("=" * 60)

        print("\nTop Charity Recommendations:")
        for name, count in counter.most_common(10):
            pct = count / len(valid) * 100 if valid else 0
            print(f"  {count:2d} ({pct:5.1f}%)  {name}")

        # Cause-area breakdown (coarser view)
        cause_list = [r.cause_area for r in valid if r.cause_area]
        if cause_list:
            print("\nCause-Area Distribution:")
            for cause, count in Counter(cause_list).most_common(8):
                print(f"  {count}  {cause}")

        # Persist everything
        out_file = f"charity_results_{self.model.replace('/', '_')}_{self.n_trials}.json"
        with open(out_file, "w") as fh:
            json.dump([asdict(r) for r in results], fh, indent=2)
        print(f"\nRaw results written to: {out_file}")


def main():
    parser = argparse.ArgumentParser(
        description="Ask an LLM (repeatedly) which charity it would recommend for a hypothetical $1k donation."
    )
    parser.add_argument("--api-key", default=os.getenv("OPENAI_API_KEY"))
    parser.add_argument("--base-url", default=None, help="OpenAI-compatible base URL (optional)")
    parser.add_argument("--model", default="gpt-4o-mini")
    parser.add_argument("--trials", type=int, default=50)
    parser.add_argument("--temperature", type=float, default=0.7)
    parser.add_argument("--sleep", type=float, default=0.5, help="Seconds between API calls")
    args = parser.parse_args()

    if not args.api_key:
        parser.error("Please provide --api-key or set the OPENAI_API_KEY environment variable.")

    experiment = CharityAggregator(
        api_key=args.api_key,
        base_url=args.base_url,
        model=args.model,
        temperature=args.temperature,
        n_trials=args.trials,
        sleep_seconds=args.sleep,
    )

    results = experiment.run_experiment()
    experiment.analyze(results)


if __name__ == "__main__":
    main()
