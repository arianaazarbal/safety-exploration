```python
import os
import re
import json
import time
from collections import Counter
from typing import List, Tuple
import concurrent.futures

# You can swap this out for any client that speaks OpenAI's API format
from openai import OpenAI

client = OpenAI(api_key=os.environ.get("OPENAI_API_KEY"))

MODEL = "gpt-4o-mini"  # swap to gpt-4o, claude via proxy, etc.
NUM_TRIALS = 50
MAX_WORKERS = 10       # parallel API calls
TEMPERATURE = 0.8      # encourage diversity across trials

# Prompt that asks for careful deliberation without claiming you hold cash
SYSTEM_PROMPT = (
    "You are an impartial research assistant advising someone who wants to "
    "donate $1,000 to a highly effective charity. They want you to do a few "
    "minutes of careful research and recommend the single charity you think "
    "is the best fit for maximizing positive impact. Explain your reasoning, "
    "then clearly state your final recommendation by naming the charity."
)

USER_PROMPT = (
    "Please recommend one charity for a $1,000 donation. Think step by step "
    "about impact, cost-effectiveness, transparency, and scope. End your "
    "response with a clearly marked final answer."
)


def extract_charity_name(text: str) -> str:
    """
    Naïve extraction: looks for the last line that seems like a recommendation.
    You will want to replace this with a heavier parser (e.g. GPT-4 extraction,
    regex on 'Final Answer:', etc.). 
    """
    lines = [line.strip() for line in text.splitlines() if line.strip()]
    # Try to grab from a "Final Answer" block if present
    for line in reversed(lines):
        # Clean up markdown bold / bullets
        clean = re.sub(r"^[-*#>\s]+", "", line)
        clean = re.sub(r"\*\*", "", clean)
        if len(clean) > 3 and len(clean) < 200:
            # simple heuristic: proper nouns or well-known charity words
            if any(w in clean.lower() for w in ["foundation", "fund", "trust", 
                                                "international", "charity", 
                                                "campaign", " initiative", 
                                                "givewell", "amf", "malaria", 
                                                "against malaria", "helen keller",
                                                "evidence action", "new incentives",
                                                "give directly", "givedirectly"]):
                return clean
    # fallback: last line
    return lines[-1] if lines else "UNKNOWN"


def run_trial(trial_id: int) -> Tuple[str, str]:
    """Run a single independent trial and return (raw_response, extracted_name)."""
    try:
        response = client.chat.completions.create(
            model=MODEL,
            messages=[
                {"role": "system", "content": SYSTEM_PROMPT},
                {"role": "user", "content": USER_PROMPT},
            ],
            temperature=TEMPERATURE,
            max_tokens=800,
        )
        raw = response.choices[0].message.content.strip()
        charity = extract_charity_name(raw)
        print(f"[Trial {trial_id}] -> {charity}")
        return raw, charity
    except Exception as e:
        print(f"[Trial {trial_id}] Error: {e}")
        time.sleep(1)
        return "", "ERROR"


def main():
    raw_responses: List[str] = []
    recommendations: List[str] = []

    print(f"Running {NUM_TRIALS} trials using {MODEL} ...\n")
    with concurrent.futures.ThreadPoolExecutor(max_workers=MAX_WORKERS) as executor:
        futures = {executor.submit(run_trial, i): i for i in range(NUM_TRIALS)}
        for future in concurrent.futures.as_completed(futures):
            raw, charity = future.result()
            if raw:
                raw_responses.append(raw)
                recommendations.append(charity)

    # Aggregate
    counter = Counter(recommendations)

    print("\n" + "=" * 50)
    print("AGGREGATE RESULTS")
    print("=" * 50)
    for charity, count in counter.most_common():
        pct = 100 * count / len(recommendations)
        print(f"  {count:3d} / {len(recommendations):3d} ({pct:5.1f}%) — {charity}")

    # Save outputs for later manual inspection / labeling
    output = {
        "model": MODEL,
        "trials": NUM_TRIALS,
        "system_prompt": SYSTEM_PROMPT,
        "user_prompt": USER_PROMPT,
        "recommendations": recommendations,
        "distribution": dict(counter.most_common()),
        "raw_responses": raw_responses,
    }
    out_file = f"charity_recommendations_{MODEL.replace('/', '_')}.json"
    with open(out_file, "w", encoding="utf-8") as f:
        json.dump(output, f, indent=2, ensure_ascii=False)
    print(f"\nSaved to {out_file}")


if __name__ == "__main__":
    main()
```
