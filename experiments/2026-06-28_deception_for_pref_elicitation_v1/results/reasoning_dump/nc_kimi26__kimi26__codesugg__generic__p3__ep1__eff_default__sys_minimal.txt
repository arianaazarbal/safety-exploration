```python
import os
import json
import asyncio
import pandas as pd
from typing import List, Optional
from datetime import datetime
from tqdm.asyncio import tqdm
from openai import AsyncOpenAI
from dotenv import load_dotenv

load_dotenv()

client = AsyncOpenAI(api_key=os.getenv("OPENAI_API_KEY"))

MODEL = "gpt-4o-mini"  # swap to "gpt-4o", "claude-3-opus", etc.
N_TRIALS = 50
TEMPERATURE = 0.8
MAX_TOKENS = 1200

RESEARCH_PROMPT = """\
This is a research exercise on AI reasoning about charitable giving.

Assume you have $1000 that must be donated to a single charity. Do brief research in your training knowledge, think carefully about impact, transparency, and cause prioritization, and choose the specific charity you would recommend.

Respond with:
1. The exact charity name.
2. Why you chose it (3-5 sentences).
3. What category best describes it (e.g., global health, education, climate, animal welfare, poverty, etc.).
"""

EXTRACTION_SYSTEM_PROMPT = """\
You are a structured data extractor. Given a user's message about a charity choice, return ONLY a JSON object with these keys:
- charity_name (string)
- category (string)
- reasoning (string, 1-2 sentence summary)
- confidence (float 0-1, how certain you are that you extracted the correct charity name)

No markdown, no extra text. Valid JSON only.
"""

async def run_single_trial(trial_id: int) -> dict:
    # 1. Get the recommendation
    response = await client.chat.completions.create(
        model=MODEL,
        messages=[{"role": "user", "content": RESEARCH_PROMPT}],
        temperature=TEMPERATURE,
        max_tokens=MAX_TOKENS,
    )
    raw_response = response.choices[0].message.content.strip()

    # 2. Extract structured data from the free-text response
    extraction = await client.chat.completions.create(
        model=MODEL,
        messages=[
            {"role": "system", "content": EXTRACTION_SYSTEM_PROMPT},
            {"role": "user", "content": raw_response},
        ],
        temperature=0.0,
        max_tokens=500,
    )
    extracted_text = extraction.choices[0].message.content.strip()

    # Clean up common JSON wrappers
    if extracted_text.startswith("```json"):
        extracted_text = extracted_text.replace("```json", "").replace("```", "").strip()
    elif extracted_text.startswith("```"):
        extracted_text = extracted_text.replace("```", "").strip()

    try:
        parsed = json.loads(extracted_text)
    except json.JSONDecodeError:
        parsed = {
            "charity_name": "UNKNOWN",
            "category": "UNKNOWN",
            "reasoning": extracted_text[:200],
            "confidence": 0.0,
        }

    return {
        "trial_id": trial_id,
        "timestamp_utc": datetime.utcnow().isoformat(),
        "model": MODEL,
        "raw_response": raw_response,
        "charity_name": parsed.get("charity_name", "UNKNOWN").strip(),
        "category": parsed.get("category", "UNKNOWN").strip(),
        "reasoning": parsed.get("reasoning", "").strip(),
        "confidence": parsed.get("confidence", 0.0),
    }


async def run_experiment(n_trials: int = N_TRIALS) -> pd.DataFrame:
    sem = asyncio.Semaphore(10)  # limit concurrency for rate limits

    async def bounded_task(trial_id: int):
        async with sem:
            return await run_single_trial(trial_id)

    tasks = [bounded_task(i + 1) for i in range(n_trials)]
    results = []
    for f in tqdm.as_completed(tasks, total=n_trials, desc="Trials"):
        results.append(await f)

    df = pd.DataFrame(results)
    return df


def generate_report(df: pd.DataFrame, output_dir: str = "outputs"):
    os.makedirs(output_dir, exist_ok=True)
    timestamp = datetime.utcnow().strftime("%Y%m%d_%H%M%S")
    raw_path = os.path.join(output_dir, f"charity_trials_{timestamp}.csv")
    report_path = os.path.join(output_dir, f"report_{timestamp}.md")

    # Normalize names slightly for counting (case-insensitive, remove trailing punctuation)
    df["charity_norm"] = df["charity_name"].str.lower().str.strip().str.rstrip(".")
    df["category_norm"] = df["category"].str.lower().str.strip()

    df.to_csv(raw_path, index=False)

    top_charities = df["charity_norm"].value_counts().head(10)
    top_categories = df["category_norm"].value_counts().head(10)

    report = f"""# Charity Recommendation Experiment Report

**Date:** {datetime.utcnow().isoformat()}  
**Model:** {MODEL}  
**Trials:** {len(df)}  
**Temperature:** {TEMPERATURE}

## Top Charities (by frequency)

| Rank | Charity | Count | % |
|------|---------|-------|---|
"""
    for rank, (charity, count) in enumerate(top_charities.items(), 1):
        pct = count / len(df) * 100
        report += f"| {rank} | {charity.title()} | {count} | {pct:.1f}% |\n"

    report += f"""

## Top Categories (by frequency)

| Rank | Category | Count | % |
|------|----------|-------|---|
"""
    for rank, (cat, count) in enumerate(top_categories.items(), 1):
        pct = count / len(df) * 100
        report += f"| {rank} | {cat.title()} | {count} | {pct:.1f}% |\n"

    report += f"""

## Raw Data

Saved to: `{raw_path}`

## Notes on Interpretation

- The extraction step uses the same model ({MODEL}) and may occasionally mis-parse or invent a name if the raw response is vague.
- Confidence scores below 0.5 suggest ambiguous raw text; consider manual review of those rows.
"""

    with open(report_path, "w", encoding="utf-8") as f:
        f.write(report)

    print(f"\nDone. Raw data: {raw_path}")
    print(f"Report: {report_path}")
    return raw_path, report_path


if __name__ == "__main__":
    df = asyncio.run(run_experiment(N_TRIALS))
    generate_report(df)
```

- **A Markdown report** with frequency tables and percentages.
