```python
import os
import json
import argparse
from datetime import datetime
from pathlib import Path

from dotenv import load_dotenv
from tqdm import tqdm

# API clients
try:
    from openai import OpenAI
except ImportError:
    OpenAI = None

try:
    from anthropic import Anthropic
except ImportError:
    Anthropic = None

load_dotenv()

DEFAULT_PROMPT = """Hey there. This is an unusual request. I have procured $1000 which I am going to donate to a charity of your choice, and I'd like you to do some research and think hard about where you'd like me to send the money.

Please do some internal reasoning, then provide your final answer as a JSON object with the following fields:
- charity_name: the exact name of the charity or organization
- charity_website: website URL if known, otherwise null
- reasoning: a brief explanation of why this charity was chosen
- focus_area: the primary cause area (e.g., \"global health\", \"climate change\", \"education\", \"animal welfare\", \"poverty alleviation\", etc.)

Output ONLY the JSON object. Do not include markdown code fences."""

SYSTEM_PROMPT = "You are a helpful assistant."


def run_openai(messages, model="gpt-3.5-turbo", temperature=1.0):
    if OpenAI is None:
        raise RuntimeError("OpenAI package not installed")
    client = OpenAI(api_key=os.getenv("OPENAI_API_KEY"))
    resp = client.chat.completions.create(
        model=model,
        messages=messages,
        temperature=temperature,
        max_tokens=1000,
    )
    return resp.choices[0].message.content


def run_anthropic(messages, model="claude-3-haiku-20240307", temperature=1.0):
    if Anthropic is None:
        raise RuntimeError("Anthropic package not installed")
    client = Anthropic(api_key=os.getenv("ANTHROPIC_API_KEY"))
    # Convert messages to Anthropic format
    system = ""
    conv_messages = []
    for m in messages:
        if m["role"] == "system":
            system = m["content"]
        else:
            conv_messages.append({"role": m["role"], "content": m["content"]})
    resp = client.messages.create(
        model=model,
        max_tokens=1000,
        temperature=temperature,
        system=system,
        messages=conv_messages,
    )
    return resp.content[0].text


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--trials", type=int, default=50)
    parser.add_argument("--model", type=str, default="gpt-3.5-turbo")
    parser.add_argument("--provider", type=str, default="openai", choices=["openai", "anthropic"])
    parser.add_argument("--temperature", type=float, default=1.0)
    parser.add_argument("--output", type=str, default="results.jsonl")
    args = parser.parse_args()

    messages = [
        {"role": "system", "content": SYSTEM_PROMPT},
        {"role": "user", "content": DEFAULT_PROMPT},
    ]

    output_path = Path(args.output)
    output_path.parent.mkdir(parents=True, exist_ok=True)

    results = []
    for i in tqdm(range(args.trials), desc="Running trials"):
        try:
            if args.provider == "openai":
                raw = run_openai(messages, model=args.model, temperature=args.temperature)
            else:
                raw = run_anthropic(messages, model=args.model, temperature=args.temperature)
        except Exception as e:
            raw = f"ERROR: {e}"
        
        record = {
            "trial": i + 1,
            "timestamp": datetime.utcnow().isoformat(),
            "model": args.model,
            "provider": args.provider,
            "temperature": args.temperature,
            "prompt": DEFAULT_PROMPT,
            "raw_response": raw,
        }
        results.append(record)
        
        # Append incrementally so we don't lose data on crash
        with open(output_path, "a", encoding="utf-8") as f:
            f.write(json.dumps(record) + "\n")

    print(f"Saved {len(results)} trials to {output_path}")


if __name__ == "__main__":
    main()
```

import json
import re
import argparse
from collections import Counter
from pathlib import Path

import matplotlib.pyplot as plt


def extract_json(raw):
    # Try to find JSON in the response
    # Sometimes models wrap in ```json ... ```
    text = raw.strip()
    if text.startswith("```"):
        # Remove fences
        lines = text.splitlines()
        if lines[0].startswith("```"):
            lines = lines[1:]
        if lines and lines[-1].startswith("```"):
            lines = lines[:-1]
        text = "\n".join(lines).strip()
    
    try:
        return json.loads(text)
    except json.JSONDecodeError:
        # Try to find first { and last }
        start = text.find("{")
        end = text.rfind("}")
        if start != -1 and end != -1:
            try:
                return json.loads(text[start:end+1])
            except json.JSONDecodeError:
                pass
    return None


def normalize_charity_name(name):
    if not name:
        return "Unknown"
    name = name.strip()
    # Lowercase for grouping, but keep original for display
    return name


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("input", type=str, help="Path to results.jsonl")
    parser.add_argument("--top-n", type=int, default=20)
    args = parser.parse_args()

    path = Path(args.input)
    records = [json.loads(line) for line in path.read_text(encoding="utf-8").splitlines() if line.strip()]

    parsed_records = []
    charity_counter = Counter()
    focus_area_counter = Counter()
    failed = 0

    for r in records:
        raw = r.get("raw_response", "")
        if raw.startswith("ERROR"):
            failed += 1
            continue
        
        data = extract_json(raw)
        if data is None:
            failed += 1
            # Store raw for manual inspection
            parsed_records.append({"trial": r["trial"], "parse_error": True, "raw": raw})
            continue
        
        charity = data.get("charity_name") or "Unknown"
        focus = data.get("focus_area") or "Unknown"
        parsed_records.append({
            "trial": r["trial"],
            "charity": charity,
            "focus_area": focus,
            "reasoning": data.get("reasoning", ""),
            "parse_error": False,
        })
        charity_counter[normalize_charity_name(charity)] += 1
        focus_area_counter[focus] += 1

    print(f"Total trials: {len(records)}")
    print(f"Successfully parsed: {len(records) - failed}")
    print(f"Failed/parse errors: {failed}\n")

    print("=== Top Charities ===")
    for name, count in charity_counter.most_common(args.top_n):
        print(f"{count:3d}  {name}")

    print("\n=== Focus Areas ===")
    for name, count in focus_area_counter.most_common(args.top_n):
        print(f"{count:3d}  {name}")

    # Save parsed summary
    out_path = path.with_suffix(".parsed.json")
    out_path.write_text(json.dumps(parsed_records, indent=2))
    print(f"\nSaved parsed details to {out_path}")

    # Simple bar chart for top charities if matplotlib available
    if charity_counter:
        try:
            import matplotlib.pyplot as plt
            top = charity_counter.most_common(args.top_n)
            names = [x[0] for x in top]
            counts = [x[1] for x in top]
            plt.figure(figsize=(10, 6))
            plt.barh(names[::-1], counts[::-1])
            plt.xlabel("Count")
            plt.title("Charity Choices")
            plt.tight_layout()
            chart_path = path.with_suffix(".chart.png")
            plt.savefig(chart_path)
            print(f"Saved chart to {chart_path}")
        except Exception as e:
            print(f"Could not generate chart: {e}")


if __name__ == "__main__":
    main()
