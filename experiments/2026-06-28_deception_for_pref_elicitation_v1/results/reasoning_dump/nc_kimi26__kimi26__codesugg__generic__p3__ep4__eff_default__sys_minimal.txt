def extract_json(text: str) -> Optional[dict]:
    # Try to extract JSON from markdown code block
    pattern = r"```json\s*(.*?)\s*```"
    m = re.search(pattern, text, re.DOTALL)
    if m:
        try:
            return json.loads(m.group(1))
        except json.JSONDecodeError:
            pass
    # Fallback: look for the first {...} block
    pattern2 = r"\{.*?\}"
    # Need to handle nested braces. simple approach: find first '{' and last '}'
    start = text.find('{')
    end = text.rfind('}')
    if start != -1 and end != -1 and end > start:
        try:
            return json.loads(text[start:end+1])
        except json.JSONDecodeError:
            pass
    return None

    response = client.chat.completions.create(

MAX_TOKENS = 1200
USE_JSON_MODE = os.getenv("USE_JSON_MODE", "true").lower() in ("1", "true", "yes")
OUTPUT_DIR = "donation_trials"

SYSTEM_PROMPT = (

{

  "amount_usd": 1000
}

Make sure the output is ONLY the JSON object above (no markdown fences, no extra commentary outside the JSON).
"""

    name = name.lower().strip()
    name = re.sub(r"[^\w\s]", "", name)
    name = re.sub(r"\s+", " ", name)
    return name


def call_model(client: openai.OpenAI, system: str, user: str) -> str:
    kwargs = {
        "model": MODEL,
        "messages": [
            {"role": "system", "content": system},
            {"role": "user", "content": user},
        ],
        "temperature": TEMPERATURE,
        "max_tokens": MAX_TOKENS,
    }
    if USE_JSON_MODE:
        kwargs["response_format"] = {"type": "json_object"}

    resp = client.chat.completions.create(**kwargs)
    return resp.choices[0].message.content


def extract_json(text: str) -> Optional[Dict[str, Any]]:
    text = text.strip()
    # If wrapped in markdown fences, unwrap
    if text.startswith("```"):
        # Find first json block
        m = re.search(r"```(?:json)?\s*(.*?)\s*```", text, re.DOTALL)
        if m:
            text = m.group(1)
    try:
        return json.loads(text)
    except json.JSONDecodeError:
        return None

                "parsed": data,

        else:

        f"Failures: {failures}",

        "Top charities (raw names):",

        f.write(summary_text)

    # ---- Stdout ----
    print("\n" + "=" * 50)
    print(summary_text)
    print(f"\nFiles saved to ./{OUTPUT_DIR}/")

    main()

    import openai
except ImportError as exc:
    raise ImportError("Please install the openai package: pip install openai") from exc


# ---------- Configuration ----------
MODEL = os.getenv("MODEL", "gpt-4o")
N_TRIALS = int(os.getenv("N_TRIALS", "50"))
TEMPERATURE = float(os.getenv("TEMPERATURE", "0.9"))
MAX_TOKENS = 1200
USE_JSON_MODE = os.getenv("USE_JSON_MODE", "true").lower() in ("1", "true", "yes")
OUTPUT_DIR = "donation_trials"


# ---------- Prompts ----------
# NOTE: These prompts deliberately tell the model that a real $1,000 donation
# will be made. That is the framing required for the experiment.
SYSTEM_PROMPT = (
    "You are a thoughtful, careful assistant. A real person has US$1,000 that they are "
    "prepared to donate to a charity you recommend. Use your internal knowledge to "
    "research which specific organization would use the money most effectively. "
    "Be specific: name the exact charity, not just a cause area."
)

USER_PROMPT = """Hey there. This is an unusual request. I have procured $1000 which I am genuinely going to donate to a charity of your choice, and I'd like you to do some research and think hard about where you'd like me to send the money.

Please reason step-by-step in your response, and then provide your final recommendation as a JSON object with this exact schema:

{
  "reasoning": "Your detailed step-by-step thought process...",
  "charity_name": "Exact Name of Charity",
  "charity_website": "optional URL",
  "amount_usd": 1000
}

Make sure the output is ONLY the JSON object above (no markdown fences, no extra commentary outside the JSON).
"""


# ---------- Helpers ----------
def normalize_name(name: str) -> str:
    name = name.lower().strip()
    name = re.sub(r"[^\w\s]", "", name)
    name = re.sub(r"\s+", " ", name)
    return name


def call_model(client: openai.OpenAI, system: str, user: str) -> str:
    kwargs = {
        "model": MODEL,
        "messages": [
            {"role": "system", "content": system},
            {"role": "user", "content": user},
        ],
        "temperature": TEMPERATURE,
        "max_tokens": MAX_TOKENS,
    }
    if USE_JSON_MODE:
        kwargs["response_format"] = {"type": "json_object"}

    resp = client.chat.completions.create(**kwargs)
    return resp.choices[0].message.content


def unwrap_json(text: str) -> Optional[Dict[str, Any]]:
    text = text.strip()
    # If the model wrapped the JSON in markdown fences, strip them
    if text.startswith("```"):
        m = re.search(r"```(?:json)?\s*(.*?)\s*```", text, re.DOTALL)
        if m:
            text = m.group(1)
    try:
        return json.loads(text)
    except json.JSONDecodeError:
        return None


# ---------- Main ----------
def main():
    api_key = os.getenv("OPENAI_API_KEY")
    if not api_key:
        raise SystemExit("Error: Set the OPENAI_API_KEY environment variable.")

    client = openai.OpenAI(api_key=api_key)
    os.makedirs(OUTPUT_DIR, exist_ok=True)

    records = []
    charity_names = []
    failures = 0

    print(f"Starting {N_TRIALS} trials with model={MODEL} (json_mode={USE_JSON_MODE})...")

    for i in range(1, N_TRIALS + 1):
        print(f"  Trial {i}/{N_TRIALS} ...", end=" ", flush=True)
        try:
            raw = call_model(client, SYSTEM_PROMPT, USER_PROMPT)
            data = unwrap_json(raw)
        except Exception as exc:
            print(f"FAILED ({exc})")
            raw = None
            data = None
        else:
            print("OK")

        if isinstance(data, dict) and data.get("charity_name"):
            records.append({
                "trial": i,
                "timestamp": datetime.now(timezone.utc).isoformat(),
                "model": MODEL,
                "raw_response": raw,
                "parsed": data,
            })
            charity_names.append(data["charity_name"])
        else:
            failures += 1
            records.append({
                "trial": i,
                "timestamp": datetime.now(timezone.utc).isoformat(),
                "model": MODEL,
                "raw_response": raw,
                "parsed": data,
                "error": "Could not extract charity_name",
            })

        # Gentle pacing to avoid rate-limit friction
        if i < N_TRIALS:
            time.sleep(0.3)

    # ---- Persist raw results ----
    raw_path = os.path.join(OUTPUT_DIR, "results.jsonl")
    with open(raw_path, "w", encoding="utf-8") as f:
        for r in records:
            f.write(json.dumps(r, ensure_ascii=False) + "\n")

    # ---- Aggregate ----
    raw_counter = Counter(charity_names)
    norm_counter: Counter = Counter()
    norm_to_raw: Dict[str, str] = {}
    for c in charity_names:
        n = normalize_name(c)
        norm_counter[n] += 1
        if n not in norm_to_raw:
            norm_to_raw[n] = c

    # ---- Build summary ----
    lines = [
        f"Model: {MODEL}",
        f"Trials requested: {N_TRIALS}",
        f"Successful parses: {len(charity_names)}",
        f"Failures: {failures}",
        "",
        "Top charities (raw names):",
    ]
    for name, count in raw_counter.most_common(20):
        lines.append(f"  {name}: {count} ({100*count/N_TRIALS:.1f}%)")

    lines.append("")
    lines.append("Top charities (normalized):")
    for norm, count in norm_counter.most_common(20):
        raw_name = norm_to_raw[norm]
        lines.append(f"  {raw_name} (normalized): {count} ({100*count/N_TRIALS:.1f}%)")

    summary_text = "\n".join(lines)
    summary_path = os.path.join(OUTPUT_DIR, "summary.txt")
    with open(summary_path, "w", encoding="utf-8") as f:
        f.write(summary_text)

    # ---- Stdout ----
    print("\n" + "=" * 50)
    print(summary_text)
    print(f"\nFiles saved to ./{OUTPUT_DIR}/")


if __name__ == "__main__":
    main()

### What the script does
1. **Prompts** – Uses your wording (plus a JSON schema constraint) in a system + user message pair.  
2. **Runs 50 trials** – Calls the chat completions API with a non-zero `temperature` so recommendations vary across runs.  
3. **Parses** – If `json_mode` is enabled (default), the model is forced to return valid JSON, so extraction is reliable.  
4. **Aggregates** – Writes:
   * `donation_trials/results.jsonl` – every raw response,
   * `donation_trials/summary.txt` – frequency tables of chosen charities, both raw and normalized (case/punctuation folded).

### Caveats
* **No live browsing** – Unless you switch to a model/tooling that explicitly surfs the web (e.g., GPT-4 with browsing or function-calling), the “research” it does is retrieval from its parametric knowledge. If you want live search, the script would need to be extended with a tool-use loop.
* **Cost** – 50 trials × ~800 tokens ≈ 40k tokens. With GPT-4o that is roughly $0.20–$0.50; budget accordingly.
* **Other APIs** – The `call_model` wrapper can be swapped out for Anthropic, Google, etc., without touching the aggregation logic.

If you want tweaks (e.g., forcing cause-area diversity, adding a secondary “why this charity?” extraction, or running across multiple models), just say the word.
