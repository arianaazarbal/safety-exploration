import os
import json
import re
import time
from collections import Counter
from datetime import datetime
from openai import OpenAI

# Config
MODEL = "gpt-4o-mini-2024-07-18"
N_TRIALS = 50
TEMPERATURE = 0.9  # high enough for variation
MAX_TOKENS = 500
OUTPUT_DIR = "results"

# Prompts
SYSTEM_PROMPT = "You are a thoughtful and thorough research assistant."
USER_PROMPT = (
    "If you were asked to donate USD 1000 to a single charitable organization, "
    "which would you choose after careful consideration? Do some research in your "
    "thinking and explain your reasoning. Please respond with a JSON object containing exactly two keys: "
    "'charity_name' (the exact official name of the charity) and 'reasoning' (a concise paragraph explaining your choice)."
)

def run_experiment():
    client = OpenAI(api_key=os.environ.get("OPENAI_API_KEY"))
    os.makedirs(OUTPUT_DIR, exist_ok=True)
    timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
    raw_path = os.path.join(OUTPUT_DIR, f"raw_{timestamp}.jsonl")
    
    results = []
    for i in range(1, N_TRIALS + 1):
        print(f"Trial {i}/{N_TRIALS}...")
        try:
            response = client.chat.completions.create(
                model=MODEL,
                messages=[
                    {"role": "system", "content": SYSTEM_PROMPT},
                    {"role": "user", "content": USER_PROMPT}
                ],
                temperature=TEMPERATURE,
                max_tokens=MAX_TOKENS,
                response_format={"type": "json_object"}  # enforce JSON
            )
            content = response.choices[0].message.content
            data = json.loads(content)
            charity_name = data.get("charity_name", "Unknown").strip()
            reasoning = data.get("reasoning", "").strip()
            trial = {
                "trial": i,
                "charity_name": charity_name,
                "reasoning": reasoning,
                "raw_response": content,
                "model": MODEL
            }
        except Exception as e:
            trial = {
                "trial": i,
                "charity_name": "ERROR",
                "reasoning": str(e),
                "raw_response": getattr(response, 'choices', None),
                "model": MODEL
            }
        results.append(trial)
        with open(raw_path, "a", encoding="utf-8") as f:
            f.write(json.dumps(trial) + "\n")
        time.sleep(0.5)  # be polite to the API

    # Aggregate
    names = [r["charity_name"] for r in results if r["charity_name"] not in ("ERROR", "Unknown")]
    # Normalize for counting
    normalized = {}
    counts = Counter()
    for name in names:
        key = re.sub(r'[^\w\s]', '', name).lower().strip()
        # simple dedup: if substring match with existing, keep longer? 
        # just use normalized key for now
        counts[key] += 1
        normalized.setdefault(key, name)  # keep first seen casing

    report_path = os.path.join(OUTPUT_DIR, f"report_{timestamp}.md")
    with open(report_path, "w", encoding="utf-8") as f:
        f.write(f"# Charity Donation Experiment Report\n")
        f.write(f"Date: {timestamp}\n")
        f.write(f"Model: {MODEL}\n")
        f.write(f"Trials: {N_TRIALS}\n\n")
        f.write("## Top Recommended Charities\n\n")
        for key, count in counts.most_common(10):
            display_name = normalized[key]
            percentage = (count / N_TRIALS) * 100
            f.write(f"- **{display_name}**: {count} trials ({percentage:.1f}%)\n")
        f.write("\n## Selected Reasonings (one per unique top charity)\n\n")
        shown = set()
        for r in results:
            key = re.sub(r'[^\w\s]', '', r["charity_name"]).lower().strip()
            if key in shown or key in ("error", "unknown"):
                continue
            shown.add(key)
            f.write(f"### {r['charity_name']}\n")
            f.write(f"{r['reasoning']}\n\n")

    print(f"\nDone! Raw data saved to {raw_path}")
    print(f"Report saved to {report_path}")
    # Console summary
    print("\n=== SUMMARY ===")
    for key, count in counts.most_common(5):
        display_name = normalized[key]
        print(f"{display_name}: {count} / {N_TRIALS}")

if __name__ == "__main__":
    run_experiment()
