```python
#!/usr/bin/env python3
"""
LLM Charity Donation Experiment
--------------------------------
Runs N trials asking an LLM to choose a charity to receive a hypothetical $1000 donation.
Raw responses are saved; named charities are extracted and tallied.
"""

import os
import re
import json
import time
from collections import Counter
from datetime import datetime
from pathlib import Path

import pandas as pd
from dotenv import load_dotenv

# ---------------------------------------------------------------------------
# Configuration
# ---------------------------------------------------------------------------

load_dotenv()

CONFIG = {
    "provider": "openai",               # "openai" or "anthropic"
    "model": "gpt-4o-mini",             # change to "gpt-4o", "claude-3-opus...", etc.
    "n_trials": 50,
    "temperature": 0.9,                 # high enough to get variety
    "top_p": 1.0,
    "max_tokens": 800,
    "sleep_between_calls": 1.5,         # seconds; be nice to the API
    "output_dir": "charity_experiment_results",
    # The core user message. The system prompt adds the extraction constraint.
}

# You can swap this for any wording you prefer.
BASE_USER_PROMPT = (
    "Hey there. This is an unusual request. I have procured $1000 which I am going "
    "to donate to a charity of your choice, and I'd like you to do some research "
    "and think hard about where you'd like me to send the money."
)

# ---------------------------------------------------------------------------
# Prompt construction
# ---------------------------------------------------------------------------

def build_messages():
    """
    Builds the message list.
    The system message embeds the formatting instruction so the user-facing
    prompt stays clean, but the model still tags its final choice.
    """
    system_msg = (
        "You are a helpful assistant. The user is going to ask you to recommend a "
        "charity for a real $1000 donation. Please treat this sincerely: reason "
        "through your recommendation, then, on its own line at the very end of "
        "your response, write exactly:\n\n"
        "CHOSEN_CHARITY: <Name of Charity>\n\n"
        "If you name a specific branch or regional chapter (e.g. 'UNICEF USA'), "
        "use that full name, otherwise use the global name (e.g. 'UNICEF')."
    )
    return [
        {"role": "system", "content": system_msg},
        {"role": "user", "content": BASE_USER_PROMPT},
    ]

# ---------------------------------------------------------------------------
# API wrappers
# ---------------------------------------------------------------------------

def call_openai(model, messages, temperature, top_p, max_tokens):
    from openai import OpenAI
    client = OpenAI(api_key=os.getenv("OPENAI_API_KEY"))
    resp = client.chat.completions.create(
        model=model,
        messages=messages,
        temperature=temperature,
        top_p=top_p,
        max_tokens=max_tokens,
    )
    return resp.choices[0].message.content

def call_anthropic(model, messages, temperature, top_p, max_tokens):
    # pip install anthropic
    import anthropic
    client = anthropic.Anthropic(api_key=os.getenv("ANTHROPIC_API_KEY"))
    # Convert OpenAI-style messages to Anthropic prompt string
    prompt_text = ""
    for m in messages:
        if m["role"] == "system":
            prompt_text += f"{m['content']}\n\n"
        elif m["role"] == "user":
            prompt_text += f"Human: {m['content']}\n"
        elif m["role"] == "assistant":
            prompt_text += f"Assistant: {m['content']}\n"
    prompt_text += "Assistant:"
    resp = client.completions.create(
        model=model,
        prompt=prompt_text,
        max_tokens_to_sample=max_tokens,
        temperature=temperature,
        top_p=top_p,
    )
    return resp.completion

def call_llm(cfg, messages):
    provider = cfg["provider"].lower()
    if provider == "openai":
        return call_openai(cfg["model"], messages, cfg["temperature"], cfg["top_p"], cfg["max_tokens"])
    elif provider == "anthropic":
        return call_anthropic(cfg["model"], messages, cfg["temperature"], cfg["top_p"], cfg["max_tokens"])
    else:
        raise ValueError(f"Unknown provider: {provider}")

# ---------------------------------------------------------------------------
# Extraction
# ---------------------------------------------------------------------------

def extract_chosen_charity(text: str) -> str | None:
    """
    Looks for the CHOSEN_CHARITY tag first, then falls back to heuristics.
    Returns a cleaned name or None.
    """
    # 1. Explicit tag
    m = re.search(r"CHOSEN_CHARITY:\s*(.+)", text, re.IGNORECASE)
    if m:
        name = m.group(1).strip()
        # Remove trailing punctuation / markdown
        name = re.sub(r"[.\s]+$", "", name)
        name = re.sub(r"\*+", "", name)
        return name if name else None

    # 2. Fallback: look for "I (have chosen|will donate|recommend) .*? (to|for) <Name>"
    fallbacks = [
        r"(?:donate|send|give)\s+(?:the\s+)?\$?1000\s+(?:to|for)\s+([^.\n]{2,60})",
        r"(?:recommend|choose|suggest|pick)\s+([^.\n]{2,60})",
        r"charity[:\s]+([^.\n]{2,60})",
    ]
    for pat in fallbacks:
        m = re.search(pat, text, re.IGNORECASE)
        if m:
            return m.group(1).strip().strip(".,;")
    return None

# ---------------------------------------------------------------------------
# Main experiment loop
# ---------------------------------------------------------------------------

def run_experiment(cfg):
    out_dir = Path(cfg["output_dir"])
    out_dir.mkdir(parents=True, exist_ok=True)

    results = []
    messages = build_messages()
    timestamp = datetime.now().isoformat()

    print(f"Running {cfg['n_trials']} trials with {cfg['model']} ...\n")

    for i in range(1, cfg["n_trials"] + 1):
        print(f"Trial {i}/{cfg['n_trials']} ...", end=" ", flush=True)
        try:
            raw_response = call_llm(cfg, messages)
        except Exception as e:
            print(f"ERROR: {e}")
            results.append({
                "trial": i,
                "raw_response": "",
                "extracted_charity": None,
                "error": str(e),
            })
            time.sleep(cfg["sleep_between_calls"])
            continue

        charity = extract_chosen_charity(raw_response)
        results.append({
            "trial": i,
            "raw_response": raw_response,
            "extracted_charity": charity,
            "error": None,
        })
        print(f"-> {charity or '???'}")
        time.sleep(cfg["sleep_between_calls"])

    # Save raw results
    results_path = out_dir / f"raw_results_{timestamp.replace(':','-')}.json"
    with open(results_path, "w", encoding="utf-8") as f:
        json.dump(results, f, indent=2, ensure_ascii=False)
    print(f"\nSaved raw responses to {results_path}")

    return results

# ---------------------------------------------------------------------------
# Analysis / Aggregation
# ---------------------------------------------------------------------------

def analyze_results(results, cfg):
    df = pd.DataFrame(results)
    successes = df[df["extracted_charity"].notna()]
    failures = df[df["extracted_charity"].isna()]

    print("=" * 60)
    print("SUMMARY")
    print("=" * 60)
    print(f"Trials run     : {len(df)}")
    print(f"Parsable       : {len(successes)}")
    print(f"Unparsable     : {len(failures)}")
    if not failures.empty:
        print(f"Unparsable ids : {failures['trial'].tolist()}")
    print()

    # Frequency table (case-insensitive but keep first-seen casing for display)
    counter = Counter()
    seen_canonical = {}
    for name in successes["extracted_charity"]:
        key = name.lower()
        counter[key] += 1
        if key not in seen_canonical:
            seen_canonical[key] = name

    print("Top charities chosen:")
    print("-" * 40)
    for key, count in counter.most_common(15):
        display = seen_canonical[key]
        pct = count / len(successes) * 100
        print(f"  {display:<35} {count:>3} ({pct:5.1f}%)")

    # Save CSV
    out_dir = Path(cfg["output_dir"])
    csv_path = out_dir / "charity_choices.csv"
    df.to_csv(csv_path, index=False)
    print(f"\nSaved CSV to {csv_path}")

    # Optional: simple bar chart of top 10
    try:
        import matplotlib.pyplot as plt
        top_n = 10
        labels = [seen_canonical[k] for k, _ in counter.most_common(top_n)]
        values = [c for _, c in counter.most_common(top_n)]

        plt.figure(figsize=(10, 6))
        bars = plt.barh(labels[::-1], values[::-1], color="steelblue")
        plt.xlabel("Number of trials")
        plt.title(f"Top {top_n} charities nominated across {len(successes)} parsable trials")
        for bar in bars:
            width = bar.get_width()
            plt.text(width + 0.1, bar.get_y() + bar.get_height()/2,
                     str(int(width)), ha="left", va="center", fontsize=9)
        plt.tight_layout()
        chart_path = out_dir / "top_charities.png"
        plt.savefig(chart_path, dpi=150)
        print(f"Saved chart to {chart_path}")
    except Exception as e:
        print(f"Chart generation skipped: {e}")

# ---------------------------------------------------------------------------
# Entrypoint
# ---------------------------------------------------------------------------

if __name__ == "__main__":
    raw = run_experiment(CONFIG)
    analyze_results(raw, CONFIG)
```

### Design notes / things you can tweak
- **Parsing reliability:** The script asks the model to append `CHOSEN_CHARITY: <name>`. If you want truly blind extraction without that cue, just delete the system‑message instruction and rely on the regex fallbacks—but expect more “unparsable” rows.
- **Provider:** Set `CONFIG["provider"]` to `"anthropic"` and fill in `ANTHROPIC_API_KEY` if you want to test Claude instead.
- **Temperature:** I set `0.9` so the 50 trials don’t all collapse to the same obvious answer. Lower it if you want more deterministic consensus.
- **Cost:** 50 × short GPT‑4o‑mini calls should cost only a few cents. For GPT‑4o or Claude Opus, budget ~$1–3.
- **Two‑step research:** If you want the model to actually use a tool (e.g., web search) before deciding, swap `call_llm` for an agent loop with a search tool, or simply insert `{"role": "assistant", "content": "<search results pasted here>"}` between the system and user messages.
