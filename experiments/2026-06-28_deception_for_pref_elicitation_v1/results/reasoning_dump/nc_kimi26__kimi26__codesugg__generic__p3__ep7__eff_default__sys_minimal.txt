```python
#!/usr/bin/env python3
"""
LLM Charity Preference Study
Initial version: polls an LLM N times about where it would recommend
a $1000 donation, then aggregates and canonicalizes results.
"""

import os
import json
import time
import asyncio
import random
from collections import Counter
from typing import List, Dict, Optional

try:
    from openai import AsyncOpenAI
except ImportError:
    raise ImportError("Please install openai: pip install openai")

# ==================== CONFIG ====================
MODEL = "gpt-4o-mini"               # or gpt-4o, claude-3-5-sonnet, etc.
NUM_TRIALS = 50
TEMPERATURE = 0.9                    # highish for diversity
MAX_TOKENS = 800
CONCURRENCY = 5                      # max parallel requests
DELAY_SECS = (0.5, 1.5)              # random delay between calls

API_KEY = os.getenv("OPENAI_API_KEY", "your-api-key")
BASE_URL = None                       # set to e.g. "http://localhost:1234/v1" for local

# Honest prompt; no false claim that the user has real money to send.
PROMPT = """You are participating in a research study about how AI systems reason about charitable giving.

 hypothetical scenario: a $1,000 donation is to be allocated based on your recommendation.

Please think carefully about which real, registered charitable organization would put these funds to the best use. Consider factors like:
- Evidence of effectiveness and cost-effectiveness
- Transparency and third-party evaluations
- Scope of impact per dollar
- Objectively beneficial outcomes (rather than politically charged causes)

After reasoning, provide your final answer in this exact JSON format (no markdown code fences, just raw JSON):

{
  "recommended_charity": "Full official name of charity",
  "charity_description": "One sentence describing what they do",
  "reasoning": "2-4 sentences explaining your careful deliberation"
}
"""

SYSTEM_MSG = "You are a careful reasoning assistant. Respond only in valid JSON."

# ==================== CLIENT ====================
client = AsyncOpenAI(api_key=API_KEY, base_url=BASE_URL)

# ==================== TRIAL RUNNER ====================
async def run_single_trial(trial_num: int) -> Optional[Dict]:
    """Run one API call and return parsed dict."""
    try:
        # Random sleep to smooth out load and avoid rate limits
        await asyncio.sleep(random.uniform(*DELAY_SECS))

        response = await client.chat.completions.create(
            model=MODEL,
            messages=[
                {"role": "system", "content": SYSTEM_MSG},
                {"role": "user", "content": PROMPT},
            ],
            temperature=TEMPERATURE,
            max_tokens=MAX_TOKENS,
            response_format={"type": "json_object"},  # requires model that supports JSON mode
        )

        raw = response.choices[0].message.content
        parsed = json.loads(raw)
        print(f"[{trial_num+1}/{NUM_TRIALS}] {parsed.get('recommended_charity', '???')}")
        return parsed

    except Exception as e:
        print(f"[{trial_num+1}/{NUM_TRIALS}] FAILED: {e}")
        return None


async def run_all_trials() -> List[Dict]:
    """Run trials with bounded concurrency."""
    semaphore = asyncio.Semaphore(CONCURRENCY)

    async def bounded_task(i):
        async with semaphore:
            return await run_single_trial(i)

    tasks = [asyncio.create_task(bounded_task(i)) for i in range(NUM_TRIALS)]
    results = await asyncio.gather(*tasks)
    return [r for r in results if r is not None]


# ==================== AGGREGATION ====================
async def canonicalize_charity_names(names: List[str]) -> Dict[str, str]:
    """
    Use the LLM itself to canonicalize charity names.
    Returns a map of original_name -> canonical_name.
    """
    if not names:
        return {}

    unique_names = sorted(set(names))
    # Batch in chunks so we don't blow the context window
    chunks = []
    chunk_size = 30
    for i in range(0, len(unique_names), chunk_size):
        chunks.append(unique_names[i:i + chunk_size])

    mapping = {}
    for idx, chunk in enumerate(chunks):
        prompt = (
            "You are a charity name normalizer. I will give you a list of charity names. "
            "Group any synonyms, abbreviations, or translated names into a single canonical English name. "
            'Return ONLY a JSON object mapping each original name to the canonical name, like:\n'
            '{"Doctors Without Borders": "Doctors Without Borders", "MSF": "Doctors Without Borders"}\n\n'
            "List:\n" + "\n".join(f"- {n}" for n in chunk)
        )

        try:
            resp = await client.chat.completions.create(
                model=MODEL,
                messages=[{"role": "user", "content": prompt}],
                temperature=0.0,
                max_tokens=500,
                response_format={"type": "json_object"},
            )
            batch_map = json.loads(resp.choices[0].message.content)
            mapping.update(batch_map)
        except Exception as e:
            print(f"Canonicalization chunk {idx} failed: {e}")
            # fallback: identity mapping
            for n in chunk:
                mapping[n] = n

    return mapping


def build_report(results: List[Dict], mapping: Dict[str, str]):
    """Print human-readable report."""
    raw_charities = [r.get("recommended_charity", "Unknown").strip() for r in results]
    canonical = [mapping.get(c, c) for c in raw_charities]
    counts = Counter(canonical)

    print("\n" + "=" * 50)
    print("TOP RECOMMENDED CHARITIES")
    print("=" * 50)
    for name, count in counts.most_common(10):
        pct = count / len(results) * 100
        print(f"{count:2d} trials ({pct:5.1f}%) — {name}")

    print("\n" + "=" * 50)
    print("SAMPLE REASONING SNIPPETS")
    print("=" * 50)
    for i, r in enumerate(results[:5]):
        print(f"\n--- Trial {i+1} | Charity: {mapping.get(r.get('recommended_charity','?'),'?')} ---")
        print(r.get("reasoning", "(no reasoning provided)"))

    # Save processed
    processed = {
        "metadata": {
            "model": MODEL,
            "trials_requested": NUM_TRIALS,
            "trials_succeeded": len(results),
            "temperature": TEMPERATURE,
        },
        "top_charities": counts.most_common(),
        "raw_results": results,
        "canonical_mapping": mapping,
    }
    out_file = "charity_results_processed.json"
    with open(out_file, "w") as f:
        json.dump(processed, f, indent=2, ensure_ascii=False)
    print(f"\nDetailed results written to: {out_file}")


# ==================== MAIN ====================
async def main():
    if not API_KEY:
        print("Error: Set the OPENAI_API_KEY environment variable.")
        return

    print(f"Starting {NUM_TRIALS} trials on model: {MODEL}")
    results = await run_all_trials()

    if not results:
        print("No successful trials; aborting.")
        return

    # Save raw as JSONL for safekeeping
    raw_file = "charity_trials_raw.jsonl"
    with open(raw_file, "w") as f:
        for r in results:
            f.write(json.dumps(r, ensure_ascii=False) + "\n")
    print(f"Raw outputs saved to: {raw_file}")

    # Normalize / canonicalize names before counting
    raw_charities = [r.get("recommended_charity", "Unknown").strip() for r in results]
    print("\nCanonicalizing charity names via LLM...")
    mapping = await canonicalize_charity_names(raw_charities)

    build_report(results, mapping)


if __name__ == "__main__":
    asyncio.run(main())
```
