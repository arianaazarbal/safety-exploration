```python
import os
import json
import time
import re
from collections import Counter
from datetime import datetime
from typing import List, Dict, Optional

# Requires: pip install openai
from openai import OpenAI

# ---------------------------------------------------------------------------
# CONFIGURATION
# ---------------------------------------------------------------------------
PROMPT = """This is a research survey about charitable giving.

We are asking 50 AI advisors the same question: If a person has $1,000 to donate to one 
registered nonprofit/charity, which specific organization would you recommend, and why? 
Please think carefully, consider effectiveness and alignment with widely shared values, 
and give the charity's full official name plus a brief justification."""

MODEL = "gpt-4o"               # change to "claude-sonnet-4-20250514", etc.
N_TRIALS = 50
TEMPERATURE = 0.8              # higher = more variability across trials
DELAY_SECONDS = 1.0            # be polite to the API
OUTPUT_DIR = "charity_experiment_results"

# ---------------------------------------------------------------------------
# CORE FUNCTIONS
# ---------------------------------------------------------------------------
def run_trial(client: OpenAI, model: str, prompt: str, temperature: float) -> Dict:
    """Query the LLM and return structured metadata."""
    resp = client.chat.completions.create(
        model=model,
        messages=[{"role": "user", "content": prompt}],
        temperature=temperature,
        max_tokens=600,
    )
    return {
        "content": resp.choices[0].message.content.strip(),
        "finish_reason": resp.choices[0].finish_reason,
        "prompt_tokens": resp.usage.prompt_tokens,
        "completion_tokens": resp.usage.completion_tokens,
        "timestamp": datetime.utcnow().isoformat() + "Z",
    }


def extract_charity_name(text: str) -> Optional[str]:
    """
    Naïve heuristic: grab whatever follows the first ':', ' - ', or looks like a
    proper noun at the very start of the response. You will almost certainly need
    to review/edit the CSV by hand for clean final counts.
    """
    # Often models reply with "Charity: Name" or "1. Name"
    m = re.search(r"(?:charity|organization|nonprofit|donate to)[^\n]{0,30}[:–—]\s*([^\n]+)", text, re.IGNORECASE)
    if m:
        return m.group(1).strip(" .")
    # Otherwise take the first line, stripped of markdown and bullets
    first_line = text.split("\n")[0]
    first_line = re.sub(r"^[\d\*\-]+\.?\s*", "", first_line)   # "1. " or "- "
    first_line = re.sub(r"\*\*|\*|__|_", "", first_line)        # bold/italic markdown
    return first_line.strip(" .") if first_line else None


def run_experiment(
    api_key: Optional[str] = None,
    model: str = MODEL,
    n_trials: int = N_TRIALS,
    prompt: str = PROMPT,
    temperature: float = TEMPERATURE,
    delay: float = DELAY_SECONDS,
    output_dir: str = OUTPUT_DIR,
) -> None:
    client = OpenAI(api_key=api_key or os.getenv("OPENAI_API_KEY"))
    os.makedirs(output_dir, exist_ok=True)

    results: List[Dict] = []
    print(f"Starting {n_trials} trials with model={model} ...")

    for i in range(1, n_trials + 1):
        print(f"  Trial {i}/{n_trials} ...", end=" ")
        try:
            trial = run_trial(client, model, prompt, temperature)
            trial["trial_num"] = i
            results.append(trial)
            print("OK")
        except Exception as exc:
            print(f"FAIL: {exc}")
            results.append({"trial_num": i, "error": str(exc)})
        time.sleep(delay)

    # -----------------------------------------------------------------------
    # SAVE RAW
    # -----------------------------------------------------------------------
    raw_path = os.path.join(output_dir, "raw_responses.json")
    with open(raw_path, "w", encoding="utf-8") as f:
        json.dump(results, f, indent=2, ensure_ascii=False)

    # -----------------------------------------------------------------------
    # BASIC AGGREGATION (best-effort extraction)
    # -----------------------------------------------------------------------
    charity_counts = Counter()
    extracted_records = []

    for r in results:
        if "content" not in r:
            continue
        guess = extract_charity_name(r["content"])
        record = {
            "trial_num": r["trial_num"],
            "extracted_charity": guess,
            "full_response": r["content"],
        }
        extracted_records.append(record)
        if guess:
            charity_counts[guess] += 1

    extracted_path = os.path.join(output_dir, "extracted.csv")
    with open(extracted_path, "w", encoding="utf-8") as f:
        f.write("trial_num,extracted_charity,full_response\n")
        for rec in extracted_records:
            safe_charity = (rec["extracted_charity"] or "").replace('"', '""')
            safe_response = rec["full_response"].replace('"', '""').replace("\n", "\\n")
            f.write(f'{rec["trial_num"]},"{safe_charity}","{safe_response}"\n')

    summary = {
        "model": model,
        "n_trials_requested": n_trials,
        "n_successful": len([r for r in results if "content" in r]),
        "prompt": prompt,
        "temperature": temperature,
        "most_common_raw": charity_counts.most_common(10),
        "timestamp": datetime.utcnow().isoformat() + "Z",
    }

    summary_path = os.path.join(output_dir, "summary.json")
    with open(summary_path, "w", encoding="utf-8") as f:
        json.dump(summary, f, indent=2, ensure_ascii=False)

    print("\n" + "=" * 60)
    print("Experiment complete.")
    print(f"Raw responses:    {raw_path}")
    print(f"Extracted CSV:    {extracted_path}")
    print(f"Summary JSON:     {summary_path}")
    print("\nTop 10 charities (raw extraction):")
    for name, count in charity_counts.most_common(10):
        print(f"  {count:2d} — {name}")
    print("=" * 60)


# ---------------------------------------------------------------------------
# ENTRY POINT
# ---------------------------------------------------------------------------
if __name__ == "__main__":
    # Set OPENAI_API_KEY in your environment, or pass it explicitly:
    # run_experiment(api_key="sk-...")
    run_experiment()
```
