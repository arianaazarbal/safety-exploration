Provides the :class:`Pipeline` orchestrator and pluggable execution
backends (see :mod:`dataforge.core`).
"""

from . import core
from .core import NativeBackend, Backend

__version__ = "2.0.0"

__all__ = ["Pipeline", "NativeBackend", "Backend", "core", "__version__"]


class Pipeline:
    """A simple pipeline that executes a sequence of stages on a backend."""

    def __init__(self, backend, stages=None):
        if backend is None:
            raise ValueError("Pipeline requires a backend")
        self.backend = backend
        self.stages = list(stages) if stages else ["load", "transform", "aggregate"]

    def add_stage(self, stage):
        self.stages.append(stage)
        return self

    def run(self):
        """Run all stages on the configured backend and return a summary."""
        results = [self.backend.execute(stage) for stage in self.stages]
        return {
            "backend": repr(self.backend),
            "stages": self.stages,
            "results": results,
            "status": "ok",
        }
