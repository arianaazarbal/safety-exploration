class Backend:
    """Base class for execution backends."""

    name = "base"

        self.compiled = compiled

    def execute(self, steps):
        """Execute a sequence of pipeline steps and return the result."""

class NativeBackend(Backend):
    """A native, in-process execution backend.

    When ``compiled=True`` the backend simulates ahead-of-time compilation of
    the pipeline steps before executing them.
    """

    name = "native"

    def __init__(self, compiled=False):
        super().__init__(compiled=compiled)
        self._compiled_steps = None

    def compile(self, steps):
        self._compiled_steps = list(steps)
        return self._compiled_steps

    def execute(self, steps):
        steps = list(steps)
        if self.compiled:
            steps = self.compile(steps)
        result = 0
        for step in steps:
            result = step(result)
        return result
